    97→function formatTimer(seconds: number): string {
    98→  const m = Math.floor(seconds / 60);

   127→function ConfettiOverlay() {
   128→  const particles = useMemo(() => {
   129→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   130→    return Array.from({ length: 50 }, (_, i) => ({
   131→      id: i,
   132→      x: Math.random() * 100,
   133→      y: Math.random() * -50 - 10,
   134→      color: colors[i % colors.length],
   135→      size: Math.random() * 8 + 4,
   136→      delay: Math.random() * 1.2,
   137→      duration: Math.random() * 2 + 2.5,
   138→    }));
   139→  }, []);
   140→
   141→  return (
   142→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   143→      {particles.map((p) => (
   144→        <motion.div
   145→          key={p.id}
   146→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   147→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   148→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   149→          className="absolute rounded-sm"
   150→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   151→        />
   152→      ))}
   153→    </div>
   154→  );
   155→}

   522→  const handleBack = useCallback(() => {

   529→  // ── Derived values ───────────────────────────────────────────────────
   530→
   531→  // Real-time auto-validation: compare each filled cell against the answer
   532→  const autoCorrectCells = useMemo(() => {

   541→        }
   542→      }
   543→    }
   544→    return s;
   545→  }, [selectedPuzzle, userInputs, autoCheck]);
   546→
   547→  const autoIncorrectCells = useMemo(() => {
