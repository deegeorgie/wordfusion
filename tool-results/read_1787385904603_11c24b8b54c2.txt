   161→function StreakHeatmap({ streakData }: { streakData: StreakCompletion[] }) {
   162→  const days = useMemo(() => {
   163→    const today = new Date();
   164→    const result: { date: string; completed: boolean; count: number }[] = [];
   165→    for (let i = 29; i >= 0; i--) {
   166→      const d = new Date(today);
   167→      d.setDate(d.getDate() - i);
   168→      const dateStr = d.toISOString().slice(0, 10);
   169→      const completions = streakData.filter((c) => c.date === dateStr);
   170→      result.push({
   171→        date: dateStr,
   172→        completed: completions.length > 0,
   173→        count: completions.length,
   174→      });
   175→    }
   176→    return result;
   177→  }, [streakData]);
   178→
   179→  if (streakData.length === 0) return null;
   180→
   181→  return (
   182→    <div className="flex flex-wrap gap-1">
   183→      {days.map((day) => (
   184→        <div
   185→          key={day.date}
   186→          title={`${day.date}${day.completed ? ` — ${day.count} puzzle${day.count > 1 ? 's' : ''}` : ''}`}
   187→          className={cn(
   188→            'h-5 w-5 rounded-sm transition-colors',
   189→            day.completed
   190→              ? 'bg-emerald-500 dark:bg-emerald-400'
   191→              : day.date <= new Date().toISOString().slice(0, 10)
   192→                ? 'bg-muted'
   193→                : 'bg-transparent',
   194→          )}
   195→        />
   196→      ))}
   197→    </div>
   198→  );
   199→}

   203→function ConfettiOverlay() {
   204→  const particles = useMemo(() => {
   205→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   206→    return Array.from({ length: 50 }, (_, i) => ({
   207→      id: i,
   208→      x: Math.random() * 100,
   209→      y: Math.random() * -50 - 10,
   210→      color: colors[i % colors.length],
   211→      size: Math.random() * 8 + 4,
   212→      delay: Math.random() * 1.2,
   213→      duration: Math.random() * 2 + 2.5,
   214→    }));
   215→  }, []);
   216→
   217→  return (
   218→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   219→      {particles.map((p) => (
   220→        <motion.div
   221→          key={p.id}
   222→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   223→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   224→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   225→          className="absolute rounded-sm"
   226→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   227→        />
   228→      ))}
   229→    </div>
   230→  );
   231→}

   491→        if (w.direction !== dir) return false;
   492→        return dir === 'across'
   493→          ? w.row === row && col >= w.col && col < w.col + w.length
   494→          : w.col === col && row >= w.row && row < w.row + w.length;
   495→      });
   496→
   497→      if (!found) {
   498→        found = selectedPuzzle.words.find((w: WordPlacement) => {
   499→          if (w.direction !== otherDir) return false;
   500→          return otherDir === 'across'
   501→            ? w.row === row && col >= w.col && col < w.col + w.length
   502→            : w.col === col && row >= w.row && row < w.row + w.length;
   503→        });
   504→        if (found) {
   505→          setDirection(otherDir);
   506→          setActiveClueDirection(otherDir);
   507→        }
   508→      } else {
   509→        setActiveClueDirection(dir);
   510→      }

   512→      if (found) setActiveClueNumber(found.clueNumber);
   513→    },
   514→    [selectedPuzzle],

   532→      }
