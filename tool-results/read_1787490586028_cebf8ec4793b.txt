   159→   159→// ── Streak Heatmap ──────────────────────────────────────────────
   160→   160→
   161→   161→function StreakHeatmap({ streakData }: { streakData: StreakCompletion[] }) {
   162→   162→  const days = useMemo(() => {
   163→   163→    const today = new Date();
   164→   164→    const result: { date: string; completed: boolean; count: number }[] = [];
   165→   165→    for (let i = 29; i >= 0; i--) {
   166→   166→      const d = new Date(today);
   167→   167→      d.setDate(d.getDate() - i);
   168→   168→      const dateStr = d.toISOString().slice(0, 10);
   169→   169→      const completions = streakData.filter((c) => c.date === dateStr);
   170→   170→      result.push({
   171→   171→        date: dateStr,
   172→   172→        completed: completions.length > 0,
   173→   173→        count: completions.length,
   174→   174→      });
   175→   175→    }
   176→   176→    return result;
   177→   177→  }, [streakData]);
   178→   178→
   179→   179→  if (streakData.length === 0) return null;
   180→   180→
   181→   181→  return (
   182→   182→    <div className="flex flex-wrap gap-1">
   183→   183→      {days.map((day) => (
   184→   184→        <div
   185→   185→          key={day.date}
   186→   186→          title={`${day.date}${day.completed ? ` — ${day.count} puzzle${day.count > 1 ? 's' : ''}` : ''}`}
   187→   187→          className={cn(
   188→   188→            'h-5 w-5 rounded-sm transition-colors',
   189→   189→            day.completed
   190→   190→              ? 'bg-emerald-500 dark:bg-emerald-400'
   191→   191→              : day.date <= new Date().toISOString().slice(0, 10)
   192→   192→                ? 'bg-muted'
   193→   193→                : 'bg-transparent',
   194→   194→          )}
   195→   195→        />
   196→   196→      ))}
   197→   197→    </div>
   198→   198→  );
   199→   199→}

   205→   205→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];

   208→   208→      x: Math.random() * 100,
   209→   209→      y: Math.random() * -50 - 10,
   210→   210→      color: colors[i % colors.length],
   211→   211→      size: Math.random() * 8 + 4,
   212→   212→      delay: Math.random() * 1.2,
   213→   213→      duration: Math.random() * 2 + 2.5,

   215→   215→  }, []);
   216→   216→
   217→   217→  return (
   218→   218→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   219→   219→      {particles.map((p) => (
   220→   220→        <motion.div
   221→   221→          key={p.id}
   222→   222→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   223→   223→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   224→   224→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   225→   225→          className="absolute rounded-sm"
   226→   226→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   227→   227→        />

   229→   229→    </div>
   230→   230→  );
   231→   231→}
