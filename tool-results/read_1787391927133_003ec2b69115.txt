   214→   214→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {

   216→   216→  const [newIcon, setNewIcon] = useState('🏷️');
   217→   217→  const [newLanguage, setNewLanguage] = useState('fr');
   218→   218→  const [creating, setCreating] = useState(false);
   219→   219→
   220→   220→  // Edit state
   221→   221→  const [editingId, setEditingId] = useState<string | null>(null);
   222→   222→  const [editName, setEditName] = useState('');
   223→   223→  const [editIcon, setEditIcon] = useState('');
   224→   224→  const [editLanguage, setEditLanguage] = useState('');
   225→   225→  const [saving, setSaving] = useState(false);
   226→   226→
   227→   227→  // Delete state
   228→   228→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   229→   229→  const [deleting, setDeleting] = useState(false);
   230→   230→
   231→   231→  const handleCreate = async () => {
   232→   232→    if (!newName.trim()) return;
   233→   233→    setCreating(true);
   234→   234→    try {
   235→   235→      const res = await fetch('/api/categories', {
   236→   236→        method: 'POST',
   237→   237→        headers: { 'Content-Type': 'application/json' },
   238→   238→        body: JSON.stringify({ name: newName.trim(), icon: newIcon, language: newLanguage }),
   239→   239→      });
   240→   240→      if (!res.ok) {
   241→   241→        const err = await res.json().catch(() => ({}));
   242→   242→        throw new Error((err as { error?: string }).error || 'Erreur');
   243→   243→      }
   244→   244→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   245→   245→      setNewName('');
   246→   246→      setNewIcon('🏷️');
   247→   247→      onRefresh();
   248→   248→    } catch (err) {
   249→   249→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   250→   250→    } finally {
   251→   251→      setCreating(false);
   252→   252→    }
   253→   253→  };
   254→   254→
   255→   255→  const startEdit = (cat: CategoryItem) => {

   257→   257→    setEditName(cat.name);
   258→   258→    setEditIcon(cat.icon || '🏷️');
   259→   259→    setEditLanguage(cat.language);
   260→   260→  };
   261→   261→
   262→   262→  const handleSaveEdit = async () => {
   263→   263→    if (!editingId || !editName.trim()) return;
   264→   264→    setSaving(true);
   265→   265→    try {
   266→   266→      const res = await fetch('/api/categories', {
   267→   267→        method: 'PUT',
   268→   268→        headers: { 'Content-Type': 'application/json' },
   269→   269→        body: JSON.stringify({ id: editingId, name: editName.trim(), icon: editIcon, language: editLanguage }),
   270→   270→      });
   271→   271→      if (!res.ok) {
   272→   272→        const err = await res.json().catch(() => ({}));
   273→   273→        throw new Error((err as { error?: string }).error || 'Erreur');
   274→   274→      }
   275→   275→      toast.success('Catégorie mise à jour');
   276→   276→      setEditingId(null);
   277→   277→      onRefresh();
   278→   278→    } catch (err) {
   279→   279→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   280→   280→    } finally {
   281→   281→      setSaving(false);
   282→   282→    }
   283→   283→  };
   284→   284→
   285→   285→  const handleDelete = async () => {
   286→   286→    if (!deleteTarget) return;
   287→   287→    setDeleting(true);
   288→   288→    try {
   289→   289→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   290→   290→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   291→   291→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   292→   292→      setDeleteTarget(null);
   293→   293→      onRefresh();
   294→   294→    } catch (err) {
   295→   295→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   296→   296→    } finally {
   297→   297→      setDeleting(false);
   298→   298→    }
   299→   299→  };
   300→   300→
   301→   301→  return (
   302→   302→    <>
   303→   303→      {/* Create new category row */}
   304→   304→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   305→   305→        <div className="w-16 sm:w-20 space-y-1">
   306→   306→          <Label className="text-xs text-muted-foreground">Icône</Label>
   307→   307→          <Input
   308→   308→            placeholder="🏷️"
   309→   309→            value={newIcon}
   310→   310→            onChange={(e) => setNewIcon(e.target.value)}
   311→   311→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   312→   312→            className="h-8 text-sm text-center"
   313→   313→            maxLength={4}
   314→   314→          />
   315→   315→        </div>
   316→   316→        <div className="flex-1 w-full sm:w-auto space-y-1">
   317→   317→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   318→   318→          <Input
   319→   319→            placeholder="Ex: Animaux, Sciences..."
   320→   320→            value={newName}
   321→   321→            onChange={(e) => setNewName(e.target.value)}
   322→   322→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   323→   323→            className="h-8 text-sm"
   324→   324→          />
   325→   325→        </div>
   326→   326→        <div className="w-full sm:w-28 space-y-1">
   327→   327→          <Label className="text-xs text-muted-foreground">Langue</Label>
   328→   328→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   329→   329→            <SelectTrigger className="h-8 text-sm">
   330→   330→              <SelectValue />

   338→   338→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   339→   339→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   340→   340→          Ajouter

   343→   343→
   344→   344→      {/* Category list */}

   359→   359→            </TableHeader>
   360→   360→            <TableBody>
   361→   361→              {categories.map((cat) => (
   362→   362→                <TableRow key={cat.id}>
   363→   363→                  <TableCell className="font-medium pl-3">
   364→   364→                    {editingId === cat.id ? (
   365→   365→                      <div className="flex items-center gap-1.5">
   366→   366→                        <Input
   367→   367→                          value={editIcon}
   368→   368→                          onChange={(e) => setEditIcon(e.target.value)}
   369→   369→                          className="h-7 text-sm w-12 text-center"

   373→   373→                        <Input
