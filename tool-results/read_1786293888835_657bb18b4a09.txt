    64→function formatFrenchDate(): string {
    65→  return new Date().toLocaleDateString('fr-FR', {
    66→    weekday: 'long',

    73→function formatTimer(seconds: number): string {
    74→  const m = Math.floor(seconds / 60);
    75→  const s = seconds % 60;
    76→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    77→}

    79→function difficultyLabel(d: number): string {

    83→  return 'Expert';

   113→function ConfettiOverlay() {
   114→  const particles = useMemo(() => {
   115→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   116→    return Array.from({ length: 50 }, (_, i) => ({
   117→      id: i,
   118→      x: Math.random() * 100,
   119→      y: Math.random() * -50 - 10,
   120→      color: colors[i % colors.length],
   121→      size: Math.random() * 8 + 4,
   122→      delay: Math.random() * 1.2,
   123→      duration: Math.random() * 2 + 2.5,
   124→    }));
   125→  }, []);
   126→
   127→  return (
   128→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   129→      {particles.map((p) => (
   130→        <motion.div
   131→          key={p.id}
   132→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   133→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   134→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   135→          className="absolute rounded-sm"
   136→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   137→        />
   138→      ))}
   139→    </div>
   140→  );
   141→}

   166→export default function Home() {
   167→  // ── View state ──────────────────────────────────────────────────────
   168→  const [currentView, setCurrentView] = useState<'selection' | 'playing'>('selection');
   169→  const [dailyPuzzles, setDailyPuzzles] = useState<PuzzleSummary[]>([]);
   170→  const [isLoadingPuzzles, setIsLoadingPuzzles] = useState(true);
   171→
   172→  // ── Puzzle data ─────────────────────────────────────────────────────
   173→  const [selectedPuzzle, setSelectedPuzzle] = useState<CrosswordPuzzleData | null>(null);
   174→  const [puzzleId, setPuzzleId] = useState<string | null>(null);
   175→  const [userInputs, setUserInputs] = useState<(string | null)[][]>([]);
   176→  const [selectedCell, setSelectedCell] = useState<{ row: number; col: number } | null>(null);
   177→  const [direction, setDirection] = useState<'across' | 'down'>('across');
   178→  const [activeClueNumber, setActiveClueNumber] = useState<number | null>(null);
   179→  const [activeClueDirection, setActiveClueDirection] = useState<'across' | 'down'>('across');
   180→
   181→  // ── Cell status ─────────────────────────────────────────────────────
   182→  const [revealedCells, setRevealedCells] = useState<Set<string>>(new Set());
   183→  const [correctCells, setCorrectCells] = useState<Set<string>>(new Set());
   184→  const [incorrectCells, setIncorrectCells] = useState<Set<string>>(new Set());
   185→  const [completedClues, setCompletedClues] = useState<Set<string>>(new Set());
   186→
   187→  // ── Timer / completion ──────────────────────────────────────────────
   188→  const [timer, setTimer] = useState(0);
   189→  const [isTimerRunning, setIsTimerRunning] = useState(false);
   190→  const [isCompleted, setIsCompleted] = useState(false);
   191→  const [isChecking, setIsChecking] = useState(false);
   192→  const [isLoadingPuzzle, setIsLoadingPuzzle] = useState(false);
   193→
   194→  // ── Timer interval ──────────────────────────────────────────────────
   195→  useEffect(() => {
   196→    if (!isTimerRunning) return;
   197→    const id = setInterval(() => setTimer((t) => t + 1), 1000);
   198→    return () => clearInterval(id);
   199→  }, [isTimerRunning]);
   200→
   201→  // ── Load daily puzzles on mount ─────────────────────────────────────
   202→  useEffect(() => {
   203→    let cancelled = false;
   204→    (async () => {
   205→      setIsLoadingPuzzles(true);
   206→      try {
   207→        const res = await fetch('/api/puzzles/daily');
   208→        if (!res.ok) throw new Error();
   209→        const data = await res.json();
   210→        if (!cancelled) setDailyPuzzles(data.puzzles ?? []);
   211→      } catch {
   212→        if (!cancelled) toast.error('Impossible de charger les puzzles du jour');
   213→      } finally {
   214→        if (!cancelled) setIsLoadingPuzzles(false);
   215→      }
   216→    })();
   217→    return () => { cancelled = true; };
   218→  }, []);
   219→
   220→  // ── Fetch a specific puzzle ─────────────────────────────────────────
   221→  const fetchPuzzle = useCallback(async (id: string) => {
   222→    setIsLoadingPuzzle(true);
   223→    setCurrentView('playing');
   224→    try {
   225→      const res = await fetch(`/api/puzzles/${id}`);
   226→      if (!res.ok) throw new Error();
   227→      const data = await res.json();
   228→      const puzzle: CrosswordPuzzleData = data.puzzle;
   229→
   230→      const inputs: (string | null)[][] = Array.from({ length: puzzle.rows }, () =>
   231→        Array.from<null>({ length: puzzle.cols }).fill(null),
   232→      );
   233→
   234→      // Restore saved progress
   235→      if (data.progress?.cells) {
   236→        for (const cell of data.progress.cells) {
   237→          const { row, col, letter } = cell as { row: number; col: number; letter: string };
   238→          if (row >= 0 && row < puzzle.rows && col >= 0 && col < puzzle.cols) {
   239→            inputs[row][col] = letter;
   240→          }
   241→        }
   242→      }
   243→
   244→      setSelectedPuzzle(puzzle);
   245→      setPuzzleId(id);
   246→      setUserInputs(inputs);
   247→      setSelectedCell(null);
   248→      setDirection('across');
   249→      setActiveClueNumber(null);
   250→      setActiveClueDirection('across');
   251→      setRevealedCells(new Set());
   252→      setCorrectCells(new Set());
   253→      setIncorrectCells(new Set());
   254→      setCompletedClues(new Set());
   255→      setTimer(0);
   256→      setIsTimerRunning(false);
   257→      setIsCompleted(false);
   258→      setIsChecking(false);
   259→    } catch {
   260→      toast.error('Impossible de charger le puzzle');
   261→      setCurrentView('selection');
   262→    } finally {
   263→      setIsLoadingPuzzle(false);
   264→    }
   265→  }, []);
   266→
   267→  // ── Get active word cells for highlighting ──────────────────────────
   268→  const getActiveWordCells = useCallback(
   269→    (row: number, col: number, dir: 'across' | 'down') => {
   270→      if (!selectedPuzzle) return [];
   271→      const word = selectedPuzzle.words.find((w: WordPlacement) => {
   272→        if (w.direction !== dir) return false;
   273→        return dir === 'across'
   274→          ? w.row === row && col >= w.col && col < w.col + w.length
   275→          : w.col === col && row >= w.row && row < w.row + w.length;
   276→      });
   277→      if (!word) return [];
   278→      return Array.from({ length: word.length }, (_, i) => ({
   279→        row: dir === 'across' ? word.row : word.row + i,
   280→        col: dir === 'across' ? word.col + i : word.col,
   281→      }));
   282→    },
   283→    [selectedPuzzle],
   284→  );
   285→
   286→  // ── Resolve which clue is active for a cell ────────────────────────
   287→  const resolveActiveClue = useCallback(
   288→    (row: number, col: number, dir: 'across' | 'down') => {
   289→      if (!selectedPuzzle) return;
   290→      const otherDir: 'across' | 'down' = dir === 'across' ? 'down' : 'across';
   291→
   292→      let found = selectedPuzzle.words.find((w: WordPlacement) => {
   293→        if (w.direction !== dir) return false;
   294→        return dir === 'across'
   295→          ? w.row === row && col >= w.col && col < w.col + w.length
   296→          : w.col === col && row >= w.row && row < w.row + w.length;
   297→      });
   298→
   299→      if (!found) {
   300→        found = selectedPuzzle.words.find((w: WordPlacement) => {
   301→          if (w.direction !== otherDir) return false;
   302→          return otherDir === 'across'
   303→            ? w.row === row && col >= w.col && col < w.col + w.length
   304→            : w.col === col && row >= w.row && row < w.row + w.length;
   305→        });
   306→        if (found) {
   307→          setDirection(otherDir);
   308→          setActiveClueDirection(otherDir);
   309→        }
   310→      } else {
   311→        setActiveClueDirection(dir);
   312→      }
   313→
   314→      if (found) setActiveClueNumber(found.clueNumber);
   315→    },
   316→    [selectedPuzzle],
   317→  );
   318→
   319→  // ── Handlers ─────────────────────────────────────────────────────────
   320→
   321→  const handleCellChange = useCallback(
   322→    (row: number, col: number, value: string) => {
   323→      if (isCompleted) return;
   324→      setUserInputs((prev) => {
   325→        const next = prev.map((r) => [...r]);
   326→        next[row][col] = value || null;
   327→        return next;
   328→      });
   329→      if (!isTimerRunning && value) setIsTimerRunning(true);
   330→      // Clear check highlighting on new input
   331→      setCorrectCells(new Set());
   332→      setIncorrectCells(new Set());
   333→    },
   334→    [isCompleted, isTimerRunning],
   335→  );
   336→
   337→  const handleSelectCell = useCallback(
   338→    (row: number, col: number) => {
   339→      if (isCompleted) return;
   340→      setSelectedCell({ row, col });
   341→      resolveActiveClue(row, col, direction);
   342→    },
   343→    [isCompleted, direction, resolveActiveClue],
   344→  );
   345→
   346→  const handleToggleDirection = useCallback(() => {
   347→    setDirection((prev) => {
   348→      const next: 'across' | 'down' = prev === 'across' ? 'down' : 'across';
   349→      setActiveClueDirection(next);
   350→      if (selectedCell) resolveActiveClue(selectedCell.row, selectedCell.col, next);
   351→      return next;
   352→    });
   353→  }, [selectedCell, resolveActiveClue]);
   354→
   355→  const handleSelectClue = useCallback(
   356→    (number: number, dir: 'across' | 'down') => {
   357→      if (!selectedPuzzle) return;
   358→      const word = selectedPuzzle.words.find(
   359→        (w: WordPlacement) => w.clueNumber === number && w.direction === dir,
   360→      );
   361→      if (!word) return;
   362→      setDirection(dir);
   363→      setActiveClueNumber(number);
   364→      setActiveClueDirection(dir);
   365→      setSelectedCell({ row: word.row, col: word.col });
   366→    },
   367→    [selectedPuzzle],
   368→  );
   369→
   370→  const handleCheck = useCallback(async () => {
   371→    if (!puzzleId || !selectedPuzzle || isCompleted) return;
   372→    setIsChecking(true);
   373→    try {
   374→      const res = await fetch('/api/puzzles/check', {
   375→        method: 'POST',
   376→        headers: { 'Content-Type': 'application/json' },
   377→        body: JSON.stringify({ puzzleId, userInputs }),
   378→      });
   379→      if (!res.ok) throw new Error();
   380→      const data = await res.json();
   381→
   382→      const incorrect = new Set<string>(
   383→        (data.incorrectCells ?? []).map((c: { row: number; col: number }) => toCellKey(c.row, c.col)),
   384→      );
   385→      setIncorrectCells(incorrect);
   386→
   387→      const correct = new Set<string>();
   388→      for (let r = 0; r < selectedPuzzle.rows; r++) {
   389→        for (let c = 0; c < selectedPuzzle.cols; c++) {
   390→          if (userInputs[r][c] && !incorrect.has(toCellKey(r, c))) correct.add(toCellKey(r, c));
   391→        }
   392→      }
   393→      setCorrectCells(correct);
   394→
   395→      // Update completed clues
   396→      const newCompleted = new Set<string>(completedClues);
   397→      for (const word of selectedPuzzle.words) {
   398→        const ok = Array.from({ length: word.length }, (_, i) => {
   399→          const r = word.direction === 'across' ? word.row : word.row + i;
   400→          const c = word.direction === 'across' ? word.col + i : word.col;
   401→          return !!userInputs[r]?.[c] && !incorrect.has(toCellKey(r, c));
   402→        }).every(Boolean);
   403→        if (ok) newCompleted.add(clueKey(word.clueNumber, word.direction));
   404→      }
   405→      setCompletedClues(newCompleted);
   406→
   407→      if (data.completionPercent === 100 && data.correct) {
   408→        setIsCompleted(true);
   409→        setIsTimerRunning(false);
   410→        toast.success('Bravo ! Puzzle complété !', { duration: 6000 });
   411→      } else if (incorrect.size === 0) {
   412→        toast.success('Tout est correct pour le moment !');
   413→      } else {
   414→        toast.error(`${incorrect.size} lettre${incorrect.size > 1 ? 's' : ''} incorrecte${incorrect.size > 1 ? 's' : ''}`);
   415→      }
   416→    } catch {
   417→      toast.error('Erreur lors de la vérification');
   418→    } finally {
   419→      setIsChecking(false);
   420→    }
   421→  }, [puzzleId, selectedPuzzle, userInputs, isCompleted, completedClues]);
   422→
   423→  const handleHint = useCallback(async () => {
   424→    if (!puzzleId || !selectedCell || !selectedPuzzle || isCompleted) return;
   425→    try {
   426→      const res = await fetch('/api/puzzles/hint', {
   427→        method: 'POST',
   428→        headers: { 'Content-Type': 'application/json' },
   429→        body: JSON.stringify({ puzzleId, row: selectedCell.row, col: selectedCell.col }),
   430→      });
   431→      if (!res.ok) {
   432→        const err = await res.json().catch(() => ({}));
   433→        toast.error((err as { error?: string }).error || 'Indice indisponible');
   434→        return;
   435→      }
   436→      const data = await res.json();
   437→      const { row, col } = selectedCell;
   438→      setUserInputs((prev) => {
   439→        const next = prev.map((r) => [...r]);
   440→        next[row][col] = data.letter;
   441→        return next;
   442→      });
   443→      setRevealedCells((prev) => {
   444→        const next = new Set(prev);
   445→        next.add(toCellKey(row, col));
   446→        return next;
   447→      });
   448→      toast.success(`Lettre révélée : ${data.letter}`);
   449→    } catch {
   450→      toast.error('Erreur lors de la révélation');
   451→    }
   452→  }, [puzzleId, selectedCell, selectedPuzzle, isCompleted]);
   453→
   454→  const handleClear = useCallback(() => {
   455→    if (!selectedPuzzle || isCompleted) return;
   456→    setUserInputs(
   457→      Array.from({ length: selectedPuzzle.rows }, () =>
   458→        Array.from<null>({ length: selectedPuzzle.cols }).fill(null),
   459→      ),
   460→    );
   461→    setCorrectCells(new Set());
   462→    setIncorrectCells(new Set());
   463→    setRevealedCells(new Set());
   464→    setCompletedClues(new Set());
   465→    toast.info('Grille effacée');
   466→  }, [selectedPuzzle, isCompleted]);
   467→
   468→  const handleBack = useCallback(() => {
   469→    setIsTimerRunning(false);
   470→    setSelectedPuzzle(null);
   471→    setPuzzleId(null);
   472→    setCurrentView('selection');
   473→  }, []);
   474→
   475→  // ── Derived values ───────────────────────────────────────────────────
   476→
   477→  const activeWordCells = useMemo(() => {
   478→    if (!selectedCell || !selectedPuzzle) return [];
   479→    return getActiveWordCells(selectedCell.row, selectedCell.col, direction);
   480→  }, [selectedCell, selectedPuzzle, direction, getActiveWordCells]);
   481→
   482→  const activeClueText = useMemo(() => {
   483→    if (!selectedPuzzle || activeClueNumber === null) return null;
   484→    return selectedPuzzle.clues.find(
   485→      (c: Clue) => c.number === activeClueNumber && c.direction === activeClueDirection,
   486→    ) ?? null;
   487→  }, [selectedPuzzle, activeClueNumber, activeClueDirection]);
   488→
   489→  // ══════════════════════════════════════════════════════════════════════
   490→  // RENDER
   491→  // ══════════════════════════════════════════════════════════════════════
   492→
   493→  return (
   494→    <div className="min-h-screen flex flex-col bg-background">
   495→      <AnimatePresence mode="wait">
   496→        {/* ═════════════════ PUZZLE SELECTION VIEW ═════════════════ */}
   497→        {currentView === 'selection' && (
   498→          <motion.div
   499→            key="selection"
   500→            {...fadeIn}
   501→            className="flex flex-1 flex-col"
   502→          >
   503→            {/* Subtle crossword grid background */}
   504→            <div
   505→              className="pointer-events-none fixed inset-0 -z-10 opacity-[0.025]"
   506→              style={{
   507→                backgroundImage:
   508→                  'linear-gradient(90deg, hsl(var(--foreground)) 1px, transparent 1px), linear-gradient(180deg, hsl(var(--foreground)) 1px, transparent 1px)',
   509→                backgroundSize: '28px 28px',
   510→              }}
   511→            />
   512→
   513→            {/* ── Header ─────────────────────────────────────────── */}
   514→            <header className="border-b bg-background/95 backdrop-blur-sm">
   515→              <div className="mx-auto flex max-w-5xl items-center justify-between px-4 py-6 sm:px-6">
   516→                <div className="flex items-center gap-3">
   517→                  <div className="flex h-11 w-11 items-center justify-center rounded-xl bg-emerald-600 text-white shadow-md shadow-emerald-600/20">
   518→                    <Grid3X3 className="size-6" />
   519→                  </div>
   520→                  <div>
   521→                    <h1
   522→                      className="text-3xl font-bold tracking-tight sm:text-4xl"
   523→                      style={{ fontFamily: 'Georgia, "Times New Roman", serif' }}
   524→                    >
   525→                      Mots Croisés
   526→                    </h1>
   527→                    <p className="text-sm text-muted-foreground mt-0.5">
   528→                      Gratuits &bull; Uniques &bull; Quotidiens
   529→                    </p>
   530→                  </div>
   531→                </div>
   532→                <a
   533→                  href="#admin"
   534→                  className="text-xs text-muted-foreground hover:text-foreground transition-colors"
   535→                >
   536→                  Administration
   537→                </a>
   538→              </div>
   539→            </header>
   540→
   541→            {/* ── Main Content ───────────────────────────────────── */}
   542→            <main className="flex-1 mx-auto w-full max-w-5xl px-4 py-8 sm:px-6">
