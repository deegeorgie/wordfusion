   167→// ── Streak Heatmap ──────────────────────────────────────────────

   169→function StreakHeatmap({ streakData }: { streakData: StreakCompletion[] }) {
   170→  const days = useMemo(() => {
   171→    const today = new Date();
   172→    const result: { date: string; completed: boolean; count: number }[] = [];
   173→    for (let i = 29; i >= 0; i--) {
   174→      const d = new Date(today);
   175→      d.setDate(d.getDate() - i);
   176→      const dateStr = d.toISOString().slice(0, 10);
   177→      const completions = streakData.filter((c) => c.date === dateStr);
   178→      result.push({
   179→        date: dateStr,
   180→        completed: completions.length > 0,
   181→        count: completions.length,
   182→      });
   183→    }
   184→    return result;
   185→  }, [streakData]);
   186→
   187→  if (streakData.length === 0) return null;
   188→
   189→  return (
   190→    <div className="flex flex-wrap gap-1">
   191→      {days.map((day) => (
   192→        <div
   193→          key={day.date}
   194→          title={`${day.date}${day.completed ? ` — ${day.count} puzzle${day.count > 1 ? 's' : ''}` : ''}`}
   195→          className={cn(
   196→            'h-5 w-5 rounded-sm transition-colors',
   197→            day.completed
   198→              ? 'bg-emerald-500 dark:bg-emerald-400'
   199→              : day.date <= new Date().toISOString().slice(0, 10)
   200→                ? 'bg-muted'
   201→                : 'bg-transparent',
   202→          )}
   203→        />
   204→      ))}
   205→    </div>
   206→  );
   207→}

   211→function ConfettiOverlay() {
   212→  const particles = useMemo(() => {
   213→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   214→    return Array.from({ length: 50 }, (_, i) => ({
   215→      id: i,
   216→      x: Math.random() * 100,
   217→      y: Math.random() * -50 - 10,
   218→      color: colors[i % colors.length],
   219→      size: Math.random() * 8 + 4,
   220→      delay: Math.random() * 1.2,
   221→      duration: Math.random() * 2 + 2.5,
   222→    }));
   223→  }, []);
   224→
   225→  return (
   226→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   227→      {particles.map((p) => (
   228→        <motion.div
   229→          key={p.id}
   230→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   231→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   232→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   233→          className="absolute rounded-sm"
   234→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   235→        />
   236→      ))}
   237→    </div>
   238→  );
   239→}
