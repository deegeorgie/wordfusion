   121→function formatTimer(seconds: number): string {
   122→  const m = Math.floor(seconds / 60);
   123→  const s = seconds % 60;

   151→function ConfettiOverlay() {
   152→  const particles = useMemo(() => {
   153→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   154→    return Array.from({ length: 50 }, (_, i) => ({
   155→      id: i,
   156→      x: Math.random() * 100,
   157→      y: Math.random() * -50 - 10,
   158→      color: colors[i % colors.length],
   159→      size: Math.random() * 8 + 4,
   160→      delay: Math.random() * 1.2,
   161→      duration: Math.random() * 2 + 2.5,
   162→    }));
   163→  }, []);
   164→
   165→  return (
   166→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   167→      {particles.map((p) => (
   168→        <motion.div
   169→          key={p.id}
   170→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   171→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   172→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   173→          className="absolute rounded-sm"
   174→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   175→        />
   176→      ))}
   177→    </div>
   178→  );
   179→}

   322→  // ── Calculate daily streak ───────────────────────────────────────────
   323→  const streak = useMemo(() => {
   324→    if (streakData.length === 0) return 0;
   325→    const dates = [...new Set(streakData.map((c) => c.date))].sort().reverse();
   326→    if (dates.length === 0) return 0;
   327→
   328→    const today = new Date().toISOString().slice(0, 10);
   329→    const yesterday = new Date(Date.now() - 86400000).toISOString().slice(0, 10);
   330→
   331→    if (dates[0] !== today && dates[0] !== yesterday) return 0;
   332→
   333→    let count = 1;
   334→    for (let i = 1; i < dates.length; i++) {
   335→      const prev = new Date(dates[i - 1]);
   336→      const curr = new Date(dates[i]);
   337→      const diff = (prev.getTime() - curr.getTime()) / 86400000;
   338→      if (diff === 1) count++;
   339→      else break;
   340→    }
   341→    return count;
   342→  }, [streakData]);
   343→
   344→  // ── Save streak on puzzle completion ─────────────────────────────────
   345→  useEffect(() => {

   348→      date: new Date().toISOString().slice(0, 10),

   391→      setActiveClueNumber(null);
   392→      setActiveClueDirection('across');

   394→      setCorrectCellsManual(new Set());
   395→      setIncorrectCellsManual(new Set());

   398→      setIsTimerRunning(false);
   399→      setIsCompleted(false);
   400→      setIsChecking(false);

   404→    } finally {
   405→      setIsLoadingPuzzle(false);
   406→    }
   407→  }, []);
   408→
   409→  // ── Get active word cells for highlighting ──────────────────────────
   410→  const getActiveWordCells = useCallback(
   411→    (row: number, col: number, dir: 'across' | 'down') => {
   412→      if (!selectedPuzzle) return [];
   413→      const word = selectedPuzzle.words.find((w: WordPlacement) => {
   414→        if (w.direction !== dir) return false;
   415→        return dir === 'across'
   416→          ? w.row === row && col >= w.col && col < w.col + w.length
   417→          : w.col === col && row >= w.row && row < w.row + w.length;
   418→      });
   419→      if (!word) return [];
   420→      return Array.from({ length: word.length }, (_, i) => ({
   421→        row: dir === 'across' ? word.row : word.row + i,
   422→        col: dir === 'across' ? word.col + i : word.col,
   423→      }));
   424→    },
   425→    [selectedPuzzle],
   426→  );
   427→
   428→  // ── Resolve which clue is active for a cell ────────────────────────
   429→  const resolveActiveClue = useCallback(
   430→    (row: number, col: number, dir: 'across' | 'down') => {
   431→      if (!selectedPuzzle) return;
   432→      const otherDir: 'across' | 'down' = dir === 'across' ? 'down' : 'across';
   433→
   434→      let found = selectedPuzzle.words.find((w: WordPlacement) => {
   435→        if (w.direction !== dir) return false;
   436→        return dir === 'across'
   437→          ? w.row === row && col >= w.col && col < w.col + w.length
   438→          : w.col === col && row >= w.row && row < w.row + w.length;
   439→      });
   440→
   441→      if (!found) {
   442→        found = selectedPuzzle.words.find((w: WordPlacement) => {
   443→          if (w.direction !== otherDir) return false;
   444→          return otherDir === 'across'
   445→            ? w.row === row && col >= w.col && col < w.col + w.length
   446→            : w.col === col && row >= w.row && row < w.row + w.length;
   447→        });
   448→        if (found) {
   449→          setDirection(otherDir);
   450→          setActiveClueDirection(otherDir);
   451→        }
   452→      } else {
   453→        setActiveClueDirection(dir);
   454→      }
   455→
   456→      if (found) setActiveClueNumber(found.clueNumber);
   457→    },
   458→    [selectedPuzzle],
   459→  );
   460→
   461→  // ── Handlers ─────────────────────────────────────────────────────────
   462→
   463→  const handleCellChange = useCallback(
   464→    (row: number, col: number, value: string) => {
   465→      if (isCompleted) return;
   466→      setUserInputs((prev) => {
   467→        const next = prev.map((r) => [...r]);
   468→        next[row][col] = value || null;
   469→        return next;
   470→      });
   471→      if (!isTimerRunning && value) setIsTimerRunning(true);
   472→      // Only clear manual check highlights on new input (auto-check handles its own)
   473→      if (!autoCheck) {
   474→        setCorrectCellsManual(new Set());
   475→        setIncorrectCellsManual(new Set());
   476→      }
   477→    },
   478→    [isCompleted, isTimerRunning, autoCheck],
   479→  );
   480→
   481→  const handleSelectCell = useCallback(
   482→    (row: number, col: number) => {
   483→      if (isCompleted) return;
   484→      setSelectedCell({ row, col });
   485→      resolveActiveClue(row, col, direction);
   486→    },
   487→    [isCompleted, direction, resolveActiveClue],
   488→  );
   489→
   490→  const handleToggleDirection = useCallback(() => {
   491→    setDirection((prev) => {
   492→      const next: 'across' | 'down' = prev === 'across' ? 'down' : 'across';
   493→      setActiveClueDirection(next);
   494→      if (selectedCell) resolveActiveClue(selectedCell.row, selectedCell.col, next);
   495→      return next;
   496→    });
   497→  }, [selectedCell, resolveActiveClue]);
   498→
   499→  const handleSelectClue = useCallback(
   500→    (number: number, dir: 'across' | 'down') => {
   501→      if (!selectedPuzzle) return;
   502→      const word = selectedPuzzle.words.find(
   503→        (w: WordPlacement) => w.clueNumber === number && w.direction === dir,
   504→      );
   505→      if (!word) return;
   506→      setDirection(dir);
   507→      setActiveClueNumber(number);
   508→      setActiveClueDirection(dir);
   509→      setSelectedCell({ row: word.row, col: word.col });
   510→    },
   511→    [selectedPuzzle],
   512→  );
   513→
   514→  const handleCheck = useCallback(async () => {
   515→    if (!puzzleId || !selectedPuzzle || isCompleted) return;
   516→    setIsChecking(true);
   517→    try {
   518→      const res = await fetch('/api/puzzles/check', {
   519→        method: 'POST',
   520→        headers: { 'Content-Type': 'application/json' },
   521→        body: JSON.stringify({ puzzleId, userInputs }),
   522→      });
   523→      if (!res.ok) throw new Error();
   524→      const data = await res.json();
   525→
   526→      const incorrect = new Set<string>(
   527→        (data.incorrectCells ?? []).map((c: { row: number; col: number }) => toCellKey(c.row, c.col)),
   528→      );
   529→      setIncorrectCellsManual(incorrect);
   530→
   531→      const correct = new Set<string>();
   532→      for (let r = 0; r < selectedPuzzle.rows; r++) {
   533→        for (let c = 0; c < selectedPuzzle.cols; c++) {
   534→          if (userInputs[r][c] && !incorrect.has(toCellKey(r, c))) correct.add(toCellKey(r, c));
   535→        }
   536→      }
   537→      setCorrectCellsManual(correct);
   538→
   539→      // Update completed clues
   540→      const newCompleted = new Set<string>(completedClues);
   541→      for (const word of selectedPuzzle.words) {
   542→        const ok = Array.from({ length: word.length }, (_, i) => {
   543→          const r = word.direction === 'across' ? word.row : word.row + i;
   544→          const c = word.direction === 'across' ? word.col + i : word.col;
   545→          return !!userInputs[r]?.[c] && !incorrect.has(toCellKey(r, c));
   546→        }).every(Boolean);
   547→        if (ok) newCompleted.add(clueKey(word.clueNumber, word.direction));
   548→      }
   549→      setCompletedClues(newCompleted);
   550→
   551→      if (data.completionPercent === 100 && data.correct) {
   552→        setIsCompleted(true);
   553→        setIsTimerRunning(false);
   554→        toast.success('Bravo ! Puzzle complété !', { duration: 6000 });
   555→      } else if (incorrect.size === 0) {
