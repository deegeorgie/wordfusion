   127→   127→function ConfettiOverlay() {
   128→   128→  const particles = useMemo(() => {
   129→   129→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   130→   130→    return Array.from({ length: 50 }, (_, i) => ({
   131→   131→      id: i,
   132→   132→      x: Math.random() * 100,
   133→   133→      y: Math.random() * -50 - 10,
   134→   134→      color: colors[i % colors.length],
   135→   135→      size: Math.random() * 8 + 4,
   136→   136→      delay: Math.random() * 1.2,
   137→   137→      duration: Math.random() * 2 + 2.5,
   138→   138→    }));
   139→   139→  }, []);
   140→   140→
   141→   141→  return (
   142→   142→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   143→   143→      {particles.map((p) => (
   144→   144→        <motion.div
   145→   145→          key={p.id}
   146→   146→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   147→   147→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   148→   148→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   149→   149→          className="absolute rounded-sm"
   150→   150→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   151→   151→        />
   152→   152→      ))}
   153→   153→    </div>
   154→   154→  );
   155→   155→}

   210→   210→  const [isTimerRunning, setIsTimerRunning] = useState(false);

   275→   275→    setCurrentView('playing');

   277→   277→      const res = await fetch(`/api/puzzles/${id}`);

   279→   279→      const data = await res.json();

   283→   283→        Array.from<null>({ length: puzzle.cols }).fill(null),

   301→   301→      setActiveClueNumber(null);
   302→   302→      setActiveClueDirection('across');
   303→   303→      setRevealedCells(new Set());
   304→   304→      setCorrectCellsManual(new Set());
   305→   305→      setIncorrectCellsManual(new Set());
   306→   306→      setCompletedClues(new Set());

   308→   308→      setIsTimerRunning(false);
   309→   309→      setIsCompleted(false);
   310→   310→      setIsChecking(false);
   311→   311→    } catch {
   312→   312→      toast.error('Impossible de charger le puzzle');
   313→   313→      setCurrentView('selection');
   314→   314→    } finally {
   315→   315→      setIsLoadingPuzzle(false);
   316→   316→    }
   317→   317→  }, []);
   318→   318→
   319→   319→  // ── Get active word cells for highlighting ──────────────────────────
   320→   320→  const getActiveWordCells = useCallback(
   321→   321→    (row: number, col: number, dir: 'across' | 'down') => {

   323→   323→      const word = selectedPuzzle.words.find((w: WordPlacement) => {
   324→   324→        if (w.direction !== dir) return false;
   325→   325→        return dir === 'across'
   326→   326→          ? w.row === row && col >= w.col && col < w.col + w.length
   327→   327→          : w.col === col && row >= w.row && row < w.row + w.length;
   328→   328→      });
   329→   329→      if (!word) return [];
   330→   330→      return Array.from({ length: word.length }, (_, i) => ({
   331→   331→        row: dir === 'across' ? word.row : word.row + i,
   332→   332→        col: dir === 'across' ? word.col + i : word.col,
   333→   333→      }));
   334→   334→    },
   335→   335→    [selectedPuzzle],
   336→   336→  );
   337→   337→
   338→   338→  // ── Resolve which clue is active for a cell ────────────────────────
   339→   339→  const resolveActiveClue = useCallback(
   340→   340→    (row: number, col: number, dir: 'across' | 'down') => {
   341→   341→      if (!selectedPuzzle) return;
   342→   342→      const otherDir: 'across' | 'down' = dir === 'across' ? 'down' : 'across';
   343→   343→
   344→   344→      let found = selectedPuzzle.words.find((w: WordPlacement) => {
   345→   345→        if (w.direction !== dir) return false;
   346→   346→        return dir === 'across'
   347→   347→          ? w.row === row && col >= w.col && col < w.col + w.length
   348→   348→          : w.col === col && row >= w.row && row < w.row + w.length;
   349→   349→      });
   350→   350→
   351→   351→      if (!found) {
   352→   352→        found = selectedPuzzle.words.find((w: WordPlacement) => {
   353→   353→          if (w.direction !== otherDir) return false;
   354→   354→          return otherDir === 'across'
   355→   355→            ? w.row === row && col >= w.col && col < w.col + w.length
   356→   356→            : w.col === col && row >= w.row && row < w.row + w.length;
   357→   357→        });
   358→   358→        if (found) {
   359→   359→          setDirection(otherDir);
   360→   360→          setActiveClueDirection(otherDir);
   361→   361→        }
   362→   362→      } else {
   363→   363→        setActiveClueDirection(dir);
   364→   364→      }
   365→   365→
   366→   366→      if (found) setActiveClueNumber(found.clueNumber);

   368→   368→    [selectedPuzzle],
   369→   369→  );

   384→   384→        setCorrectCellsManual(new Set());
   385→   385→        setIncorrectCellsManual(new Set());
   386→   386→      }
   387→   387→    },
   388→   388→    [isCompleted, isTimerRunning, autoCheck],

   402→   402→      const next: 'across' | 'down' = prev === 'across' ? 'down' : 'across';
   403→   403→      setActiveClueDirection(next);
   404→   404→      if (selectedCell) resolveActiveClue(selectedCell.row, selectedCell.col, next);
   405→   405→      return next;
   406→   406→    });
   407→   407→  }, [selectedCell, resolveActiveClue]);
   408→   408→
   409→   409→  const handleSelectClue = useCallback(
   410→   410→    (number: number, dir: 'across' | 'down') => {
   411→   411→      if (!selectedPuzzle) return;
   412→   412→      const word = selectedPuzzle.words.find(
   413→   413→        (w: WordPlacement) => w.clueNumber === number && w.direction === dir,
   414→   414→      );
   415→   415→      if (!word) return;
   416→   416→      setDirection(dir);
   417→   417→      setActiveClueNumber(number);
   418→   418→      setActiveClueDirection(dir);
   419→   419→      setSelectedCell({ row: word.row, col: word.col });
   420→   420→    },
   421→   421→    [selectedPuzzle],
   422→   422→  );
   423→   423→
   424→   424→  const handleCheck = useCallback(async () => {
   425→   425→    if (!puzzleId || !selectedPuzzle || isCompleted) return;
   426→   426→    setIsChecking(true);
   427→   427→    try {
   428→   428→      const res = await fetch('/api/puzzles/check', {
   429→   429→        method: 'POST',
   430→   430→        headers: { 'Content-Type': 'application/json' },
   431→   431→        body: JSON.stringify({ puzzleId, userInputs }),
   432→   432→      });
   433→   433→      if (!res.ok) throw new Error();
   434→   434→      const data = await res.json();
   435→   435→
   436→   436→      const incorrect = new Set<string>(
   437→   437→        (data.incorrectCells ?? []).map((c: { row: number; col: number }) => toCellKey(c.row, c.col)),
   438→   438→      );
   439→   439→      setIncorrectCellsManual(incorrect);
   440→   440→
   441→   441→      const correct = new Set<string>();
   442→   442→      for (let r = 0; r < selectedPuzzle.rows; r++) {
   443→   443→        for (let c = 0; c < selectedPuzzle.cols; c++) {
   444→   444→          if (userInputs[r][c] && !incorrect.has(toCellKey(r, c))) correct.add(toCellKey(r, c));
   445→   445→        }
   446→   446→      }
   447→   447→      setCorrectCellsManual(correct);
   448→   448→
   449→   449→      // Update completed clues
   450→   450→      const newCompleted = new Set<string>(completedClues);
   451→   451→      for (const word of selectedPuzzle.words) {
   452→   452→        const ok = Array.from({ length: word.length }, (_, i) => {
   453→   453→          const r = word.direction === 'across' ? word.row : word.row + i;
   454→   454→          const c = word.direction === 'across' ? word.col + i : word.col;
   455→   455→          return !!userInputs[r]?.[c] && !incorrect.has(toCellKey(r, c));
   456→   456→        }).every(Boolean);
   457→   457→        if (ok) newCompleted.add(clueKey(word.clueNumber, word.direction));
