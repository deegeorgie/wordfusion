    56→function toCellKey(row: number, col: number): string {
    57→  return `${row},${col}`;
    58→}
    59→
    60→function clueKey(number: number, direction: 'across' | 'down'): string {
    61→  return `${number}-${direction}`;
    62→}
    63→
    64→function formatFrenchDate(): string {
    65→  return new Date().toLocaleDateString('fr-FR', {
    66→    weekday: 'long',
    67→    year: 'numeric',
    68→    month: 'long',
    69→    day: 'numeric',
    70→  });
    71→}
    72→
    73→function formatTimer(seconds: number): string {
    74→  const m = Math.floor(seconds / 60);
    75→  const s = seconds % 60;
    76→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    77→}

    79→function difficultyLabel(d: number): string {
    80→  if (d <= 1) return 'Facile';
    81→  if (d <= 2) return 'Moyen';
    82→  if (d <= 3) return 'Difficile';
    83→  return 'Expert';
    84→}
    85→
    86→function difficultyColor(d: number): string {
    87→  if (d <= 1) return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400';
    88→  if (d <= 2) return 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400';
    89→  if (d <= 3) return 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400';
    90→  return 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400';
    91→}

    95→function ConfettiOverlay() {
    96→  const particles = useMemo(() => {
    97→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
    98→    return Array.from({ length: 50 }, (_, i) => ({
    99→      id: i,
   100→      x: Math.random() * 100,
   101→      y: Math.random() * -50 - 10,
   102→      color: colors[i % colors.length],
   103→      size: Math.random() * 8 + 4,
   104→      delay: Math.random() * 1.2,
   105→      duration: Math.random() * 2 + 2.5,
   106→    }));
   107→  }, []);
   108→
   109→  return (
   110→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   111→      {particles.map((p) => (
   112→        <motion.div
   113→          key={p.id}
   114→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   115→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   116→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   117→          className="absolute rounded-sm"
   118→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   119→        />
   120→      ))}
   121→    </div>
   122→  );
   123→}

   148→export default function Home() {

   150→  const [currentView, setCurrentView] = useState<'selection' | 'playing'>('selection');
   151→  const [dailyPuzzles, setDailyPuzzles] = useState<PuzzleSummary[]>([]);

   153→

   252→      if (!selectedPuzzle) return [];

   257→          : w.col === col && row >= w.row && row < w.row + w.length;
   258→      });
   259→      if (!word) return [];
   260→      return Array.from({ length: word.length }, (_, i) => ({
   261→        row: dir === 'across' ? word.row : word.row + i,
   262→        col: dir === 'across' ? word.col + i : word.col,
   263→      }));
   264→    },
   265→    [selectedPuzzle],
   266→  );
   267→
   268→  // ── Resolve which clue is active for a cell ────────────────────────

   271→      if (!selectedPuzzle) return;
   272→      const otherDir: 'across' | 'down' = dir === 'across' ? 'down' : 'across';
   273→
   274→      let found = selectedPuzzle.words.find((w: WordPlacement) => {
   275→        if (w.direction !== dir) return false;
   276→        return dir === 'across'
   277→          ? w.row === row && col >= w.col && col < w.col + w.length
   278→          : w.col === col && row >= w.row && row < w.row + w.length;
   279→      });
   280→
   281→      if (!found) {
   282→        found = selectedPuzzle.words.find((w: WordPlacement) => {
   283→          if (w.direction !== otherDir) return false;
   284→          return otherDir === 'across'
   285→            ? w.row === row && col >= w.col && col < w.col + w.length
   286→            : w.col === col && row >= w.row && row < w.row + w.length;
   287→        });
   288→        if (found) {
   289→          setDirection(otherDir);
   290→          setActiveClueDirection(otherDir);
   291→        }
   292→      } else {
   293→        setActiveClueDirection(dir);
   294→      }
   295→
   296→      if (found) setActiveClueNumber(found.clueNumber);
   297→    },
   298→    [selectedPuzzle],
   299→  );

   316→    [isCompleted, isTimerRunning],

   319→  const handleSelectCell = useCallback(
   320→    (row: number, col: number) => {
   321→      if (isCompleted) return;
   322→      setSelectedCell({ row, col });
   323→      resolveActiveClue(row, col, direction);
   324→    },
   325→    [isCompleted, direction, resolveActiveClue],
   326→  );
   327→
   328→  const handleToggleDirection = useCallback(() => {

   331→      setActiveClueDirection(next);

   337→  const handleSelectClue = useCallback(
   338→    (number: number, dir: 'across' | 'down') => {

   340→      const word = selectedPuzzle.words.find(
   341→        (w: WordPlacement) => w.clueNumber === number && w.direction === dir,

   345→      setActiveClueNumber(number);
   346→      setActiveClueDirection(dir);

   352→  const handleCheck = useCallback(async () => {
   353→    if (!puzzleId || !selectedPuzzle || isCompleted) return;
   354→    setIsChecking(true);
   355→    try {
   356→      const res = await fetch('/api/puzzles/check', {
   357→        method: 'POST',
   358→        headers: { 'Content-Type': 'application/json' },

   364→      const incorrect = new Set<string>(
   365→        (data.incorrectCells ?? []).map((c: { row: number; col: number }) => toCellKey(c.row, c.col)),
   366→      );
   367→      setIncorrectCells(incorrect);
   368→
   369→      const correct = new Set<string>();
   370→      for (let r = 0; r < selectedPuzzle.rows; r++) {
   371→        for (let c = 0; c < selectedPuzzle.cols; c++) {
   372→          if (userInputs[r][c] && !incorrect.has(toCellKey(r, c))) correct.add(toCellKey(r, c));
   373→        }
   374→      }
   375→      setCorrectCells(correct);
   376→
   377→      // Update completed clues
   378→      const newCompleted = new Set<string>(completedClues);
   379→      for (const word of selectedPuzzle.words) {
   380→        const ok = Array.from({ length: word.length }, (_, i) => {

   382→          const c = word.direction === 'across' ? word.col + i : word.col;
   383→          return !!userInputs[r]?.[c] && !incorrect.has(toCellKey(r, c));
   384→        }).every(Boolean);
   385→        if (ok) newCompleted.add(clueKey(word.clueNumber, word.direction));
   386→      }
   387→      setCompletedClues(newCompleted);
   388→
   389→      if (data.completionPercent === 100 && data.correct) {
   390→        setIsCompleted(true);
   391→        setIsTimerRunning(false);
   392→        toast.success('Bravo ! Puzzle complété !', { duration: 6000 });
   393→      } else if (incorrect.size === 0) {
   394→        toast.success('Tout est correct pour le moment !');
   395→      } else {
   396→        toast.error(`${incorrect.size} lettre${incorrect.size > 1 ? 's' : ''} incorrecte${incorrect.size > 1 ? 's' : ''}`);
   397→      }
   398→    } catch {
   399→      toast.error('Erreur lors de la vérification');
   400→    } finally {
   401→      setIsChecking(false);

   493→              <div className="flex items-center gap-3">
