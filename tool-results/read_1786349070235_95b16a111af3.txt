    95→    95→function ConfettiOverlay() {
    96→    96→  const particles = useMemo(() => {
    97→    97→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
    98→    98→    return Array.from({ length: 50 }, (_, i) => ({
    99→    99→      id: i,
   100→   100→      x: Math.random() * 100,
   101→   101→      y: Math.random() * -50 - 10,
   102→   102→      color: colors[i % colors.length],
   103→   103→      size: Math.random() * 8 + 4,
   104→   104→      delay: Math.random() * 1.2,
   105→   105→      duration: Math.random() * 2 + 2.5,
   106→   106→    }));
   107→   107→  }, []);
   108→   108→
   109→   109→  return (
   110→   110→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   111→   111→      {particles.map((p) => (
   112→   112→        <motion.div
   113→   113→          key={p.id}
   114→   114→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   115→   115→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   116→   116→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   117→   117→          className="absolute rounded-sm"
   118→   118→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   119→   119→        />
   120→   120→      ))}
   121→   121→    </div>
   122→   122→  );
   123→   123→}

   148→   148→export default function Home() {
   149→   149→  // ── View state ──────────────────────────────────────────────────────
   150→   150→  const [currentView, setCurrentView] = useState<'selection' | 'playing'>('selection');
   151→   151→  const [dailyPuzzles, setDailyPuzzles] = useState<PuzzleSummary[]>([]);
   152→   152→  const [isLoadingPuzzles, setIsLoadingPuzzles] = useState(true);
   153→   153→
   154→   154→  // ── Puzzle data ─────────────────────────────────────────────────────
   155→   155→  const [selectedPuzzle, setSelectedPuzzle] = useState<CrosswordPuzzleData | null>(null);
   156→   156→  const [puzzleId, setPuzzleId] = useState<string | null>(null);
   157→   157→  const [userInputs, setUserInputs] = useState<(string | null)[][]>([]);
   158→   158→  const [selectedCell, setSelectedCell] = useState<{ row: number; col: number } | null>(null);
   159→   159→  const [direction, setDirection] = useState<'across' | 'down'>('across');
   160→   160→  const [activeClueNumber, setActiveClueNumber] = useState<number | null>(null);
   161→   161→  const [activeClueDirection, setActiveClueDirection] = useState<'across' | 'down'>('across');
   162→   162→
   163→   163→  // ── Cell status ─────────────────────────────────────────────────────
   164→   164→  const [revealedCells, setRevealedCells] = useState<Set<string>>(new Set());
   165→   165→  const [correctCells, setCorrectCells] = useState<Set<string>>(new Set());
   166→   166→  const [incorrectCells, setIncorrectCells] = useState<Set<string>>(new Set());
   167→   167→  const [completedClues, setCompletedClues] = useState<Set<string>>(new Set());
   168→   168→
   169→   169→  // ── Timer / completion ──────────────────────────────────────────────
   170→   170→  const [timer, setTimer] = useState(0);
   171→   171→  const [isTimerRunning, setIsTimerRunning] = useState(false);
   172→   172→  const [isCompleted, setIsCompleted] = useState(false);
   173→   173→  const [isChecking, setIsChecking] = useState(false);
   174→   174→  const [isLoadingPuzzle, setIsLoadingPuzzle] = useState(false);
   175→   175→
   176→   176→  // ── Timer interval ──────────────────────────────────────────────────
   177→   177→  useEffect(() => {
   178→   178→    if (!isTimerRunning) return;
   179→   179→    const id = setInterval(() => setTimer((t) => t + 1), 1000);
   180→   180→    return () => clearInterval(id);
   181→   181→  }, [isTimerRunning]);
   182→   182→
   183→   183→  // ── Load daily puzzles on mount ─────────────────────────────────────
   184→   184→  useEffect(() => {

   186→   186→    (async () => {

   193→   193→      } catch {
   194→   194→        if (!cancelled) toast.error('Impossible de charger les puzzles du jour');
   195→   195→      } finally {
   196→   196→        if (!cancelled) setIsLoadingPuzzles(false);
   197→   197→      }
   198→   198→    })();
   199→   199→    return () => { cancelled = true; };
   200→   200→  }, []);
   201→   201→
   202→   202→  // ── Fetch a specific puzzle ─────────────────────────────────────────
   203→   203→  const fetchPuzzle = useCallback(async (id: string) => {
   204→   204→    setIsLoadingPuzzle(true);
   205→   205→    setCurrentView('playing');
   206→   206→    try {
   207→   207→      const res = await fetch(`/api/puzzles/${id}`);
   208→   208→      if (!res.ok) throw new Error();
   209→   209→      const data = await res.json();
   210→   210→      const puzzle: CrosswordPuzzleData = data.puzzle;
   211→   211→
   212→   212→      const inputs: (string | null)[][] = Array.from({ length: puzzle.rows }, () =>
   213→   213→        Array.from<null>({ length: puzzle.cols }).fill(null),
   214→   214→      );
   215→   215→
   216→   216→      // Restore saved progress
   217→   217→      if (data.progress?.cells) {
   218→   218→        for (const cell of data.progress.cells) {
   219→   219→          const { row, col, letter } = cell as { row: number; col: number; letter: string };
   220→   220→          if (row >= 0 && row < puzzle.rows && col >= 0 && col < puzzle.cols) {
   221→   221→            inputs[row][col] = letter;
   222→   222→          }
   223→   223→        }
   224→   224→      }
   225→   225→
   226→   226→      setSelectedPuzzle(puzzle);
   227→   227→      setPuzzleId(id);
   228→   228→      setUserInputs(inputs);
   229→   229→      setSelectedCell(null);
   230→   230→      setDirection('across');
   231→   231→      setActiveClueNumber(null);
   232→   232→      setActiveClueDirection('across');
   233→   233→      setRevealedCells(new Set());
   234→   234→      setCorrectCells(new Set());
   235→   235→      setIncorrectCells(new Set());
   236→   236→      setCompletedClues(new Set());

   238→   238→      setIsTimerRunning(false);
   239→   239→      setIsCompleted(false);
   240→   240→      setIsChecking(false);
   241→   241→    } catch {
   242→   242→      toast.error('Impossible de charger le puzzle');
   243→   243→      setCurrentView('selection');
   244→   244→    } finally {
   245→   245→      setIsLoadingPuzzle(false);
   246→   246→    }
   247→   247→  }, []);
   248→   248→
   249→   249→  // ── Get active word cells for highlighting ──────────────────────────
   250→   250→  const getActiveWordCells = useCallback(
   251→   251→    (row: number, col: number, dir: 'across' | 'down') => {
   252→   252→      if (!selectedPuzzle) return [];
   253→   253→      const word = selectedPuzzle.words.find((w: WordPlacement) => {
   254→   254→        if (w.direction !== dir) return false;
   255→   255→        return dir === 'across'
   256→   256→          ? w.row === row && col >= w.col && col < w.col + w.length
   257→   257→          : w.col === col && row >= w.row && row < w.row + w.length;
   258→   258→      });
   259→   259→      if (!word) return [];
   260→   260→      return Array.from({ length: word.length }, (_, i) => ({
   261→   261→        row: dir === 'across' ? word.row : word.row + i,
   262→   262→        col: dir === 'across' ? word.col + i : word.col,
   263→   263→      }));
   264→   264→    },
   265→   265→    [selectedPuzzle],
   266→   266→  );
   267→   267→
   268→   268→  // ── Resolve which clue is active for a cell ────────────────────────
   269→   269→  const resolveActiveClue = useCallback(
   270→   270→    (row: number, col: number, dir: 'across' | 'down') => {
   271→   271→      if (!selectedPuzzle) return;
   272→   272→      const otherDir: 'across' | 'down' = dir === 'across' ? 'down' : 'across';
   273→   273→
   274→   274→      let found = selectedPuzzle.words.find((w: WordPlacement) => {
   275→   275→        if (w.direction !== dir) return false;
   276→   276→        return dir === 'across'
   277→   277→          ? w.row === row && col >= w.col && col < w.col + w.length
   278→   278→          : w.col === col && row >= w.row && row < w.row + w.length;
   279→   279→      });
   280→   280→
   281→   281→      if (!found) {
   282→   282→        found = selectedPuzzle.words.find((w: WordPlacement) => {
   283→   283→          if (w.direction !== otherDir) return false;
   284→   284→          return otherDir === 'across'
   285→   285→            ? w.row === row && col >= w.col && col < w.col + w.length
   286→   286→            : w.col === col && row >= w.row && row < w.row + w.length;
   287→   287→        });
   288→   288→        if (found) {
   289→   289→          setDirection(otherDir);
   290→   290→          setActiveClueDirection(otherDir);
   291→   291→        }
   292→   292→      } else {
   293→   293→        setActiveClueDirection(dir);
   294→   294→      }
   295→   295→
   296→   296→      if (found) setActiveClueNumber(found.clueNumber);
   297→   297→    },
   298→   298→    [selectedPuzzle],
   299→   299→  );
   300→   300→
   301→   301→  // ── Handlers ─────────────────────────────────────────────────────────
   302→   302→
   303→   303→  const handleCellChange = useCallback(
   304→   304→    (row: number, col: number, value: string) => {
   305→   305→      if (isCompleted) return;
   306→   306→      setUserInputs((prev) => {
   307→   307→        const next = prev.map((r) => [...r]);
   308→   308→        next[row][col] = value || null;
   309→   309→        return next;
   310→   310→      });
   311→   311→      if (!isTimerRunning && value) setIsTimerRunning(true);
   312→   312→      // Clear check highlighting on new input
   313→   313→      setCorrectCells(new Set());
   314→   314→      setIncorrectCells(new Set());
   315→   315→    },
   316→   316→    [isCompleted, isTimerRunning],
   317→   317→  );
   318→   318→
   319→   319→  const handleSelectCell = useCallback(
   320→   320→    (row: number, col: number) => {
   321→   321→      if (isCompleted) return;
   322→   322→      setSelectedCell({ row, col });
   323→   323→      resolveActiveClue(row, col, direction);
   324→   324→    },
   325→   325→    [isCompleted, direction, resolveActiveClue],
   326→   326→  );
   327→   327→
   328→   328→  const handleToggleDirection = useCallback(() => {
   329→   329→    setDirection((prev) => {
   330→   330→      const next: 'across' | 'down' = prev === 'across' ? 'down' : 'across';
   331→   331→      setActiveClueDirection(next);

   333→   333→      return next;
   334→   334→    });
   335→   335→  }, [selectedCell, resolveActiveClue]);
   336→   336→

   338→   338→    (number: number, dir: 'across' | 'down') => {
   339→   339→      if (!selectedPuzzle) return;
   340→   340→      const word = selectedPuzzle.words.find(
   341→   341→        (w: WordPlacement) => w.clueNumber === number && w.direction === dir,

   345→   345→      setActiveClueNumber(number);
   346→   346→      setActiveClueDirection(dir);
   347→   347→      setSelectedCell({ row: word.row, col: word.col });
   348→   348→    },
   349→   349→    [selectedPuzzle],
   350→   350→  );
   351→   351→

   353→   353→    if (!puzzleId || !selectedPuzzle || isCompleted) return;
   354→   354→    setIsChecking(true);
   355→   355→    try {
   356→   356→      const res = await fetch('/api/puzzles/check', {
   357→   357→        method: 'POST',
   358→   358→        headers: { 'Content-Type': 'application/json' },

   361→   361→      if (!res.ok) throw new Error();
   362→   362→      const data = await res.json();
   363→   363→
   364→   364→      const incorrect = new Set<string>(
   365→   365→        (data.incorrectCells ?? []).map((c: { row: number; col: number }) => toCellKey(c.row, c.col)),
   366→   366→      );
   367→   367→      setIncorrectCells(incorrect);
   368→   368→
   369→   369→      const correct = new Set<string>();
   370→   370→      for (let r = 0; r < selectedPuzzle.rows; r++) {
   371→   371→        for (let c = 0; c < selectedPuzzle.cols; c++) {
   372→   372→          if (userInputs[r][c] && !incorrect.has(toCellKey(r, c))) correct.add(toCellKey(r, c));
   373→   373→        }
   374→   374→      }
   375→   375→      setCorrectCells(correct);
   376→   376→
   377→   377→      // Update completed clues
   378→   378→      const newCompleted = new Set<string>(completedClues);
   379→   379→      for (const word of selectedPuzzle.words) {
   380→   380→        const ok = Array.from({ length: word.length }, (_, i) => {
   381→   381→          const r = word.direction === 'across' ? word.row : word.row + i;
   382→   382→          const c = word.direction === 'across' ? word.col + i : word.col;
   383→   383→          return !!userInputs[r]?.[c] && !incorrect.has(toCellKey(r, c));
   384→   384→        }).every(Boolean);
   385→   385→        if (ok) newCompleted.add(clueKey(word.clueNumber, word.direction));
   386→   386→      }
   387→   387→      setCompletedClues(newCompleted);
   388→   388→
   389→   389→      if (data.completionPercent === 100 && data.correct) {
   390→   390→        setIsCompleted(true);
   391→   391→        setIsTimerRunning(false);
   392→   392→        toast.success('Bravo ! Puzzle complété !', { duration: 6000 });
   393→   393→      } else if (incorrect.size === 0) {
   394→   394→        toast.success('Tout est correct pour le moment !');
   395→   395→      } else {
   396→   396→        toast.error(`${incorrect.size} lettre${incorrect.size > 1 ? 's' : ''} incorrecte${incorrect.size > 1 ? 's' : ''}`);
   397→   397→      }
   398→   398→    } catch {
   399→   399→      toast.error('Erreur lors de la vérification');
   400→   400→    } finally {
   401→   401→      setIsChecking(false);
   402→   402→    }
   403→   403→  }, [puzzleId, selectedPuzzle, userInputs, isCompleted, completedClues]);
   404→   404→
   405→   405→  const handleHint = useCallback(async () => {
   406→   406→    if (!puzzleId || !selectedCell || !selectedPuzzle || isCompleted) return;
   407→   407→    try {
   408→   408→      const res = await fetch('/api/puzzles/hint', {
   409→   409→        method: 'POST',
   410→   410→        headers: { 'Content-Type': 'application/json' },
   411→   411→        body: JSON.stringify({ puzzleId, row: selectedCell.row, col: selectedCell.col }),
   412→   412→      });
   413→   413→      if (!res.ok) {
   414→   414→        const err = await res.json().catch(() => ({}));
   415→   415→        toast.error((err as { error?: string }).error || 'Indice indisponible');
   416→   416→        return;
   417→   417→      }
   418→   418→      const data = await res.json();
   419→   419→      const { row, col } = selectedCell;
   420→   420→      setUserInputs((prev) => {
   421→   421→        const next = prev.map((r) => [...r]);
   422→   422→        next[row][col] = data.letter;
   423→   423→        return next;
   424→   424→      });
   425→   425→      setRevealedCells((prev) => {
   426→   426→        const next = new Set(prev);
   427→   427→        next.add(toCellKey(row, col));
   428→   428→        return next;
   429→   429→      });
   430→   430→      toast.success(`Lettre révélée : ${data.letter}`);
   431→   431→    } catch {
   432→   432→      toast.error('Erreur lors de la révélation');
   433→   433→    }
   434→   434→  }, [puzzleId, selectedCell, selectedPuzzle, isCompleted]);
   435→   435→
   436→   436→  const handleClear = useCallback(() => {
   437→   437→    if (!selectedPuzzle || isCompleted) return;
   438→   438→    setUserInputs(
   439→   439→      Array.from({ length: selectedPuzzle.rows }, () =>
   440→   440→        Array.from<null>({ length: selectedPuzzle.cols }).fill(null),
   441→   441→      ),
   442→   442→    );
   443→   443→    setCorrectCells(new Set());
   444→   444→    setIncorrectCells(new Set());
   445→   445→    setRevealedCells(new Set());
   446→   446→    setCompletedClues(new Set());
   447→   447→    toast.info('Grille effacée');
   448→   448→  }, [selectedPuzzle, isCompleted]);
   449→   449→
   450→   450→  const handleBack = useCallback(() => {
   451→   451→    setIsTimerRunning(false);
   452→   452→    setSelectedPuzzle(null);
   453→   453→    setPuzzleId(null);
   454→   454→    setCurrentView('selection');
