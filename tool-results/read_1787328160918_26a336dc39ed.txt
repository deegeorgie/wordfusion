   152→function StreakHeatmap({ streakData }: { streakData: StreakCompletion[] }) {
   153→  const days = useMemo(() => {
   154→    const today = new Date();
   155→    const result: { date: string; completed: boolean; count: number }[] = [];
   156→    for (let i = 29; i >= 0; i--) {
   157→      const d = new Date(today);
   158→      d.setDate(d.getDate() - i);
   159→      const dateStr = d.toISOString().slice(0, 10);
   160→      const completions = streakData.filter((c) => c.date === dateStr);
   161→      result.push({
   162→        date: dateStr,
   163→        completed: completions.length > 0,
   164→        count: completions.length,
   165→      });
   166→    }
   167→    return result;
   168→  }, [streakData]);
   169→
   170→  if (streakData.length === 0) return null;
   171→
   172→  return (
   173→    <div className="flex flex-wrap gap-1">
   174→      {days.map((day) => (
   175→        <div
   176→          key={day.date}
   177→          title={`${day.date}${day.completed ? ` — ${day.count} puzzle${day.count > 1 ? 's' : ''}` : ''}`}
   178→          className={cn(
   179→            'h-5 w-5 rounded-sm transition-colors',
   180→            day.completed
   181→              ? 'bg-emerald-500 dark:bg-emerald-400'
   182→              : day.date <= new Date().toISOString().slice(0, 10)
   183→                ? 'bg-muted'
   184→                : 'bg-transparent',
   185→          )}
   186→        />
   187→      ))}
   188→    </div>
   189→  );
   190→}

   194→function ConfettiOverlay() {
   195→  const particles = useMemo(() => {
   196→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   197→    return Array.from({ length: 50 }, (_, i) => ({
   198→      id: i,
   199→      x: Math.random() * 100,
   200→      y: Math.random() * -50 - 10,
   201→      color: colors[i % colors.length],
   202→      size: Math.random() * 8 + 4,
   203→      delay: Math.random() * 1.2,
   204→      duration: Math.random() * 2 + 2.5,
   205→    }));
   206→  }, []);
   207→
   208→  return (
   209→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   210→      {particles.map((p) => (
   211→        <motion.div
   212→          key={p.id}
   213→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   214→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   215→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   216→          className="absolute rounded-sm"
   217→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   218→        />
   219→      ))}
   220→    </div>
   221→  );
   222→}

   369→    if (dates.length === 0) return 0;

   378→      const prev = new Date(dates[i - 1]);

   380→      const diff = (prev.getTime() - curr.getTime()) / 86400000;

   474→      if (!selectedPuzzle) return;
   475→      const otherDir: 'across' | 'down' = dir === 'across' ? 'down' : 'across';
   476→
   477→      let found = selectedPuzzle.words.find((w: WordPlacement) => {
   478→        if (w.direction !== dir) return false;
   479→        return dir === 'across'
   480→          ? w.row === row && col >= w.col && col < w.col + w.length
   481→          : w.col === col && row >= w.row && row < w.row + w.length;
   482→      });
   483→
   484→      if (!found) {
   485→        found = selectedPuzzle.words.find((w: WordPlacement) => {
   486→          if (w.direction !== otherDir) return false;
   487→          return otherDir === 'across'
   488→            ? w.row === row && col >= w.col && col < w.col + w.length
   489→            : w.col === col && row >= w.row && row < w.row + w.length;
   490→        });
   491→        if (found) {
   492→          setDirection(otherDir);
   493→          setActiveClueDirection(otherDir);
   494→        }
   495→      } else {
   496→        setActiveClueDirection(dir);
   497→      }
   498→
   499→      if (found) setActiveClueNumber(found.clueNumber);
   500→    },
   501→    [selectedPuzzle],
   502→  );

   517→        setCorrectCellsManual(new Set());
   518→        setIncorrectCellsManual(new Set());
   519→      }
   520→    },
   521→    [isCompleted, isTimerRunning, autoCheck],
   522→  );
   523→

   525→    (row: number, col: number) => {

   527→      setSelectedCell({ row, col });
   528→      resolveActiveClue(row, col, direction);
   529→    },
   530→    [isCompleted, direction, resolveActiveClue],
   531→  );
   532→
   533→  const handleToggleDirection = useCallback(() => {

   538→      return next;
   539→    });
   540→  }, [selectedCell, resolveActiveClue]);
   541→
   542→  const handleSelectClue = useCallback(
   543→    (number: number, dir: 'across' | 'down') => {

   546→        (w: WordPlacement) => w.clueNumber === number && w.direction === dir,
   547→      );
   548→      if (!word) return;
   549→      setDirection(dir);
   550→      setActiveClueNumber(number);
   551→      setActiveClueDirection(dir);
   552→      setSelectedCell({ row: word.row, col: word.col });
   553→    },
   554→    [selectedPuzzle],
   555→  );
   556→
   557→  const handleCheck = useCallback(async () => {
   558→    if (!puzzleId || !selectedPuzzle || isCompleted) return;
   559→    setIsChecking(true);
   560→    try {
   561→      const res = await fetch('/api/puzzles/check', {
   562→        method: 'POST',
   563→        headers: { 'Content-Type': 'application/json' },
