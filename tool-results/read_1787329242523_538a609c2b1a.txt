   212→   212→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {
   213→   213→  const [newName, setNewName] = useState('');
   214→   214→  const [newIcon, setNewIcon] = useState('🏷️');
   215→   215→  const [newLanguage, setNewLanguage] = useState('fr');
   216→   216→  const [creating, setCreating] = useState(false);
   217→   217→
   218→   218→  // Edit state
   219→   219→  const [editingId, setEditingId] = useState<string | null>(null);
   220→   220→  const [editName, setEditName] = useState('');
   221→   221→  const [editIcon, setEditIcon] = useState('');
   222→   222→  const [editLanguage, setEditLanguage] = useState('');
   223→   223→  const [saving, setSaving] = useState(false);
   224→   224→
   225→   225→  // Delete state
   226→   226→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   227→   227→  const [deleting, setDeleting] = useState(false);
   228→   228→
   229→   229→  const handleCreate = async () => {
   230→   230→    if (!newName.trim()) return;
   231→   231→    setCreating(true);
   232→   232→    try {
   233→   233→      const res = await fetch('/api/categories', {
   234→   234→        method: 'POST',
   235→   235→        headers: { 'Content-Type': 'application/json' },
   236→   236→        body: JSON.stringify({ name: newName.trim(), icon: newIcon, language: newLanguage }),
   237→   237→      });
   238→   238→      if (!res.ok) {
   239→   239→        const err = await res.json().catch(() => ({}));
   240→   240→        throw new Error((err as { error?: string }).error || 'Erreur');
   241→   241→      }
   242→   242→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   243→   243→      setNewName('');
   244→   244→      setNewIcon('🏷️');
   245→   245→      onRefresh();
   246→   246→    } catch (err) {
   247→   247→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   248→   248→    } finally {
   249→   249→      setCreating(false);
   250→   250→    }
   251→   251→  };
   252→   252→
   253→   253→  const startEdit = (cat: CategoryItem) => {
   254→   254→    setEditingId(cat.id);
   255→   255→    setEditName(cat.name);
   256→   256→    setEditIcon(cat.icon || '🏷️');
   257→   257→    setEditLanguage(cat.language);
   258→   258→  };
   259→   259→
   260→   260→  const handleSaveEdit = async () => {
   261→   261→    if (!editingId || !editName.trim()) return;
   262→   262→    setSaving(true);
   263→   263→    try {
   264→   264→      const res = await fetch('/api/categories', {
   265→   265→        method: 'PUT',
   266→   266→        headers: { 'Content-Type': 'application/json' },
   267→   267→        body: JSON.stringify({ id: editingId, name: editName.trim(), icon: editIcon, language: editLanguage }),
   268→   268→      });
   269→   269→      if (!res.ok) {
   270→   270→        const err = await res.json().catch(() => ({}));
   271→   271→        throw new Error((err as { error?: string }).error || 'Erreur');
   272→   272→      }
   273→   273→      toast.success('Catégorie mise à jour');
   274→   274→      setEditingId(null);
   275→   275→      onRefresh();
   276→   276→    } catch (err) {
   277→   277→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   278→   278→    } finally {
   279→   279→      setSaving(false);
   280→   280→    }
   281→   281→  };
   282→   282→
   283→   283→  const handleDelete = async () => {
   284→   284→    if (!deleteTarget) return;
   285→   285→    setDeleting(true);
   286→   286→    try {
   287→   287→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   288→   288→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   289→   289→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   290→   290→      setDeleteTarget(null);
   291→   291→      onRefresh();
   292→   292→    } catch (err) {
   293→   293→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   294→   294→    } finally {
   295→   295→      setDeleting(false);
   296→   296→    }
   297→   297→  };
   298→   298→
   299→   299→  return (
   300→   300→    <>
   301→   301→      {/* Create new category row */}
   302→   302→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   303→   303→        <div className="w-16 sm:w-20 space-y-1">
   304→   304→          <Label className="text-xs text-muted-foreground">Icône</Label>
   305→   305→          <Input
   306→   306→            placeholder="🏷️"
   307→   307→            value={newIcon}
   308→   308→            onChange={(e) => setNewIcon(e.target.value)}
   309→   309→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   310→   310→            className="h-8 text-sm text-center"
   311→   311→            maxLength={4}
   312→   312→          />
   313→   313→        </div>
   314→   314→        <div className="flex-1 w-full sm:w-auto space-y-1">
   315→   315→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   316→   316→          <Input
   317→   317→            placeholder="Ex: Animaux, Sciences..."
   318→   318→            value={newName}
   319→   319→            onChange={(e) => setNewName(e.target.value)}
   320→   320→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   321→   321→            className="h-8 text-sm"
   322→   322→          />
   323→   323→        </div>
   324→   324→        <div className="w-full sm:w-28 space-y-1">
   325→   325→          <Label className="text-xs text-muted-foreground">Langue</Label>
   326→   326→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   327→   327→            <SelectTrigger className="h-8 text-sm">
   328→   328→              <SelectValue />
   329→   329→            </SelectTrigger>
   330→   330→            <SelectContent>
   331→   331→              <SelectItem value="fr">🇫🇷 Français</SelectItem>
   332→   332→              <SelectItem value="en">🇬🇧 English</SelectItem>
   333→   333→            </SelectContent>
   334→   334→          </Select>
   335→   335→        </div>
   336→   336→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   337→   337→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   338→   338→          Ajouter
   339→   339→        </Button>
   340→   340→      </div>
   341→   341→
   342→   342→      {/* Category list */}
   343→   343→      {categories.length === 0 ? (

   356→   356→              </TableRow>
   357→   357→            </TableHeader>
   358→   358→            <TableBody>
   359→   359→              {categories.map((cat) => (
   360→   360→                <TableRow key={cat.id}>
   361→   361→                  <TableCell className="font-medium pl-3">
   362→   362→                    {editingId === cat.id ? (
   363→   363→                      <div className="flex items-center gap-1.5">
   364→   364→                        <Input
   365→   365→                          value={editIcon}
   366→   366→                          onChange={(e) => setEditIcon(e.target.value)}
   367→   367→                          className="h-7 text-sm w-12 text-center"

   371→   371→                        <Input

   380→   380→                        <span className="text-sm">{cat.icon || '🏷️'}</span>

   384→   384→                  </TableCell>
