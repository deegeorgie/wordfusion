   151→   151→function ConfettiOverlay() {
   152→   152→  const particles = useMemo(() => {
   153→   153→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   154→   154→    return Array.from({ length: 50 }, (_, i) => ({
   155→   155→      id: i,
   156→   156→      x: Math.random() * 100,
   157→   157→      y: Math.random() * -50 - 10,
   158→   158→      color: colors[i % colors.length],
   159→   159→      size: Math.random() * 8 + 4,
   160→   160→      delay: Math.random() * 1.2,
   161→   161→      duration: Math.random() * 2 + 2.5,
   162→   162→    }));
   163→   163→  }, []);
   164→   164→
   165→   165→  return (
   166→   166→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   167→   167→      {particles.map((p) => (
   168→   168→        <motion.div
   169→   169→          key={p.id}
   170→   170→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   171→   171→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   172→   172→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   173→   173→          className="absolute rounded-sm"
   174→   174→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   175→   175→        />
   176→   176→      ))}
   177→   177→    </div>
   178→   178→  );
   179→   179→}
