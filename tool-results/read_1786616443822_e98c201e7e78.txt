    67→/** Create an empty grid of all-black cells */
    68→function createEmptyGrid(rows: number, cols: number): CrosswordCell[][] {
    69→  const grid: CrosswordCell[][] = [];
    70→  for (let r = 0; r < rows; r++) {
    71→    grid[r] = [];
    72→    for (let c = 0; c < cols; c++) {
    73→      grid[r][c] = { letter: "", number: null, isBlack: true, row: r, col: c };
    74→    }
    75→  }
    76→  return grid;
    77→}
    78→
    79→/** Deep-clone a grid */
    80→function cloneGrid(grid: CrosswordCell[][]): CrosswordCell[][] {
    81→  return grid.map((row) => row.map((cell) => ({ ...cell })));
    82→}

    84→/** Detect all words (2+ consecutive non-black cells with at least one letter) and assign numbers */
    85→function detectWordsAndNumbers(
    86→  grid: CrosswordCell[][]
    87→): { words: WordPlacement[]; clues: EditorClue[] } {
    88→  const rows = grid.length;
    89→  const cols = grid[0]?.length ?? 0;
    90→
    91→  // First pass: assign numbers
    92→  let clueNum = 1;
    93→  const numberMap = new Map<string, number>();
    94→
    95→  for (let r = 0; r < rows; r++) {
    96→    for (let c = 0; c < cols; c++) {
    97→      if (grid[r][c].isBlack) continue;
    98→      const key = `${r},${c}`;
    99→      if (numberMap.has(key)) continue;
   100→
   101→      const startsAcross =
   102→        (c === 0 || grid[r][c - 1].isBlack) &&
   103→        c + 1 < cols &&
   104→        !grid[r][c + 1].isBlack;
   105→      const startsDown =
   106→        (r === 0 || grid[r - 1][c].isBlack) &&
   107→        r + 1 < rows &&
   108→        !grid[r + 1][c].isBlack;
   109→
   110→      if (startsAcross || startsDown) {
   111→        numberMap.set(key, clueNum);
   112→        clueNum++;
   113→      }
   114→    }
   115→  }
   116→
   117→  // Build number grid for quick lookup
   118→  const numGrid: (number | null)[][] = [];
   119→  for (let r = 0; r < rows; r++) {
   120→    numGrid[r] = [];
   121→    for (let c = 0; c < cols; c++) {
   122→      numGrid[r][c] = grid[r][c].isBlack ? null : (numberMap.get(`${r},${c}`) ?? null);
   123→    }
   124→  }
   125→
   126→  const words: WordPlacement[] = [];
   127→  const clues: EditorClue[] = [];
   128→
   129→  // Detect across words
   130→  for (let r = 0; r < rows; r++) {
   131→    let c = 0;
   132→    while (c < cols) {
   133→      if (grid[r][c].isBlack) {
   134→        c++;
   135→        continue;
   136→      }
   137→      // Start of a horizontal run
   138→      const startC = c;
   139→      let wordStr = "";
   140→      while (c < cols && !grid[r][c].isBlack) {
   141→        wordStr += grid[r][c].letter;
   142→        c++;
   143→      }
   144→      if (wordStr.length >= 2) {
   145→        const num = numGrid[r][startC];
   146→        if (num !== null) {
   147→          words.push({
   148→            word: wordStr,
   149→            clueNumber: num,
   150→            direction: "across",
   151→            row: r,
   152→            col: startC,
   153→            length: wordStr.length,
   154→          });
   155→          clues.push({
   156→            number: num,
   157→            direction: "across",
   158→            text: "",
   159→            word: wordStr,
   160→            row: r,
   161→            col: startC,
   162→          });
   163→        }
   164→      }
   165→    }
   166→  }
   167→
   168→  // Detect down words
   169→  for (let c = 0; c < cols; c++) {
   170→    let r = 0;
   171→    while (r < rows) {
   172→      if (grid[r][c].isBlack) {
   173→        r++;
   174→        continue;
   175→      }
   176→      const startR = r;
   177→      let wordStr = "";
   178→      while (r < rows && !grid[r][c].isBlack) {
   179→        wordStr += grid[r][c].letter;
   180→        r++;
   181→      }
   182→      if (wordStr.length >= 2) {
   183→        const num = numGrid[startR][c];
   184→        if (num !== null) {
   185→          words.push({
   186→            word: wordStr,
   187→            clueNumber: num,
   188→            direction: "down",
   189→            row: startR,
   190→            col: c,
   191→            length: wordStr.length,
   192→          });
   193→          clues.push({
   194→            number: num,
   195→            direction: "down",
   196→            text: "",
   197→            word: wordStr,
   198→            row: startR,
   199→            col: c,
   200→          });
   201→        }
   202→      }
   203→    }
   204→  }
   205→
   206→  return { words, clues };
   207→}
   208→
   209→/** Update grid numbers from numberMap */
   210→function applyNumbersToGrid(
   211→  grid: CrosswordCell[][],
   212→  words: WordPlacement[]
   213→): CrosswordCell[][] {
   214→  const newGrid = cloneGrid(grid);
   215→  // Clear all numbers
   216→  for (let r = 0; r < newGrid.length; r++) {
   217→    for (let c = 0; c < newGrid[0].length; c++) {
   218→      newGrid[r][c].number = null;
   219→    }
   220→  }
   221→  // Set numbers from word placements
   222→  for (const w of words) {
   223→    if (newGrid[w.row]?.[w.col]) {
   224→      newGrid[w.row][w.col].number = w.clueNumber;
   225→    }
   226→  }
   227→  return newGrid;
   228→}

   232→export default function PuzzleEditor({
   233→  open,
   234→  onOpenChange,
   235→  editPuzzleId,
   236→  onSaved,
   237→}: PuzzleEditorProps) {
   238→  // ── State ────────────────────────────────────────────────────────
   239→  const [title, setTitle] = useState("");
   240→  const [description, setDescription] = useState("");
   241→  const [difficulty, setDifficulty] = useState<string>("2");
   242→  const [language, setLanguage] = useState<string>("fr");
   243→  const [categoryId, setCategoryId] = useState<string>(noneCategory);
   244→  const [categories, setCategories] = useState<CategoryOption[]>([]);
   245→  const [rowsInput, setRowsInput] = useState(10);
   246→  const [colsInput, setColsInput] = useState(10);
   247→  const [grid, setGrid] = useState<CrosswordCell[][]>(createEmptyGrid(10, 10));
   248→  const [selectedCell, setSelectedCell] = useState<{ row: number; col: number } | null>(null);
   249→  const [words, setWords] = useState<WordPlacement[]>([]);
   250→  const [clues, setClues] = useState<EditorClue[]>([]);
   251→  const [loading, setLoading] = useState(false);
   252→  const [saving, setSaving] = useState(false);
   253→  const [activeClueTab, setActiveClueTab] = useState<string>("across");
   254→
   255→  const gridRef = useRef<HTMLDivElement>(null);
   256→  const isEditing = !!editPuzzleId;
   257→
   258→  // ── Load categories ───────────────────────────────────────────
   259→  useEffect(() => {
   260→    if (!open) return;
   261→    fetch('/api/categories')
   262→      .then((res) => res.json())
   263→      .then((data) => {
   264→        setCategories(data.categories ?? []);
   265→      })
   266→      .catch(() => {});
   267→  }, [open]);
   268→
   269→  // ── Derived: number grid for display ─────────────────────────────
   270→  const numberGrid = useMemo(() => {
   271→    const ng: (number | null)[][] = [];
   272→    for (let r = 0; r < grid.length; r++) {
   273→      ng[r] = [];
   274→      for (let c = 0; c < (grid[0]?.length ?? 0); c++) {
   275→        ng[r][c] = grid[r][c].number;
   276→      }
   277→    }
   278→    return ng;
   279→  }, [grid]);
   280→
   281→  const acrossClues = useMemo(
   282→    () => clues.filter((c) => c.direction === "across").sort((a, b) => a.number - b.number),
   283→    [clues]
   284→  );
   285→  const downClues = useMemo(
   286→    () => clues.filter((c) => c.direction === "down").sort((a, b) => a.number - b.number),
   287→    [clues]
   288→  );
   289→
   290→  // ── Recompute words/clues from grid ──────────────────────────────
   291→  const recomputeWords = useCallback(
   292→    (g: CrosswordCell[][], preserveClueTexts: boolean = true) => {
   293→      const { words: newWords, clues: newClues } = detectWordsAndNumbers(g);
   294→      setWords(newWords);
   295→      if (preserveClueTexts) {
   296→        // Merge existing clue texts into new clues
   297→        const clueMap = new Map<string, string>();
   298→        for (const c of clues) {
   299→          clueMap.set(`${c.number}-${c.direction}`, c.text);
   300→        }
   301→        const merged = newClues.map((nc) => ({
   302→          ...nc,
   303→          text: clueMap.get(`${nc.number}-${nc.direction}`) ?? "",
   304→        }));
   305→        setClues(merged);
   306→      } else {
   307→        setClues(newClues);
   308→      }
   309→      // Update grid numbers
   310→      const updatedGrid = applyNumbersToGrid(g, newWords);
   311→      setGrid(updatedGrid);
   312→    },
   313→    [clues]
   314→  );
   315→
   316→  // ── Load puzzle for editing ──────────────────────────────────────
   317→  useEffect(() => {
   318→    if (!open) return;
   319→    if (!editPuzzleId) {
   320→      // New puzzle: reset everything
   321→      setTitle("");
   322→      setDescription("");
   323→      setDifficulty("2");
   324→      setLanguage("fr");
   325→      setCategoryId(noneCategory);
   326→      setRowsInput(10);
   327→      setColsInput(10);
   328→      const emptyGrid = createEmptyGrid(10, 10);
   329→      setGrid(emptyGrid);
   330→      setSelectedCell(null);
   331→      setWords([]);
   332→      setClues([]);
   333→      setLoading(false);
   334→      return;
   335→    }
   336→
   337→    // Fetch puzzle data
   338→    let cancelled = false;
   339→    setLoading(true);
   340→    fetch(`/api/puzzles/${editPuzzleId}`)
   341→      .then((res) => {
   342→        if (!res.ok) throw new Error("Erreur de chargement");
   343→        return res.json();
   344→      })
   345→      .then((data) => {
   346→        if (cancelled) return;
   347→        const p: CrosswordPuzzleData = data.puzzle;
   348→        setTitle(p.title);
   349→        setDescription(p.description ?? "");
   350→        setDifficulty(String(p.difficulty));
   351→        // Load language and category from puzzle metadata
   352→        if (data.language) setLanguage(data.language);
   353→        if (data.categoryId) setCategoryId(data.categoryId);
   354→        else setCategoryId(noneCategory);
   355→        setRowsInput(p.rows);
   356→        setColsInput(p.cols);
   357→        setGrid(cloneGrid(p.grid));
   358→        setSelectedCell(null);
   359→        // Build words and clues from loaded data
   360→        const editorClues: EditorClue[] = p.clues.map((cl) => {
   361→          const w = p.words.find(
   362→            (pw) => pw.clueNumber === cl.number && pw.direction === cl.direction
   363→          );
   364→          return {
   365→            ...cl,
   366→            word: w?.word ?? "",
   367→            row: w?.row ?? 0,
   368→            col: w?.col ?? 0,
   369→          };
   370→        });
   371→        setWords(p.words);
   372→        setClues(editorClues);
   373→        setLoading(false);
   374→      })
   375→      .catch((err) => {
   376→        if (cancelled) return;
   377→        console.error(err);
   378→        toast.error("Erreur lors du chargement du puzzle");
   379→        setLoading(false);
   380→      });
   381→
   382→    return () => {
   383→      cancelled = true;
   384→    };
   385→  }, [open, editPuzzleId]);
   386→
   387→  // ── Grid cell click ──────────────────────────────────────────────
   388→  const handleCellClick = useCallback(
   389→    (r: number, c: number) => {
   390→      const cell = grid[r]?.[c];
   391→      if (!cell) return;
   392→
   393→      // If clicking the SAME selected cell again, toggle black/white
   394→      if (
   395→        selectedCell &&
   396→        selectedCell.row === r &&
   397→        selectedCell.col === c
   398→      ) {
   399→        if (!cell.isBlack && cell.letter === "") {
   400→          // White empty cell → toggle to black
   401→          const newGrid = cloneGrid(grid);
   402→          newGrid[r][c] = { ...newGrid[r][c], isBlack: true, letter: "" };
   403→          recomputeWords(newGrid);
   404→          setSelectedCell(null);
   405→          return;
   406→        }
   407→        if (cell.isBlack) {
   408→          // Black cell → toggle to white
   409→          const newGrid = cloneGrid(grid);
   410→          newGrid[r][c] = { ...newGrid[r][c], isBlack: false, letter: "" };
   411→          recomputeWords(newGrid);
   412→          // Keep it selected so user can start typing
   413→          return;
   414→        }
   415→        // Has a letter — just keep selected
   416→        return;
   417→      }
   418→
   419→      // Clicking an unselected BLACK cell → toggle to white and select (draw mode)
   420→      if (cell.isBlack) {
   421→        const newGrid = cloneGrid(grid);
   422→        newGrid[r][c] = { ...newGrid[r][c], isBlack: false, letter: "" };
   423→        recomputeWords(newGrid);
   424→        setSelectedCell({ row: r, col: c });
   425→        return;
   426→      }
   427→
   428→      // Otherwise just select the cell
   429→      setSelectedCell({ row: r, col: c });
   430→    },
   431→    [grid, selectedCell, recomputeWords]
   432→  );
   433→
   434→  // ── Keyboard handler ─────────────────────────────────────────────
   435→  useEffect(() => {
   436→    if (!selectedCell) return;
   437→
   438→    const handleKeyDown = (e: KeyboardEvent) => {
   439→      const { row, col } = selectedCell;
   440→      const maxRow = grid.length - 1;
   441→      const maxCol = (grid[0]?.length ?? 1) - 1;
   442→
   443→      // Prevent default for keys we handle
   444→      if (
   445→        e.key.startsWith("Arrow") ||
   446→        e.key === "Backspace" ||
   447→        e.key === "Delete" ||
   448→        (/^[a-zA-ZéèêëàâùûîïôçÉÈÊËÀÂÙÛÎÏÔÇ]$/.test(e.key))
   449→      ) {
   450→        e.preventDefault();
   451→      }
   452→
   453→      if (e.key === "ArrowUp") {
   454→        setSelectedCell((prev) =>
   455→          prev && prev.row > 0 ? { row: prev.row - 1, col: prev.col } : prev
   456→        );
   457→      } else if (e.key === "ArrowDown") {
   458→        setSelectedCell((prev) =>
   459→          prev && prev.row < maxRow ? { row: prev.row + 1, col: prev.col } : prev
   460→        );
   461→      } else if (e.key === "ArrowLeft") {
   462→        setSelectedCell((prev) =>
   463→          prev && prev.col > 0 ? { row: prev.row, col: prev.col - 1 } : prev
   464→        );
   465→      } else if (e.key === "ArrowRight") {
   466→        setSelectedCell((prev) =>
   467→          prev && prev.col < maxCol ? { row: prev.row, col: prev.col + 1 } : prev
   468→        );
   469→      } else if (e.key === "Backspace" || e.key === "Delete") {
   470→        const cell = grid[row]?.[col];
   471→        if (cell && !cell.isBlack && cell.letter !== "") {
   472→          const newGrid = cloneGrid(grid);
   473→          newGrid[row][col] = { ...newGrid[row][col], letter: "" };
   474→          recomputeWords(newGrid);
   475→        } else if (e.key === "Backspace") {
   476→          // Move back
   477→          if (col > 0) {
   478→            setSelectedCell({ row, col: col - 1 });
   479→          } else if (row > 0) {
   480→            setSelectedCell({ row: row - 1, col: maxCol });
   481→          }
   482→        }
   483→      } else if (/^[a-zA-Z]$/.test(e.key)) {
   484→        const cell = grid[row]?.[col];
   485→        if (cell && !cell.isBlack) {
   486→          const newGrid = cloneGrid(grid);
   487→          newGrid[row][col] = {
   488→            ...newGrid[row][col],
   489→            letter: e.key.toUpperCase(),
   490→          };
   491→          recomputeWords(newGrid);
   492→          // Auto-advance
   493→          if (col < maxCol && !newGrid[row][col + 1].isBlack) {
   494→            setSelectedCell({ row, col: col + 1 });
   495→          } else if (row < maxRow) {
   496→            // Try to move to next row
   497→            setSelectedCell({ row: row + 1, col: 0 });
   498→          }
   499→        }
   500→      } else if (e.key === "Escape") {
   501→        setSelectedCell(null);
   502→      }
   503→    };
   504→
   505→    window.addEventListener("keydown", handleKeyDown);
   506→    return () => window.removeEventListener("keydown", handleKeyDown);
   507→  }, [selectedCell, grid, recomputeWords]);
   508→
   509→  // ── Resize grid ──────────────────────────────────────────────────
   510→  const handleResizeGrid = useCallback(() => {
   511→    const r = Math.max(2, Math.min(20, rowsInput));
   512→    const c = Math.max(2, Math.min(20, colsInput));
   513→    const newGrid = createEmptyGrid(r, c);
   514→    // Preserve existing data
   515→    for (let row = 0; row < Math.min(r, grid.length); row++) {
   516→      for (let col = 0; col < Math.min(c, (grid[0]?.length ?? 0)); col++) {
   517→        newGrid[row][col] = { ...grid[row][col], row, col };
   518→      }
   519→    }
   520→    setRowsInput(r);
   521→    setColsInput(c);
   522→    setGrid(newGrid);
   523→    setSelectedCell(null);
   524→    recomputeWords(newGrid);
   525→  }, [rowsInput, colsInput, grid, recomputeWords]);
   526→
   527→  // ── Clear all ────────────────────────────────────────────────────
   528→  const handleClearAll = useCallback(() => {
   529→    const newGrid = createEmptyGrid(grid.length, grid[0]?.length ?? 0);
   530→    setGrid(newGrid);
   531→    setSelectedCell(null);
   532→    setWords([]);
   533→    setClues([]);
   534→  }, [grid]);
   535→
   536→  // ── Clear letters only ───────────────────────────────────────────
   537→  const handleClearLetters = useCallback(() => {
   538→    const newGrid = cloneGrid(grid);
   539→    for (let r = 0; r < newGrid.length; r++) {
   540→      for (let c = 0; c < newGrid[0].length; c++) {
   541→        newGrid[r][c].letter = "";
   542→      }
   543→    }
   544→    setGrid(newGrid);
   545→    recomputeWords(newGrid);
   546→  }, [grid, recomputeWords]);
   547→
   548→  // ── Update clue text ─────────────────────────────────────────────
   549→  const handleClueTextChange = useCallback(
   550→    (number: number, direction: "across" | "down", text: string) => {
   551→      setClues((prev) =>
   552→        prev.map((c) =>
   553→          c.number === number && c.direction === direction ? { ...c, text } : c
   554→        )
   555→      );
   556→    },
   557→    []
   558→  );
   559→
   560→  // ── Update word text → update grid cells ─────────────────────────
   561→  const handleWordChange = useCallback(
   562→    (
   563→      number: number,
   564→      direction: "across" | "down",
   565→      newWord: string,
   566→      row: number,
   567→      col: number
   568→    ) => {
   569→      const newGrid = cloneGrid(grid);
   570→      const oldWord = words.find(
   571→        (w) => w.clueNumber === number && w.direction === direction
   572→      );
   573→      const oldLen = oldWord?.length ?? newWord.length;
   574→
   575→      // Update grid cells
   576→      for (let i = 0; i < Math.max(oldLen, newWord.length); i++) {
   577→        const r = direction === "across" ? row : row + i;
   578→        const c = direction === "across" ? col + i : col;
   579→        if (r < newGrid.length && c < (newGrid[0]?.length ?? 0)) {
   580→          if (i < newWord.length) {
   581→            newGrid[r][c] = {
   582→              ...newGrid[r][c],
   583→              letter: newWord[i].toUpperCase(),
   584→              isBlack: false,
   585→            };
   586→          } else if (i >= newWord.length && i < oldLen) {
   587→            // Clear extra letters but keep cell white
   588→            newGrid[r][c] = { ...newGrid[r][c], letter: "" };
   589→          }
   590→        }
   591→      }
   592→
   593→      setGrid(newGrid);
   594→      recomputeWords(newGrid);
   595→    },
   596→    [grid, words, recomputeWords]
   597→  );
   598→
   599→  // ── Save ─────────────────────────────────────────────────────────
