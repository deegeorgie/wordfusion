    68→function formatFrenchDate(): string {
    69→  return new Date().toLocaleDateString('fr-FR', {
    70→    weekday: 'long',
    71→    year: 'numeric',
    72→    month: 'long',
    73→    day: 'numeric',
    74→  });
    75→}

    77→function formatTimer(seconds: number): string {
    78→  const m = Math.floor(seconds / 60);
    79→  const s = seconds % 60;
    80→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    81→}

    83→function difficultyLabel(d: number): string {
    84→  if (d <= 1) return 'Facile';
    85→  if (d <= 2) return 'Moyen';
    86→  if (d <= 3) return 'Difficile';
    87→  return 'Expert';
    88→}

    90→function difficultyColor(d: number): string {
    91→  if (d <= 1) return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400';
    92→  if (d <= 2) return 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400';
    93→  if (d <= 3) return 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400';
    94→  return 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400';
    95→}

    99→function ConfettiOverlay() {
   100→  const particles = useMemo(() => {
   101→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   102→    return Array.from({ length: 50 }, (_, i) => ({
   103→      id: i,
   104→      x: Math.random() * 100,
   105→      y: Math.random() * -50 - 10,
   106→      color: colors[i % colors.length],
   107→      size: Math.random() * 8 + 4,
   108→      delay: Math.random() * 1.2,
   109→      duration: Math.random() * 2 + 2.5,
   110→    }));
   111→  }, []);
   112→
   113→  return (
   114→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   115→      {particles.map((p) => (
   116→        <motion.div
   117→          key={p.id}
   118→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   119→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   120→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   121→          className="absolute rounded-sm"
   122→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   123→        />
   124→      ))}
   125→    </div>
   126→  );
   127→}

   152→export default function Home() {
   153→  // ── View state ──────────────────────────────────────────────────────
   154→  const [currentView, setCurrentView] = useState<'selection' | 'playing'>('selection');
   155→  const [adminOpen, setAdminOpen] = useState(false);
   156→  const [dailyPuzzles, setDailyPuzzles] = useState<PuzzleSummary[]>([]);
   157→  const [isLoadingPuzzles, setIsLoadingPuzzles] = useState(true);
   158→
   159→  // ── Puzzle data ─────────────────────────────────────────────────────
   160→  const [selectedPuzzle, setSelectedPuzzle] = useState<CrosswordPuzzleData | null>(null);
   161→  const [puzzleId, setPuzzleId] = useState<string | null>(null);
   162→  const [userInputs, setUserInputs] = useState<(string | null)[][]>([]);
   163→  const [selectedCell, setSelectedCell] = useState<{ row: number; col: number } | null>(null);
   164→  const [direction, setDirection] = useState<'across' | 'down'>('across');
   165→  const [activeClueNumber, setActiveClueNumber] = useState<number | null>(null);
   166→  const [activeClueDirection, setActiveClueDirection] = useState<'across' | 'down'>('across');
   167→
   168→  // ── Cell status ─────────────────────────────────────────────────────
   169→  const [revealedCells, setRevealedCells] = useState<Set<string>>(new Set());
   170→  const [correctCellsManual, setCorrectCellsManual] = useState<Set<string>>(new Set());
   171→  const [incorrectCellsManual, setIncorrectCellsManual] = useState<Set<string>>(new Set());
   172→  const [completedClues, setCompletedClues] = useState<Set<string>>(new Set());
   173→  const [autoCheck, setAutoCheck] = useState(true); // Real-time letter validation
   174→
   175→  // ── Timer / completion ──────────────────────────────────────────────
   176→  const [timer, setTimer] = useState(0);
   177→  const [isTimerRunning, setIsTimerRunning] = useState(false);
   178→  const [isCompleted, setIsCompleted] = useState(false);
   179→  const [isChecking, setIsChecking] = useState(false);
   180→  const [isLoadingPuzzle, setIsLoadingPuzzle] = useState(false);
   181→
   182→  // ── Timer interval ──────────────────────────────────────────────────
   183→  useEffect(() => {
   184→    if (!isTimerRunning) return;
   185→    const id = setInterval(() => setTimer((t) => t + 1), 1000);
   186→    return () => clearInterval(id);
   187→  }, [isTimerRunning]);
   188→
   189→  // ── Load daily puzzles on mount ─────────────────────────────────────
   190→  useEffect(() => {
   191→    let cancelled = false;
   192→    (async () => {
   193→      setIsLoadingPuzzles(true);
   194→      try {
   195→        const res = await fetch('/api/puzzles/daily');
   196→        if (!res.ok) throw new Error();
   197→        const data = await res.json();
   198→        if (!cancelled) setDailyPuzzles(data.puzzles ?? []);
   199→      } catch {
   200→        if (!cancelled) toast.error('Impossible de charger les puzzles du jour');
   201→      } finally {
   202→        if (!cancelled) setIsLoadingPuzzles(false);
   203→      }
   204→    })();
   205→    return () => { cancelled = true; };
   206→  }, []);
   207→
   208→  // ── Fetch a specific puzzle ─────────────────────────────────────────
   209→  const fetchPuzzle = useCallback(async (id: string) => {
   210→    setIsLoadingPuzzle(true);
   211→    setCurrentView('playing');
   212→    try {
   213→      const res = await fetch(`/api/puzzles/${id}`);
   214→      if (!res.ok) throw new Error();
   215→      const data = await res.json();
   216→      const puzzle: CrosswordPuzzleData = data.puzzle;
   217→
   218→      const inputs: (string | null)[][] = Array.from({ length: puzzle.rows }, () =>
   219→        Array.from<null>({ length: puzzle.cols }).fill(null),
   220→      );
   221→
   222→      // Restore saved progress
   223→      if (data.progress?.cells) {
   224→        for (const cell of data.progress.cells) {
   225→          const { row, col, letter } = cell as { row: number; col: number; letter: string };
   226→          if (row >= 0 && row < puzzle.rows && col >= 0 && col < puzzle.cols) {
   227→            inputs[row][col] = letter;
   228→          }
   229→        }
   230→      }
   231→
   232→      setSelectedPuzzle(puzzle);
   233→      setPuzzleId(id);
   234→      setUserInputs(inputs);
   235→      setSelectedCell(null);
   236→      setDirection('across');
   237→      setActiveClueNumber(null);
   238→      setActiveClueDirection('across');
   239→      setRevealedCells(new Set());
   240→      setCorrectCellsManual(new Set());
   241→      setIncorrectCellsManual(new Set());
   242→      setCompletedClues(new Set());
   243→      setTimer(0);
   244→      setIsTimerRunning(false);
   245→      setIsCompleted(false);
   246→      setIsChecking(false);
   247→    } catch {
   248→      toast.error('Impossible de charger le puzzle');
   249→      setCurrentView('selection');
   250→    } finally {
   251→      setIsLoadingPuzzle(false);
   252→    }
   253→  }, []);
   254→
   255→  // ── Get active word cells for highlighting ──────────────────────────
   256→  const getActiveWordCells = useCallback(
   257→    (row: number, col: number, dir: 'across' | 'down') => {
   258→      if (!selectedPuzzle) return [];
   259→      const word = selectedPuzzle.words.find((w: WordPlacement) => {
   260→        if (w.direction !== dir) return false;
   261→        return dir === 'across'
   262→          ? w.row === row && col >= w.col && col < w.col + w.length
   263→          : w.col === col && row >= w.row && row < w.row + w.length;
   264→      });
   265→      if (!word) return [];
   266→      return Array.from({ length: word.length }, (_, i) => ({
   267→        row: dir === 'across' ? word.row : word.row + i,
   268→        col: dir === 'across' ? word.col + i : word.col,
   269→      }));
   270→    },
   271→    [selectedPuzzle],
   272→  );
   273→
   274→  // ── Resolve which clue is active for a cell ────────────────────────
   275→  const resolveActiveClue = useCallback(
   276→    (row: number, col: number, dir: 'across' | 'down') => {
   277→      if (!selectedPuzzle) return;
   278→      const otherDir: 'across' | 'down' = dir === 'across' ? 'down' : 'across';
   279→
   280→      let found = selectedPuzzle.words.find((w: WordPlacement) => {
   281→        if (w.direction !== dir) return false;
   282→        return dir === 'across'
   283→          ? w.row === row && col >= w.col && col < w.col + w.length
   284→          : w.col === col && row >= w.row && row < w.row + w.length;
   285→      });
   286→
   287→      if (!found) {
   288→        found = selectedPuzzle.words.find((w: WordPlacement) => {
   289→          if (w.direction !== otherDir) return false;
   290→          return otherDir === 'across'
   291→            ? w.row === row && col >= w.col && col < w.col + w.length
   292→            : w.col === col && row >= w.row && row < w.row + w.length;
   293→        });
   294→        if (found) {
   295→          setDirection(otherDir);
   296→          setActiveClueDirection(otherDir);
   297→        }
   298→      } else {
   299→        setActiveClueDirection(dir);
   300→      }
   301→
   302→      if (found) setActiveClueNumber(found.clueNumber);
   303→    },
   304→    [selectedPuzzle],
   305→  );
   306→
   307→  // ── Handlers ─────────────────────────────────────────────────────────
   308→
   309→  const handleCellChange = useCallback(
   310→    (row: number, col: number, value: string) => {
   311→      if (isCompleted) return;
   312→      setUserInputs((prev) => {
   313→        const next = prev.map((r) => [...r]);
   314→        next[row][col] = value || null;
   315→        return next;
   316→      });
   317→      if (!isTimerRunning && value) setIsTimerRunning(true);
   318→      // Only clear manual check highlights on new input (auto-check handles its own)
   319→      if (!autoCheck) {
   320→        setCorrectCellsManual(new Set());
   321→        setIncorrectCellsManual(new Set());
   322→      }
   323→    },
   324→    [isCompleted, isTimerRunning, autoCheck],
   325→  );
   326→
   327→  const handleSelectCell = useCallback(
   328→    (row: number, col: number) => {
   329→      if (isCompleted) return;
   330→      setSelectedCell({ row, col });
   331→      resolveActiveClue(row, col, direction);
   332→    },
   333→    [isCompleted, direction, resolveActiveClue],
   334→  );
   335→
   336→  const handleToggleDirection = useCallback(() => {
   337→    setDirection((prev) => {
   338→      const next: 'across' | 'down' = prev === 'across' ? 'down' : 'across';
   339→      setActiveClueDirection(next);
   340→      if (selectedCell) resolveActiveClue(selectedCell.row, selectedCell.col, next);
   341→      return next;
   342→    });
   343→  }, [selectedCell, resolveActiveClue]);
   344→
   345→  const handleSelectClue = useCallback(
   346→    (number: number, dir: 'across' | 'down') => {
   347→      if (!selectedPuzzle) return;
   348→      const word = selectedPuzzle.words.find(
   349→        (w: WordPlacement) => w.clueNumber === number && w.direction === dir,
   350→      );
   351→      if (!word) return;
   352→      setDirection(dir);
   353→      setActiveClueNumber(number);
   354→      setActiveClueDirection(dir);
   355→      setSelectedCell({ row: word.row, col: word.col });
   356→    },
   357→    [selectedPuzzle],
   358→  );
   359→
   360→  const handleCheck = useCallback(async () => {
   361→    if (!puzzleId || !selectedPuzzle || isCompleted) return;
   362→    setIsChecking(true);
   363→    try {
   364→      const res = await fetch('/api/puzzles/check', {
   365→        method: 'POST',
   366→        headers: { 'Content-Type': 'application/json' },
   367→        body: JSON.stringify({ puzzleId, userInputs }),
   368→      });
   369→      if (!res.ok) throw new Error();
   370→      const data = await res.json();
   371→
   372→      const incorrect = new Set<string>(
   373→        (data.incorrectCells ?? []).map((c: { row: number; col: number }) => toCellKey(c.row, c.col)),
   374→      );
   375→      setIncorrectCellsManual(incorrect);
   376→
   377→      const correct = new Set<string>();
   378→      for (let r = 0; r < selectedPuzzle.rows; r++) {
   379→        for (let c = 0; c < selectedPuzzle.cols; c++) {
   380→          if (userInputs[r][c] && !incorrect.has(toCellKey(r, c))) correct.add(toCellKey(r, c));
   381→        }
   382→      }
   383→      setCorrectCellsManual(correct);
   384→
   385→      // Update completed clues
   386→      const newCompleted = new Set<string>(completedClues);
   387→      for (const word of selectedPuzzle.words) {
   388→        const ok = Array.from({ length: word.length }, (_, i) => {
   389→          const r = word.direction === 'across' ? word.row : word.row + i;
   390→          const c = word.direction === 'across' ? word.col + i : word.col;
   391→          return !!userInputs[r]?.[c] && !incorrect.has(toCellKey(r, c));
   392→        }).every(Boolean);
   393→        if (ok) newCompleted.add(clueKey(word.clueNumber, word.direction));
   394→      }
   395→      setCompletedClues(newCompleted);
   396→
   397→      if (data.completionPercent === 100 && data.correct) {
   398→        setIsCompleted(true);
   399→        setIsTimerRunning(false);
   400→        toast.success('Bravo ! Puzzle complété !', { duration: 6000 });
   401→      } else if (incorrect.size === 0) {
   402→        toast.success('Tout est correct pour le moment !');
   403→      } else {
   404→        toast.error(`${incorrect.size} lettre${incorrect.size > 1 ? 's' : ''} incorrecte${incorrect.size > 1 ? 's' : ''}`);
   405→      }
   406→    } catch {
   407→      toast.error('Erreur lors de la vérification');
   408→    } finally {
   409→      setIsChecking(false);
   410→    }
   411→  }, [puzzleId, selectedPuzzle, userInputs, isCompleted, completedClues]);
   412→
   413→  const handleHint = useCallback(async () => {
   414→    if (!puzzleId || !selectedCell || !selectedPuzzle || isCompleted) return;
   415→    try {
   416→      const res = await fetch('/api/puzzles/hint', {
   417→        method: 'POST',
   418→        headers: { 'Content-Type': 'application/json' },
   419→        body: JSON.stringify({ puzzleId, row: selectedCell.row, col: selectedCell.col }),
   420→      });
   421→      if (!res.ok) {
   422→        const err = await res.json().catch(() => ({}));
   423→        toast.error((err as { error?: string }).error || 'Indice indisponible');
   424→        return;
   425→      }
   426→      const data = await res.json();
   427→      const { row, col } = selectedCell;
   428→      setUserInputs((prev) => {
   429→        const next = prev.map((r) => [...r]);
   430→        next[row][col] = data.letter;
   431→        return next;
   432→      });
   433→      setRevealedCells((prev) => {
   434→        const next = new Set(prev);
   435→        next.add(toCellKey(row, col));
   436→        return next;
   437→      });
   438→      toast.success(`Lettre révélée : ${data.letter}`);
   439→    } catch {
   440→      toast.error('Erreur lors de la révélation');
   441→    }
   442→  }, [puzzleId, selectedCell, selectedPuzzle, isCompleted]);
   443→
   444→  const handleClear = useCallback(() => {
   445→    if (!selectedPuzzle || isCompleted) return;
   446→    setUserInputs(
   447→      Array.from({ length: selectedPuzzle.rows }, () =>
   448→        Array.from<null>({ length: selectedPuzzle.cols }).fill(null),
   449→      ),
   450→    );
   451→    setCorrectCellsManual(new Set());
   452→    setIncorrectCellsManual(new Set());
   453→    setRevealedCells(new Set());
   454→    setCompletedClues(new Set());
   455→    toast.info('Grille effacée');
   456→  }, [selectedPuzzle, isCompleted]);
   457→
   458→  const handleBack = useCallback(() => {
   459→    setIsTimerRunning(false);
   460→    setSelectedPuzzle(null);
   461→    setPuzzleId(null);
   462→    setCurrentView('selection');
   463→  }, []);
   464→
   465→  // ── Derived values ───────────────────────────────────────────────────
   466→
   467→  // Real-time auto-validation: compare each filled cell against the answer
   468→  const autoCorrectCells = useMemo(() => {
   469→    const s = new Set<string>();
   470→    if (!selectedPuzzle || !autoCheck) return s;
   471→    for (let r = 0; r < selectedPuzzle.rows; r++) {
   472→      for (let c = 0; c < selectedPuzzle.cols; c++) {
   473→        const answer = selectedPuzzle.grid[r]?.[c]?.letter;
   474→        const input = userInputs[r]?.[c];
   475→        if (answer && input && input.toUpperCase() === answer.toUpperCase()) {
   476→          s.add(toCellKey(r, c));
   477→        }
   478→      }
   479→    }
   480→    return s;
   481→  }, [selectedPuzzle, userInputs, autoCheck]);
   482→
   483→  const autoIncorrectCells = useMemo(() => {
   484→    const s = new Set<string>();
   485→    if (!selectedPuzzle || !autoCheck) return s;
   486→    for (let r = 0; r < selectedPuzzle.rows; r++) {
   487→      for (let c = 0; c < selectedPuzzle.cols; c++) {
   488→        const answer = selectedPuzzle.grid[r]?.[c]?.letter;
   489→        const input = userInputs[r]?.[c];
   490→        if (answer && input && input.toUpperCase() !== answer.toUpperCase()) {
   491→          s.add(toCellKey(r, c));
   492→        }
   493→      }
   494→    }
   495→    return s;
   496→  }, [selectedPuzzle, userInputs, autoCheck]);
   497→
   498→  // Merge auto-check + manual check sets (manual takes priority if conflicting)
   499→  const correctCells = useMemo(() => {
   500→    if (!autoCheck) return correctCellsManual;
   501→    return new Set([...autoCorrectCells, ...correctCellsManual]);
   502→  }, [autoCheck, autoCorrectCells, correctCellsManual]);
   503→
   504→  const incorrectCells = useMemo(() => {
   505→    if (!autoCheck) return incorrectCellsManual;
   506→    return new Set([...autoIncorrectCells, ...incorrectCellsManual]);
   507→  }, [autoCheck, autoIncorrectCells, incorrectCellsManual]);
   508→
   509→  const activeWordCells = useMemo(() => {
   510→    if (!selectedCell || !selectedPuzzle) return [];
   511→    return getActiveWordCells(selectedCell.row, selectedCell.col, direction);
   512→  }, [selectedCell, selectedPuzzle, direction, getActiveWordCells]);
   513→
   514→  const activeClueText = useMemo(() => {
   515→    if (!selectedPuzzle || activeClueNumber === null) return null;
   516→    return selectedPuzzle.clues.find(
   517→      (c: Clue) => c.number === activeClueNumber && c.direction === activeClueDirection,
   518→    ) ?? null;
   519→  }, [selectedPuzzle, activeClueNumber, activeClueDirection]);
   520→
   521→  // ══════════════════════════════════════════════════════════════════════
   522→  // RENDER
   523→  // ══════════════════════════════════════════════════════════════════════
   524→
   525→  return (
   526→    <div className="min-h-screen flex flex-col bg-background">
   527→      {/* ═════════════════ PUZZLE SELECTION VIEW ═════════════════ */}
   528→      {currentView === 'selection' && (
   529→        <div className="flex flex-1 flex-col">
   530→          {/* Subtle crossword grid background */}
   531→          <div
   532→            className="pointer-events-none fixed inset-0 -z-10 opacity-[0.025]"
