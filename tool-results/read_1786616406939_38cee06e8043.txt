   140→function formatDate(dateStr: string | null): string {
   141→  if (!dateStr) return '—';
   142→  try {
   143→    return format(new Date(dateStr), 'dd MMM yyyy', { locale: fr });
   144→  } catch {
   145→    return '—';
   146→  }
   147→}

   191→// --- Category Management Sub-component ---

   193→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {
   194→  const [newName, setNewName] = useState('');
   195→  const [newLanguage, setNewLanguage] = useState('fr');
   196→  const [creating, setCreating] = useState(false);
   197→
   198→  // Edit state
   199→  const [editingId, setEditingId] = useState<string | null>(null);
   200→  const [editName, setEditName] = useState('');
   201→  const [editLanguage, setEditLanguage] = useState('');
   202→  const [saving, setSaving] = useState(false);
   203→
   204→  // Delete state
   205→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   206→  const [deleting, setDeleting] = useState(false);
   207→
   208→  const handleCreate = async () => {
   209→    if (!newName.trim()) return;
   210→    setCreating(true);
   211→    try {
   212→      const res = await fetch('/api/categories', {
   213→        method: 'POST',
   214→        headers: { 'Content-Type': 'application/json' },
   215→        body: JSON.stringify({ name: newName.trim(), language: newLanguage }),
   216→      });
   217→      if (!res.ok) {
   218→        const err = await res.json().catch(() => ({}));
   219→        throw new Error((err as { error?: string }).error || 'Erreur');
   220→      }
   221→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   222→      setNewName('');
   223→      onRefresh();
   224→    } catch (err) {
   225→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   226→    } finally {
   227→      setCreating(false);
   228→    }
   229→  };
   230→
   231→  const startEdit = (cat: CategoryItem) => {
   232→    setEditingId(cat.id);
   233→    setEditName(cat.name);
   234→    setEditLanguage(cat.language);
   235→  };
   236→
   237→  const handleSaveEdit = async () => {
   238→    if (!editingId || !editName.trim()) return;
   239→    setSaving(true);
   240→    try {
   241→      const res = await fetch('/api/categories', {
   242→        method: 'PUT',
   243→        headers: { 'Content-Type': 'application/json' },
   244→        body: JSON.stringify({ id: editingId, name: editName.trim(), language: editLanguage }),
   245→      });
   246→      if (!res.ok) {
   247→        const err = await res.json().catch(() => ({}));
   248→        throw new Error((err as { error?: string }).error || 'Erreur');
   249→      }
   250→      toast.success('Catégorie mise à jour');
   251→      setEditingId(null);
   252→      onRefresh();
   253→    } catch (err) {
   254→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   255→    } finally {
   256→      setSaving(false);
   257→    }
   258→  };
   259→
   260→  const handleDelete = async () => {
   261→    if (!deleteTarget) return;
   262→    setDeleting(true);
   263→    try {
   264→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   265→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   266→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   267→      setDeleteTarget(null);
   268→      onRefresh();
   269→    } catch (err) {
   270→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   271→    } finally {
   272→      setDeleting(false);
   273→    }
   274→  };
   275→
   276→  return (
   277→    <>
   278→      {/* Create new category row */}
   279→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   280→        <div className="flex-1 w-full sm:w-auto space-y-1">
   281→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   282→          <Input
   283→            placeholder="Ex: Animaux, Sciences..."
   284→            value={newName}
   285→            onChange={(e) => setNewName(e.target.value)}
   286→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   287→            className="h-8 text-sm"
   288→          />
   289→        </div>
   290→        <div className="w-full sm:w-28 space-y-1">
   291→          <Label className="text-xs text-muted-foreground">Langue</Label>
   292→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   293→            <SelectTrigger className="h-8 text-sm">
   294→              <SelectValue />
   295→            </SelectTrigger>
   296→            <SelectContent>
   297→              <SelectItem value="fr">🇫🇷 Français</SelectItem>
   298→              <SelectItem value="en">🇬🇧 English</SelectItem>
   299→            </SelectContent>
   300→          </Select>
   301→        </div>
   302→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   303→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   304→          Ajouter
   305→        </Button>
   306→      </div>
   307→
   308→      {/* Category list */}
   309→      {categories.length === 0 ? (
   310→        <div className="text-center py-6 text-muted-foreground text-sm">
   311→          Aucune catégorie. Créez-en une ci-dessus.
   312→        </div>
   313→      ) : (
   314→        <div className="rounded-md border max-h-64 overflow-y-auto">
   315→          <Table>
   316→            <TableHeader>
   317→              <TableRow className="hover:bg-transparent">
   318→                <TableHead className="pl-3">Nom</TableHead>
   319→                <TableHead>Langue</TableHead>
   320→                <TableHead>Puzzles</TableHead>
   321→                <TableHead className="text-right pr-3">Actions</TableHead>
   322→              </TableRow>
   323→            </TableHeader>
   324→            <TableBody>
   325→              {categories.map((cat) => (
   326→                <TableRow key={cat.id}>
   327→                  <TableCell className="font-medium pl-3">
   328→                    {editingId === cat.id ? (
   329→                      <Input
   330→                        value={editName}
   331→                        onChange={(e) => setEditName(e.target.value)}
   332→                        onKeyDown={(e) => e.key === 'Enter' && handleSaveEdit()}
   333→                        className="h-7 text-sm w-full"
   334→                        autoFocus
   335→                      />
   336→                    ) : (
   337→                      <span className="flex items-center gap-1.5">
   338→                        <Tag className="size-3 text-muted-foreground" />
   339→                        {cat.name}
   340→                      </span>
   341→                    )}
   342→                  </TableCell>
   343→                  <TableCell>
   344→                    {editingId === cat.id ? (
   345→                      <Select value={editLanguage} onValueChange={setEditLanguage}>
   346→                        <SelectTrigger className="h-7 text-xs w-24">
   347→                          <SelectValue />
   348→                        </SelectTrigger>
   349→                        <SelectContent>
   350→                          <SelectItem value="fr">🇫🇷 FR</SelectItem>
   351→                          <SelectItem value="en">🇬🇧 EN</SelectItem>
   352→                        </SelectContent>
   353→                      </Select>
   354→                    ) : (
   355→                      <Badge variant="outline" className="text-xs">
   356→                        {languageLabel(cat.language)}
   357→                      </Badge>
   358→                    )}
   359→                  </TableCell>
   360→                  <TableCell className="text-muted-foreground text-xs">
   361→                    {cat._count.puzzles}
   362→                  </TableCell>
   363→                  <TableCell className="text-right pr-3">
   364→                    <div className="flex items-center justify-end gap-1">
   365→                      {editingId === cat.id ? (
   366→                        <>
   367→                          <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={() => setEditingId(null)}>
   368→                            Annuler
   369→                          </Button>
   370→                          <Button size="sm" className="h-7 text-xs gap-1" onClick={handleSaveEdit} disabled={saving}>
   371→                            {saving ? <Loader2 className="size-3 animate-spin" /> : <Save className="size-3" />}
   372→                            OK
   373→                          </Button>
   374→                        </>
   375→                      ) : (
   376→                        <>
   377→                          <Button
   378→                            variant="ghost"
   379→                            size="sm"
   380→                            className="h-7 text-xs gap-1"
   381→                            onClick={() => startEdit(cat)}
   382→                          >
   383→                            <Pencil className="size-3" />
   384→                          </Button>
   385→                          <Button
   386→                            variant="ghost"
   387→                            size="sm"
   388→                            className="h-7 text-xs gap-1 text-destructive hover:text-destructive"
   389→                            onClick={() => setDeleteTarget(cat)}
   390→                          >
   391→                            <Trash2 className="size-3" />
   392→                          </Button>
   393→                        </>
   394→                      )}
   395→                    </div>
   396→                  </TableCell>
   397→                </TableRow>
   398→              ))}
   399→            </TableBody>
   400→          </Table>
   401→        </div>
   402→      )}
   403→
   404→      {/* Delete confirmation */}
   405→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   406→        <AlertDialogContent>
   407→          <AlertDialogHeader>
   408→            <AlertDialogTitle>Supprimer la catégorie</AlertDialogTitle>
   409→            <AlertDialogDescription>
   410→              Êtes-vous sûr de vouloir supprimer « {deleteTarget?.name} » ? Les puzzles associés ne seront pas supprimés mais seront décatégorisés.
   411→            </AlertDialogDescription>
   412→          </AlertDialogHeader>
   413→          <AlertDialogFooter>
   414→            <AlertDialogCancel disabled={deleting}>Annuler</AlertDialogCancel>
   415→            <AlertDialogAction
   416→              onClick={handleDelete}
   417→              disabled={deleting}
   418→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   419→            >
   420→              {deleting ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   421→              Supprimer
   422→            </AlertDialogAction>
   423→          </AlertDialogFooter>
   424→        </AlertDialogContent>
   425→      </AlertDialog>
   426→    </>
   427→  );
   428→}

   430→// --- Main AdminPanel Component ---

   432→export default function AdminPanel({ open, onOpenChange }: AdminPanelProps) {
   433→  // Data state
   434→  const [data, setData] = useState<AdminData | null>(null);
   435→  const [loading, setLoading] = useState(false);
   436→
   437→  // Schedule editing state
   438→  const [schedulePuzzlesPerDay, setSchedulePuzzlesPerDay] = useState(1);
   439→  const [scheduleIsActive, setScheduleIsActive] = useState(false);
   440→  const [savingSchedule, setSavingSchedule] = useState(false);
   441→
   442→  // Date picker state
   443→  const [datePickerPuzzleId, setDatePickerPuzzleId] = useState<string | null>(null);
   444→  const [selectedDate, setSelectedDate] = useState<Date | undefined>(undefined);
   445→  const [savingDate, setSavingDate] = useState(false);
   446→
   447→  // Publish/unpublish loading map
   448→  const [togglingPuzzleIds, setTogglingPuzzleIds] = useState<Set<string>>(new Set());
   449→
   450→  // Delete confirmation state
   451→  const [deleteTarget, setDeleteTarget] = useState<PuzzleSummary | null>(null);
   452→  const [deleting, setDeleting] = useState(false);
   453→
   454→  // Editor state
   455→  const [editorOpen, setEditorOpen] = useState(false);
   456→  const [editPuzzleId, setEditPuzzleId] = useState<string | null>(null);
   457→
   458→  // Category filter for puzzle table
   459→  const [categoryFilter, setCategoryFilter] = useState<string>('all');
   460→  const [languageFilter, setLanguageFilter] = useState<string>('all');
   461→
   462→  // Expose fetchData for editor callback
   463→  const fetchData = useCallback(async () => {
   464→    setLoading(true);
   465→    try {
   466→      const res = await fetch('/api/puzzles/admin');
   467→      if (!res.ok) {
   468→        throw new Error('Erreur lors du chargement des données');
   469→      }
   470→      const json: AdminData = await res.json();
   471→      setData(json);
   472→      setSchedulePuzzlesPerDay(json.schedule.puzzlesPerDay);
   473→      setScheduleIsActive(json.schedule.isActive);
   474→    } catch (err) {
   475→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   476→    } finally {
   477→      setLoading(false);
   478→    }
   479→  }, []);
   480→
   481→  useEffect(() => {
   482→    if (open) {
   483→      fetchData();
   484→    }
   485→  }, [open, fetchData]);
   486→
   487→  // Reset state when dialog opens/closes
   488→  useEffect(() => {
   489→    if (!open) {
   490→      setDatePickerPuzzleId(null);
   491→      setSelectedDate(undefined);
   492→    }
   493→  }, [open]);
   494→
   495→  // --- Handlers ---
   496→
   497→  const handleSaveSchedule = async () => {
   498→    setSavingSchedule(true);
   499→    try {
   500→      const res = await fetch('/api/puzzles/admin/schedule', {
   501→        method: 'POST',
   502→        headers: { 'Content-Type': 'application/json' },
   503→        body: JSON.stringify({
   504→          puzzlesPerDay: schedulePuzzlesPerDay,
   505→          isActive: scheduleIsActive,
   506→        }),
   507→      });
   508→      if (!res.ok) throw new Error('Erreur lors de la sauvegarde');
   509→      toast.success('Programme de publication mis à jour');
   510→      await fetchData();
   511→    } catch (err) {
   512→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   513→    } finally {
   514→      setSavingSchedule(false);
   515→    }
   516→  };
   517→
   518→  const handleTogglePublish = async (puzzle: PuzzleSummary) => {
   519→    const puzzleId = puzzle.id;
   520→    const newPublished = !puzzle.published;
   521→
   522→    setTogglingPuzzleIds((prev) => new Set(prev).add(puzzleId));
   523→    setData((prev) =>
   524→      prev
   525→        ? {
   526→            ...prev,
   527→            puzzles: prev.puzzles.map((p) =>
   528→              p.id === puzzleId
   529→                ? {
   530→                    ...p,
   531→                    published: newPublished,
   532→                    publishDate: newPublished
   533→                      ? p.publishDate || new Date().toISOString()
   534→                      : null,
   535→                  }
   536→                : p
   537→            ),
   538→          }
   539→        : prev
   540→    );
   541→
   542→    try {
   543→      const res = await fetch('/api/puzzles/admin/publish', {
   544→        method: 'POST',
   545→        headers: { 'Content-Type': 'application/json' },
   546→        body: JSON.stringify({ puzzleId, published: newPublished }),
   547→      });
   548→      if (!res.ok) throw new Error('Erreur lors de la modification');
   549→      toast.success(newPublished ? 'Puzzle publié avec succès' : 'Puzzle dépublié avec succès');
   550→      await fetchData();
   551→    } catch (err) {
   552→      setData((prev) =>
   553→        prev
   554→          ? {
   555→              ...prev,
   556→              puzzles: prev.puzzles.map((p) =>
   557→                p.id === puzzleId
   558→                  ? { ...p, published: !newPublished, publishDate: puzzle.publishDate }
   559→                  : p
   560→              ),
   561→            }
   562→          : prev
   563→      );
   564→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   565→    } finally {
   566→      setTogglingPuzzleIds((prev) => {
   567→        const next = new Set(prev);
   568→        next.delete(puzzleId);
   569→        return next;
   570→      });
   571→    }
   572→  };
   573→
   574→  const handleOpenDatePicker = (puzzle: PuzzleSummary) => {
   575→    setDatePickerPuzzleId(puzzle.id);
   576→    setSelectedDate(puzzle.publishDate ? new Date(puzzle.publishDate) : undefined);
   577→  };
   578→
   579→  const handleSaveDate = async () => {
   580→    if (!datePickerPuzzleId || !selectedDate) return;
   581→    setSavingDate(true);
   582→    try {
   583→      const res = await fetch('/api/puzzles/admin/publish', {
   584→        method: 'POST',
   585→        headers: { 'Content-Type': 'application/json' },
   586→        body: JSON.stringify({
   587→          puzzleId: datePickerPuzzleId,
   588→          published: true,
   589→          publishDate: selectedDate.toISOString(),
   590→        }),
   591→      });
   592→      if (!res.ok) throw new Error('Erreur lors de la mise à jour de la date');
   593→      toast.success('Date de publication mise à jour');
   594→      setDatePickerPuzzleId(null);
   595→      setSelectedDate(undefined);
   596→      await fetchData();
   597→    } catch (err) {
   598→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   599→    } finally {
   600→      setSavingDate(false);
   601→    }
   602→  };
   603→
   604→  const handleDelete = async () => {
   605→    if (!deleteTarget) return;
   606→    setDeleting(true);
   607→    try {
   608→      const res = await fetch(`/api/puzzles/admin?id=${deleteTarget.id}`, { method: 'DELETE' });
   609→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   610→      toast.success(`« ${deleteTarget.title} » supprimé`);
   611→      setDeleteTarget(null);
   612→      await fetchData();
   613→    } catch (err) {
   614→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   615→    } finally {
   616→      setDeleting(false);
   617→    }
   618→  };
   619→
   620→  const handlePuzzlesPerDayChange = (value: string) => {
   621→    const num = parseInt(value, 10);
   622→    if (!isNaN(num) && num >= 1 && num <= 10) {
   623→      setSchedulePuzzlesPerDay(num);
   624→    }
   625→  };
   626→
   627→  const openEditor = (puzzleId?: string) => {
   628→    setEditPuzzleId(puzzleId ?? null);
   629→    setEditorOpen(true);
   630→  };
   631→
   632→  const handleEditorSaved = () => {
   633→    fetchData();
   634→  };
   635→
   636→  // Filtered puzzles
   637→  const filteredPuzzles = useMemo(() => {
   638→    if (!data) return [];
   639→    let filtered = data.puzzles;
   640→    if (categoryFilter !== 'all') {
   641→      filtered = filtered.filter((p) => p.categoryId === categoryFilter);
   642→    }
   643→    if (languageFilter !== 'all') {
   644→      filtered = filtered.filter((p) => p.language === languageFilter);
   645→    }
   646→    return filtered;
   647→  }, [data, categoryFilter, languageFilter]);
   648→
   649→  // --- Render ---
   650→
   651→  return (
   652→    <>
   653→      <Dialog open={open} onOpenChange={onOpenChange}>
   654→        <DialogContent className="sm:max-w-5xl max-h-[90vh] overflow-y-auto p-0 gap-0">
   655→          <DialogHeader className="p-6 pb-0">
   656→            <DialogTitle className="text-xl font-bold flex items-center gap-2">
   657→              <Settings className="size-5" />
   658→              Administration
   659→            </DialogTitle>
   660→            <DialogDescription>
   661→              Gérez vos catégories, puzzles et programme de publication.
   662→            </DialogDescription>
   663→          </DialogHeader>
   664→
   665→          <div className="p-6 space-y-6">
   666→            <Tabs defaultValue="categories" className="w-full">
   667→              <TabsList className="grid w-full grid-cols-3">
   668→                <TabsTrigger value="categories" className="text-xs sm:text-sm gap-1.5">
   669→                  <FolderOpen className="size-3.5" />
   670→                  Catégories
   671→                </TabsTrigger>
   672→                <TabsTrigger value="puzzles" className="text-xs sm:text-sm gap-1.5">
   673→                  <Grid3X3 className="size-3.5" />
   674→                  Puzzles
   675→                </TabsTrigger>
   676→                <TabsTrigger value="schedule" className="text-xs sm:text-sm gap-1.5">
   677→                  <CalendarDays className="size-3.5" />
   678→                  Programme
   679→                </TabsTrigger>
   680→              </TabsList>
   681→
   682→              {/* ── Categories Tab ── */}
   683→              <TabsContent value="categories" className="mt-4">
   684→                <section className="rounded-lg border bg-muted/30 p-4">
   685→                  <h3 className="text-sm font-semibold mb-4 flex items-center gap-2">
   686→                    <Tag className="size-4 text-muted-foreground" />
   687→                    Gestion des Catégories
   688→                    <Badge variant="secondary" className="ml-1">
   689→                      {data?.categories.length ?? 0}
   690→                    </Badge>
   691→                  </h3>
   692→
   693→                  {loading && !data ? (
   694→                    <CategorySkeleton />
   695→                  ) : data ? (
   696→                    <CategoryManager
   697→                      categories={data.categories}
   698→                      onRefresh={fetchData}
   699→                    />
   700→                  ) : null}
   701→                </section>
   702→              </TabsContent>
   703→
   704→              {/* ── Puzzles Tab ── */}
   705→              <TabsContent value="puzzles" className="mt-4">
   706→                <section>
   707→                  {/* Filters and create button */}
   708→                  <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 mb-4">
   709→                    <h3 className="text-sm font-semibold flex items-center gap-2">
   710→                      <CalendarIcon className="size-4 text-muted-foreground" />
   711→                      Tous les Puzzles
   712→                      <Badge variant="secondary" className="ml-1">
   713→                        {filteredPuzzles.length}
   714→                      </Badge>
   715→                    </h3>
   716→
   717→                    <div className="flex items-center gap-2">
   718→                      <Select value={languageFilter} onValueChange={setLanguageFilter}>
   719→                        <SelectTrigger className="h-8 text-xs w-28">
   720→                          <Languages className="size-3 mr-1" />
   721→                          <SelectValue placeholder="Langue" />
   722→                        </SelectTrigger>
   723→                        <SelectContent>
   724→                          <SelectItem value="all">Toutes</SelectItem>
   725→                          <SelectItem value="fr">🇫🇷 Français</SelectItem>
   726→                          <SelectItem value="en">🇬🇧 English</SelectItem>
   727→                        </SelectContent>
   728→                      </Select>
   729→
   730→                      <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   731→                        <SelectTrigger className="h-8 text-xs w-36">
   732→                          <FolderOpen className="size-3 mr-1" />
   733→                          <SelectValue placeholder="Catégorie" />
   734→                        </SelectTrigger>
   735→                        <SelectContent>
   736→                          <SelectItem value="all">Toutes</SelectItem>
   737→                          {data?.categories.map((cat) => (
   738→                            <SelectItem key={cat.id} value={cat.id}>
   739→                              {languageLabel(cat.language)} {cat.name}
   740→                            </SelectItem>
   741→                          ))}
   742→                        </SelectContent>
   743→                      </Select>
   744→
   745→                      <Button size="sm" className="h-8 text-xs gap-1.5" onClick={() => openEditor()}>
   746→                        <Plus className="size-3.5" />
   747→                        Créer
   748→                      </Button>
   749→                    </div>
   750→                  </div>
   751→
   752→                  {loading && !data ? (
   753→                    <TableSkeleton />
   754→                  ) : filteredPuzzles.length === 0 ? (
   755→                    <div className="text-center py-8 text-muted-foreground text-sm">
   756→                      Aucun puzzle ne correspond aux filtres.
   757→                    </div>
   758→                  ) : (
   759→                    <div className="rounded-md border max-h-96 overflow-y-auto">
   760→                      <Table>
   761→                        <TableHeader>
   762→                          <TableRow className="hover:bg-transparent">
   763→                            <TableHead className="pl-3">Titre</TableHead>
   764→                            <TableHead>Catégorie</TableHead>
   765→                            <TableHead>Langue</TableHead>
   766→                            <TableHead>Difficulté</TableHead>
   767→                            <TableHead className="hidden md:table-cell">Taille</TableHead>
   768→                            <TableHead className="hidden lg:table-cell">Date</TableHead>
   769→                            <TableHead>Statut</TableHead>
   770→                            <TableHead className="text-right pr-3">Actions</TableHead>
   771→                          </TableRow>
   772→                        </TableHeader>
   773→                        <TableBody>
   774→                          {filteredPuzzles.map((puzzle) => {
   775→                            const isToggling = togglingPuzzleIds.has(puzzle.id);
   776→                            const isDatePicking = datePickerPuzzleId === puzzle.id;
   777→
   778→                            return (
   779→                              <TableRow key={puzzle.id}>
   780→                                <TableCell className="font-medium pl-3 max-w-[140px] truncate">
   781→                                  {puzzle.title}
   782→                                </TableCell>
   783→
   784→                                <TableCell>
   785→                                  {puzzle.categoryName ? (
   786→                                    <Badge variant="outline" className="text-xs">
   787→                                      {puzzle.categoryName}
   788→                                    </Badge>
   789→                                  ) : (
   790→                                    <span className="text-xs text-muted-foreground">—</span>
   791→                                  )}
   792→                                </TableCell>
   793→
   794→                                <TableCell>
   795→                                  <Badge variant="outline" className="text-xs">
   796→                                    {languageLabel(puzzle.language)}
   797→                                  </Badge>
   798→                                </TableCell>
   799→
   800→                                <TableCell>{renderStars(puzzle.difficulty)}</TableCell>
   801→
   802→                                <TableCell className="hidden md:table-cell text-muted-foreground text-xs">
   803→                                  {puzzle.rows}×{puzzle.cols}
   804→                                </TableCell>
   805→
   806→                                <TableCell className="hidden lg:table-cell text-sm text-muted-foreground">
   807→                                  {isDatePicking ? (
   808→                                    <Popover open={isDatePicking} onOpenChange={(o) => !o && setDatePickerPuzzleId(null)}>
   809→                                      <PopoverTrigger asChild>
   810→                                        <Button variant="outline" size="sm" className="h-7 text-xs gap-1.5 font-normal">
   811→                                          <CalendarIcon className="size-3" />
   812→                                          {selectedDate ? format(selectedDate, 'dd MMM yyyy', { locale: fr }) : 'Choisir…'}
   813→                                        </Button>
   814→                                      </PopoverTrigger>
   815→                                      <PopoverContent className="w-auto p-0" align="start">
   816→                                        <Calendar mode="single" selected={selectedDate} onSelect={setSelectedDate} initialFocus />
   817→                                        <div className="flex justify-end gap-2 border-t p-2">
   818→                                          <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={() => setDatePickerPuzzleId(null)}>
   819→                                            Annuler
   820→                                          </Button>
   821→                                          <Button size="sm" className="h-7 text-xs" onClick={handleSaveDate} disabled={!selectedDate || savingDate}>
   822→                                            {savingDate ? <Loader2 className="size-3 animate-spin" /> : <Save className="size-3" />}
   823→                                            OK
   824→                                          </Button>
   825→                                        </div>
   826→                                      </PopoverContent>
   827→                                    </Popover>
   828→                                  ) : (
   829→                                    <span>{formatDate(puzzle.publishDate)}</span>
   830→                                  )}
   831→                                </TableCell>
   832→
   833→                                <TableCell>
   834→                                  {puzzle.published ? (
   835→                                    <Badge variant="default" className="bg-emerald-600 hover:bg-emerald-700 text-white border-transparent gap-1">
   836→                                      <CheckCircle2 className="size-3" />
   837→                                      <span className="hidden sm:inline">Publié</span>
   838→                                    </Badge>
   839→                                  ) : (
   840→                                    <Badge variant="secondary" className="gap-1 text-muted-foreground">
   841→                                      <CircleOff className="size-3" />
   842→                                      <span className="hidden sm:inline">Brouillon</span>
   843→                                    </Badge>
   844→                                  )}
   845→                                </TableCell>
   846→
   847→                                <TableCell className="text-right pr-3">
   848→                                  <div className="flex items-center justify-end gap-1">
   849→                                    <Button
   850→                                      variant="ghost"
   851→                                      size="sm"
   852→                                      className="h-7 text-xs gap-1.5"
   853→                                      onClick={() => openEditor(puzzle.id)}
   854→                                      title="Modifier"
   855→                                    >
   856→                                      <Pencil className="size-3" />
   857→                                      <span className="hidden xl:inline">Modifier</span>
   858→                                    </Button>
   859→                                    <Button
   860→                                      variant={puzzle.published ? 'outline' : 'default'}
   861→                                      size="sm"
   862→                                      className="h-7 text-xs gap-1.5"
   863→                                      onClick={() => handleTogglePublish(puzzle)}
   864→                                      disabled={isToggling}
   865→                                    >
   866→                                      {isToggling ? (
   867→                                        <Loader2 className="size-3 animate-spin" />
   868→                                      ) : puzzle.published ? (
   869→                                        <EyeOff className="size-3" />
   870→                                      ) : (
   871→                                        <Eye className="size-3" />
   872→                                      )}
   873→                                      <span className="hidden xl:inline">{puzzle.published ? 'Dépublier' : 'Publier'}</span>
   874→                                    </Button>
   875→                                    <Button
   876→                                      variant="ghost"
   877→                                      size="sm"
   878→                                      className="h-7 text-xs gap-1.5"
   879→                                      onClick={() => handleOpenDatePicker(puzzle)}
   880→                                      disabled={isToggling || isDatePicking}
   881→                                      title="Date"
   882→                                    >
   883→                                      <CalendarDays className="size-3" />
   884→                                    </Button>
   885→                                    <Button
   886→                                      variant="ghost"
   887→                                      size="sm"
   888→                                      className="h-7 text-xs gap-1.5 text-destructive hover:text-destructive"
   889→                                      onClick={() => setDeleteTarget(puzzle)}
   890→                                    >
   891→                                      <Trash2 className="size-3" />
   892→                                    </Button>
   893→                                  </div>
   894→                                </TableCell>
   895→                              </TableRow>
   896→                            );
   897→                          })}
   898→                        </TableBody>
   899→                      </Table>
   900→                    </div>
   901→                  )}
   902→                </section>
   903→              </TabsContent>
   904→
   905→              {/* ── Schedule Tab ── */}
   906→              <TabsContent value="schedule" className="mt-4">
   907→                <section className="rounded-lg border bg-muted/30 p-4">
   908→                  <h3 className="text-sm font-semibold mb-4 flex items-center gap-2">
   909→                    <CalendarDays className="size-4 text-muted-foreground" />
   910→                    Programme de Publication
   911→                  </h3>
   912→
   913→                  <div className="space-y-4">
   914→                    <div className="flex items-center justify-between">
   915→                      <Label htmlFor="schedule-active" className="text-sm font-medium">
   916→                        Publication automatique active
   917→                      </Label>
   918→                      <Switch
   919→                        id="schedule-active"
   920→                        checked={scheduleIsActive}
   921→                        onCheckedChange={setScheduleIsActive}
   922→                      />
   923→                    </div>
   924→
   925→                    <div className="flex items-center gap-3">
   926→                      <Label htmlFor="puzzles-per-day" className="text-sm font-medium whitespace-nowrap">
   927→                        Puzzles par jour
   928→                      </Label>
   929→                      <Input
   930→                        id="puzzles-per-day"
   931→                        type="number"
   932→                        min={1}
   933→                        max={10}
   934→                        value={schedulePuzzlesPerDay}
   935→                        onChange={(e) => handlePuzzlesPerDayChange(e.target.value)}
   936→                        className="w-20 h-9 text-center"
   937→                        disabled={!scheduleIsActive}
   938→                      />
   939→                    </div>
   940→
   941→                    <div className="flex justify-end pt-1">
   942→                      <Button size="sm" onClick={handleSaveSchedule} disabled={savingSchedule}>
   943→                        {savingSchedule ? <Loader2 className="size-4 animate-spin" /> : <Save className="size-4" />}
   944→                        Enregistrer
   945→                      </Button>
   946→                    </div>
   947→                  </div>
   948→                </section>
   949→              </TabsContent>
   950→            </Tabs>
   951→          </div>
   952→        </DialogContent>
   953→      </Dialog>
   954→
   955→      {/* ── Puzzle Editor Dialog ──────────────────────────────── */}
   956→      <PuzzleEditor
   957→        open={editorOpen}
   958→        onOpenChange={setEditorOpen}
   959→        editPuzzleId={editPuzzleId}
   960→        onSaved={handleEditorSaved}
   961→      />
   962→
   963→      {/* ── Delete Confirmation Dialog ────────────────────────── */}
   964→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   965→        <AlertDialogContent>
   966→          <AlertDialogHeader>
   967→            <AlertDialogTitle>Supprimer le puzzle</AlertDialogTitle>
   968→            <AlertDialogDescription>
   969→              Êtes-vous sûr de vouloir supprimer « {deleteTarget?.title} » ? Cette action est irréversible.
   970→            </AlertDialogDescription>
   971→          </AlertDialogHeader>
   972→          <AlertDialogFooter>
   973→            <AlertDialogCancel disabled={deleting}>Annuler</AlertDialogCancel>
   974→            <AlertDialogAction
   975→              onClick={handleDelete}
   976→              disabled={deleting}
   977→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   978→            >
   979→              {deleting ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   980→              Supprimer
   981→            </AlertDialogAction>
   982→          </AlertDialogFooter>
   983→        </AlertDialogContent>
   984→      </AlertDialog>
   985→    </>
   986→  );
   987→}
