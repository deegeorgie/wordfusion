   164→function formatDate(dateStr: string | null): string {
   165→  if (!dateStr) return '—';
   166→  try {
   167→    return format(new Date(dateStr), 'dd MMM yyyy', { locale: fr });
   168→  } catch {
   169→    return '—';
   170→  }
   171→}

   173→function languageLabel(lang: string) {
   174→  return lang === 'en' ? '🇬🇧 EN' : '🇫🇷 FR';

   217→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {
   218→  const [newName, setNewName] = useState('');
   219→  const [newIcon, setNewIcon] = useState('🏷️');
   220→  const [newLanguage, setNewLanguage] = useState('fr');
   221→  const [creating, setCreating] = useState(false);
   222→
   223→  // Edit state
   224→  const [editingId, setEditingId] = useState<string | null>(null);
   225→  const [editName, setEditName] = useState('');
   226→  const [editIcon, setEditIcon] = useState('');
   227→  const [editLanguage, setEditLanguage] = useState('');
   228→  const [saving, setSaving] = useState(false);
   229→
   230→  // Delete state
   231→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   232→  const [deleting, setDeleting] = useState(false);
   233→
   234→  const handleCreate = async () => {
   235→    if (!newName.trim()) return;
   236→    setCreating(true);
   237→    try {
   238→      const res = await fetch('/api/categories', {
   239→        method: 'POST',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify({ name: newName.trim(), icon: newIcon, language: newLanguage }),
   242→      });
   243→      if (!res.ok) {
   244→        const err = await res.json().catch(() => ({}));
   245→        throw new Error((err as { error?: string }).error || 'Erreur');
   246→      }
   247→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   248→      setNewName('');
   249→      setNewIcon('🏷️');
   250→      onRefresh();
   251→    } catch (err) {
   252→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   253→    } finally {
   254→      setCreating(false);
   255→    }
   256→  };
   257→
   258→  const startEdit = (cat: CategoryItem) => {
   259→    setEditingId(cat.id);
   260→    setEditName(cat.name);
   261→    setEditIcon(cat.icon || '🏷️');
   262→    setEditLanguage(cat.language);
   263→  };
   264→
   265→  const handleSaveEdit = async () => {
   266→    if (!editingId || !editName.trim()) return;
   267→    setSaving(true);
   268→    try {
   269→      const res = await fetch('/api/categories', {
   270→        method: 'PUT',
   271→        headers: { 'Content-Type': 'application/json' },
   272→        body: JSON.stringify({ id: editingId, name: editName.trim(), icon: editIcon, language: editLanguage }),
   273→      });
   274→      if (!res.ok) {
   275→        const err = await res.json().catch(() => ({}));
   276→        throw new Error((err as { error?: string }).error || 'Erreur');
   277→      }
   278→      toast.success('Catégorie mise à jour');
   279→      setEditingId(null);
   280→      onRefresh();
   281→    } catch (err) {
   282→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   283→    } finally {
   284→      setSaving(false);
   285→    }
   286→  };
   287→
   288→  const handleDelete = async () => {
   289→    if (!deleteTarget) return;
   290→    setDeleting(true);
   291→    try {
   292→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   293→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   294→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   295→      setDeleteTarget(null);
   296→      onRefresh();
   297→    } catch (err) {
   298→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   299→    } finally {
   300→      setDeleting(false);
   301→    }
   302→  };
   303→
   304→  return (
   305→    <>
   306→      {/* Create new category row */}
   307→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   308→        <div className="w-16 sm:w-20 space-y-1">
   309→          <Label className="text-xs text-muted-foreground">Icône</Label>
   310→          <Input
   311→            placeholder="🏷️"
   312→            value={newIcon}
   313→            onChange={(e) => setNewIcon(e.target.value)}
   314→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   315→            className="h-8 text-sm text-center"
   316→            maxLength={4}
   317→          />
   318→        </div>
   319→        <div className="flex-1 w-full sm:w-auto space-y-1">
   320→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   321→          <Input
   322→            placeholder="Ex: Animaux, Sciences..."
   323→            value={newName}
   324→            onChange={(e) => setNewName(e.target.value)}
   325→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   326→            className="h-8 text-sm"
   327→          />
   328→        </div>
   329→        <div className="w-full sm:w-28 space-y-1">
   330→          <Label className="text-xs text-muted-foreground">Langue</Label>
   331→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   332→            <SelectTrigger className="h-8 text-sm">
   333→              <SelectValue />
   334→            </SelectTrigger>
   335→            <SelectContent>
   336→              <SelectItem value="fr">🇫🇷 Français</SelectItem>
   337→              <SelectItem value="en">🇬🇧 English</SelectItem>
   338→            </SelectContent>
   339→          </Select>
   340→        </div>
   341→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   342→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   343→          Ajouter
   344→        </Button>
   345→      </div>
   346→
   347→      {/* Category list */}
   348→      {categories.length === 0 ? (
   349→        <div className="text-center py-6 text-muted-foreground text-sm">
   350→          Aucune catégorie. Créez-en une ci-dessus.
   351→        </div>
   352→      ) : (
   353→        <div className="rounded-md border max-h-64 overflow-y-auto">
   354→          <Table>
   355→            <TableHeader>
   356→              <TableRow className="hover:bg-transparent">
   357→                <TableHead className="pl-3">Nom</TableHead>
   358→                <TableHead>Langue</TableHead>
   359→                <TableHead>Puzzles</TableHead>
   360→                <TableHead className="text-right pr-3">Actions</TableHead>
   361→              </TableRow>
   362→            </TableHeader>
   363→            <TableBody>
   364→              {categories.map((cat) => (
   365→                <TableRow key={cat.id}>
   366→                  <TableCell className="font-medium pl-3">
   367→                    {editingId === cat.id ? (
   368→                      <div className="flex items-center gap-1.5">
   369→                        <Input
   370→                          value={editIcon}
   371→                          onChange={(e) => setEditIcon(e.target.value)}
   372→                          className="h-7 text-sm w-12 text-center"
   373→                          maxLength={4}
   374→                          autoFocus
   375→                        />
   376→                        <Input
   377→                          value={editName}
   378→                          onChange={(e) => setEditName(e.target.value)}
   379→                          onKeyDown={(e) => e.key === 'Enter' && handleSaveEdit()}
   380→                          className="h-7 text-sm w-full"
   381→                        />
   382→                      </div>
   383→                    ) : (
   384→                      <span className="flex items-center gap-1.5">
   385→                        <span className="text-sm">{cat.icon || '🏷️'}</span>
   386→                        {cat.name}
   387→                      </span>
   388→                    )}
   389→                  </TableCell>
   390→                  <TableCell>
   391→                    {editingId === cat.id ? (
   392→                      <Select value={editLanguage} onValueChange={setEditLanguage}>
   393→                        <SelectTrigger className="h-7 text-xs w-24">
   394→                          <SelectValue />
   395→                        </SelectTrigger>
   396→                        <SelectContent>
   397→                          <SelectItem value="fr">🇫🇷 FR</SelectItem>
   398→                          <SelectItem value="en">🇬🇧 EN</SelectItem>
   399→                        </SelectContent>
   400→                      </Select>
   401→                    ) : (
   402→                      <Badge variant="outline" className="text-xs">
   403→                        {languageLabel(cat.language)}
   404→                      </Badge>
   405→                    )}
   406→                  </TableCell>
   407→                  <TableCell className="text-muted-foreground text-xs">
   408→                    {cat._count.puzzles}
   409→                  </TableCell>
   410→                  <TableCell className="text-right pr-3">
   411→                    <div className="flex items-center justify-end gap-1">
   412→                      {editingId === cat.id ? (
   413→                        <>
   414→                          <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={() => setEditingId(null)}>
   415→                            Annuler
   416→                          </Button>
   417→                          <Button size="sm" className="h-7 text-xs gap-1" onClick={handleSaveEdit} disabled={saving}>
   418→                            {saving ? <Loader2 className="size-3 animate-spin" /> : <Save className="size-3" />}
   419→                            OK
   420→                          </Button>
   421→                        </>
   422→                      ) : (
   423→                        <>
   424→                          <Button
   425→                            variant="ghost"
   426→                            size="sm"
   427→                            className="h-7 text-xs gap-1"
   428→                            onClick={() => startEdit(cat)}
   429→                          >
   430→                            <Pencil className="size-3" />
   431→                          </Button>
   432→                          <Button
   433→                            variant="ghost"
   434→                            size="sm"
   435→                            className="h-7 text-xs gap-1 text-destructive hover:text-destructive"
   436→                            onClick={() => setDeleteTarget(cat)}
   437→                          >
   438→                            <Trash2 className="size-3" />
   439→                          </Button>
   440→                        </>
   441→                      )}
   442→                    </div>
   443→                  </TableCell>
   444→                </TableRow>
   445→              ))}
   446→            </TableBody>
   447→          </Table>
   448→        </div>
   449→      )}
   450→
   451→      {/* Delete confirmation */}
   452→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   453→        <AlertDialogContent>
   454→          <AlertDialogHeader>
   455→            <AlertDialogTitle>Supprimer la catégorie</AlertDialogTitle>
   456→            <AlertDialogDescription>
   457→              Êtes-vous sûr de vouloir supprimer « {deleteTarget?.name} » ? Les puzzles associés ne seront pas supprimés mais seront décatégorisés.
   458→            </AlertDialogDescription>
   459→          </AlertDialogHeader>
   460→          <AlertDialogFooter>
   461→            <AlertDialogCancel disabled={deleting}>Annuler</AlertDialogCancel>
   462→            <AlertDialogAction
   463→              onClick={handleDelete}
   464→              disabled={deleting}
   465→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   466→            >
   467→              {deleting ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   468→              Supprimer
   469→            </AlertDialogAction>
   470→          </AlertDialogFooter>
   471→        </AlertDialogContent>
   472→      </AlertDialog>
   473→    </>
   474→  );
   475→}
