    85→/** Create an empty grid of all-black cells */
    86→function createEmptyGrid(rows: number, cols: number): CrosswordCell[][] {
    87→  const grid: CrosswordCell[][] = [];
    88→  for (let r = 0; r < rows; r++) {
    89→    grid[r] = [];
    90→    for (let c = 0; c < cols; c++) {
    91→      grid[r][c] = { letter: "", number: null, isBlack: true, row: r, col: c };
    92→    }
    93→  }
    94→  return grid;
    95→}
    96→
    97→/** Deep-clone a grid */
    98→function cloneGrid(grid: CrosswordCell[][]): CrosswordCell[][] {
    99→  return grid.map((row) => row.map((cell) => ({ ...cell })));
   100→}

   102→/** Detect all words (2+ consecutive non-black cells with at least one letter) and assign numbers */
   103→function detectWordsAndNumbers(
   104→  grid: CrosswordCell[][]
   105→): { words: WordPlacement[]; clues: EditorClue[] } {
   106→  const rows = grid.length;
   107→  const cols = grid[0]?.length ?? 0;
   108→
   109→  // First pass: assign numbers
   110→  let clueNum = 1;
   111→  const numberMap = new Map<string, number>();
   112→
   113→  for (let r = 0; r < rows; r++) {
   114→    for (let c = 0; c < cols; c++) {
   115→      if (grid[r][c].isBlack) continue;
   116→      const key = `${r},${c}`;
   117→      if (numberMap.has(key)) continue;
   118→
   119→      const startsAcross =
   120→        (c === 0 || grid[r][c - 1].isBlack) &&
   121→        c + 1 < cols &&
   122→        !grid[r][c + 1].isBlack;
   123→      const startsDown =
   124→        (r === 0 || grid[r - 1][c].isBlack) &&
   125→        r + 1 < rows &&
   126→        !grid[r + 1][c].isBlack;
   127→
   128→      if (startsAcross || startsDown) {
   129→        numberMap.set(key, clueNum);
   130→        clueNum++;
   131→      }
   132→    }
   133→  }
   134→
   135→  // Build number grid for quick lookup
   136→  const numGrid: (number | null)[][] = [];
   137→  for (let r = 0; r < rows; r++) {
   138→    numGrid[r] = [];
   139→    for (let c = 0; c < cols; c++) {
   140→      numGrid[r][c] = grid[r][c].isBlack ? null : (numberMap.get(`${r},${c}`) ?? null);
   141→    }
   142→  }
   143→
   144→  const words: WordPlacement[] = [];
   145→  const clues: EditorClue[] = [];
   146→
   147→  // Detect across words
   148→  for (let r = 0; r < rows; r++) {
   149→    let c = 0;
   150→    while (c < cols) {
   151→      if (grid[r][c].isBlack) {
   152→        c++;
   153→        continue;
   154→      }
   155→      // Start of a horizontal run
   156→      const startC = c;
   157→      let wordStr = "";
   158→      while (c < cols && !grid[r][c].isBlack) {
   159→        wordStr += grid[r][c].letter;
   160→        c++;
   161→      }
   162→      if (wordStr.length >= 2) {
   163→        const num = numGrid[r][startC];
   164→        if (num !== null) {
   165→          words.push({
   166→            word: wordStr,
   167→            clueNumber: num,
   168→            direction: "across",
   169→            row: r,
   170→            col: startC,
   171→            length: wordStr.length,
   172→          });
   173→          clues.push({
   174→            number: num,
   175→            direction: "across",
   176→            text: "",
   177→            word: wordStr,
   178→            row: r,
   179→            col: startC,
   180→          });
   181→        }
   182→      }
   183→    }
   184→  }
   185→
   186→  // Detect down words
   187→  for (let c = 0; c < cols; c++) {
   188→    let r = 0;
   189→    while (r < rows) {
   190→      if (grid[r][c].isBlack) {
   191→        r++;
   192→        continue;
   193→      }
   194→      const startR = r;
   195→      let wordStr = "";
   196→      while (r < rows && !grid[r][c].isBlack) {
   197→        wordStr += grid[r][c].letter;
   198→        r++;
   199→      }
   200→      if (wordStr.length >= 2) {
   201→        const num = numGrid[startR][c];
   202→        if (num !== null) {
   203→          words.push({
   204→            word: wordStr,
   205→            clueNumber: num,
   206→            direction: "down",
   207→            row: startR,
   208→            col: c,
   209→            length: wordStr.length,
   210→          });
   211→          clues.push({
   212→            number: num,
   213→            direction: "down",
   214→            text: "",
   215→            word: wordStr,
   216→            row: startR,
   217→            col: c,
   218→          });
   219→        }
   220→      }
   221→    }
   222→  }
   223→
   224→  return { words, clues };
   225→}
   226→
   227→/** Update grid numbers from numberMap */
   228→function applyNumbersToGrid(
   229→  grid: CrosswordCell[][],
   230→  words: WordPlacement[]
   231→): CrosswordCell[][] {
   232→  const newGrid = cloneGrid(grid);
   233→  // Clear all numbers
   234→  for (let r = 0; r < newGrid.length; r++) {
   235→    for (let c = 0; c < newGrid[0].length; c++) {
   236→      newGrid[r][c].number = null;
   237→    }
   238→  }
   239→  // Set numbers from word placements
   240→  for (const w of words) {
   241→    if (newGrid[w.row]?.[w.col]) {
   242→      newGrid[w.row][w.col].number = w.clueNumber;
   243→    }
   244→  }
   245→  return newGrid;
   246→}

   470→      const maxCol = (grid[0]?.length ?? 1) - 1;

   484→          prev && prev.row > 0 ? { row: prev.row - 1, col: prev.col } : prev
   485→        );
   486→      } else if (e.key === "ArrowDown") {
   487→        setSelectedCell((prev) =>
   488→          prev && prev.row < maxRow ? { row: prev.row + 1, col: prev.col } : prev
   489→        );
   490→      } else if (e.key === "ArrowLeft") {
   491→        setSelectedCell((prev) =>
   492→          prev && prev.col > 0 ? { row: prev.row, col: prev.col - 1 } : prev
   493→        );
   494→      } else if (e.key === "ArrowRight") {
   495→        setSelectedCell((prev) =>
   496→          prev && prev.col < maxCol ? { row: prev.row, col: prev.col + 1 } : prev
   497→        );
   498→      } else if (e.key === "Backspace" || e.key === "Delete") {

   500→        if (cell && !cell.isBlack && cell.letter !== "") {
   501→          const newGrid = cloneGrid(grid);
   502→          newGrid[row][col] = { ...newGrid[row][col], letter: "" };

   504→        } else if (e.key === "Backspace") {

   506→          if (col > 0) {
   507→            setSelectedCell({ row, col: col - 1 });
   508→          } else if (row > 0) {
   509→            setSelectedCell({ row: row - 1, col: maxCol });
   510→          }
   511→        }
   512→      } else if (/^[a-zA-Z]$/.test(e.key)) {
   513→        const cell = grid[row]?.[col];
   514→        if (cell && !cell.isBlack) {
   515→          const newGrid = cloneGrid(grid);
   516→          newGrid[row][col] = {
   517→            ...newGrid[row][col],
   518→            letter: e.key.toUpperCase(),
   519→          };
   520→          recomputeWords(newGrid);
   521→          // Auto-advance
   522→          if (col < maxCol && !newGrid[row][col + 1].isBlack) {
   523→            setSelectedCell({ row, col: col + 1 });
   524→          } else if (row < maxRow) {
   525→            // Try to move to next row
   526→            setSelectedCell({ row: row + 1, col: 0 });
   527→          }
   528→        }
   529→      } else if (e.key === "Escape") {
   530→        setSelectedCell(null);
   531→      }
   532→    };
   533→
   534→    window.addEventListener("keydown", handleKeyDown);
   535→    return () => window.removeEventListener("keydown", handleKeyDown);
   536→  }, [selectedCell, grid, recomputeWords]);

   552→    setSelectedCell(null);

   554→  }, [grid, recomputeWords]);

   586→  // ── Update clue text ─────────────────────────────────────────────

   588→    (number: number, direction: "across" | "down", text: string) => {
   589→      setClues((prev) =>
   590→        prev.map((c) =>
   591→          c.number === number && c.direction === direction ? { ...c, text } : c
   592→        )
   593→      );
   594→    },
   595→    []
   596→  );
   597→
   598→  // ── Update word text → update grid cells ─────────────────────────
   599→  const handleWordChange = useCallback(
   600→    (
   601→      number: number,
   602→      direction: "across" | "down",
   603→      newWord: string,
   604→      row: number,
   605→      col: number
   606→    ) => {
   607→      const newGrid = cloneGrid(grid);
   608→      const oldWord = words.find(
   609→        (w) => w.clueNumber === number && w.direction === direction
   610→      );
   611→      const oldLen = oldWord?.length ?? newWord.length;
   612→
   613→      // Update grid cells
   614→      for (let i = 0; i < Math.max(oldLen, newWord.length); i++) {
   615→        const r = direction === "across" ? row : row + i;
   616→        const c = direction === "across" ? col + i : col;
   617→        if (r < newGrid.length && c < (newGrid[0]?.length ?? 0)) {
   618→          if (i < newWord.length) {
   619→            newGrid[r][c] = {
   620→              ...newGrid[r][c],
   621→              letter: newWord[i].toUpperCase(),
   622→              isBlack: false,
   623→            };
   624→          } else if (i >= newWord.length && i < oldLen) {
   625→            // Clear extra letters but keep cell white
   626→            newGrid[r][c] = { ...newGrid[r][c], letter: "" };
   627→          }
   628→        }
   629→      }
   630→
   631→      setGrid(newGrid);
   632→      recomputeWords(newGrid);
   633→    },
   634→    [grid, words, recomputeWords]
   635→  );
   636→
   637→  // ── Save ─────────────────────────────────────────────────────────
   638→  const handleSave = useCallback(async () => {
   639→    if (!title.trim()) {
   640→      toast.error("Le titre est requis");
   641→      return;
   642→    }
   643→
   644→    const finalClues: Clue[] = clues.map(({ number, direction, text }) => ({
   645→      number,
   646→      direction,
   647→      text,
   648→    }));
   649→
   650→    setSaving(true);
   651→    try {
   652→      const body = {
   653→        title: title.trim(),
   654→        description: description.trim() || undefined,
   655→        difficulty: Number(difficulty),
   656→        language,
   657→        categoryId: categoryId === noneCategory ? null : categoryId,
   658→        packId: packId === nonePack ? null : packId,
   659→        rows: grid.length,
   660→        cols: grid[0]?.length ?? 0,
   661→        grid,
   662→        words,
   663→        clues: finalClues,
   664→      };
   665→
   666→      const url = isEditing ? "/api/puzzles/admin" : "/api/puzzles/admin";
   667→      const method = isEditing ? "PUT" : "POST";
   668→      const payload = isEditing ? { ...body, id: editPuzzleId } : body;
   669→
   670→      const res = await fetch(url, {
   671→        method,
   672→        headers: { "Content-Type": "application/json" },
   673→        body: JSON.stringify(payload),
   674→      });
   675→
   676→      if (!res.ok) {
   677→        const data = await res.json().catch(() => ({}));
   678→        throw new Error(data.error ?? "Erreur lors de la sauvegarde");
   679→      }
   680→
   681→      toast.success(
   682→        isEditing ? "Puzzle mis à jour avec succès" : "Puzzle créé avec succès"
   683→      );
   684→
   685→      onSaved?.();
   686→
   687→      if (isEditing) {
   688→        onOpenChange(false);
   689→      } else {
   690→        // Reset for new puzzle
   691→        setTitle("");
   692→        setDescription("");
   693→        setDifficulty("2");
   694→        setLanguage("fr");
   695→        setCategoryId(noneCategory);
   696→        setPackId(nonePack);
   697→        setRowsInput(10);
   698→        setColsInput(10);
   699→        setGrid(createEmptyGrid(10, 10));
   700→        setSelectedCell(null);
   701→        setWords([]);
   702→        setClues([]);
   703→      }
   704→    } catch (err) {
   705→      toast.error(err instanceof Error ? err.message : "Erreur inconnue");
   706→    } finally {
   707→      setSaving(false);
   708→    }
   709→  }, [title, description, difficulty, language, categoryId, packId, grid, words, clues, isEditing, editPuzzleId, onSaved, onOpenChange]);
   710→
   711→  // ── Render helpers ───────────────────────────────────────────────
   712→  const cellSize = "w-9 h-9 sm:w-10 sm:h-10 text-base sm:text-lg";
   713→
   714→  const renderGrid = () => {
   715→    if (loading) {
   716→      return (
   717→        <div className="flex flex-col items-center gap-1">
   718→          {Array.from({ length: Math.min(rowsInput, 8) }).map((_, r) => (
   719→            <div key={r} className="flex gap-1">
   720→              {Array.from({ length: Math.min(colsInput, 8) }).map((_, c) => (
   721→                <Skeleton key={c} className="w-9 h-9 sm:w-10 sm:h-10 rounded-sm" />
   722→              ))}
   723→            </div>
   724→          ))}
   725→        </div>
   726→      );
   727→    }
   728→
   729→    return (
   730→      <div
   731→        ref={gridRef}
   732→        className="inline-grid gap-px bg-border rounded-md overflow-hidden"
   733→        style={{
   734→          gridTemplateColumns: `repeat(${grid[0]?.length ?? 0}, minmax(0, 1fr))`,
   735→        }}
   736→        tabIndex={0}
   737→        onClick={() => gridRef.current?.focus()}
   738→      >
   739→        {grid.map((row, r) =>
   740→          row.map((cell, c) => {
   741→            const isSelected =
   742→              selectedCell?.row === r && selectedCell?.col === c;
   743→            return (
   744→              <div
   745→                key={`${r}-${c}`}
   746→                className={[
   747→                  cellSize,
   748→                  "relative flex items-center justify-center select-none cursor-pointer transition-colors",
   749→                  cell.isBlack
   750→                    ? "bg-black/90 hover:bg-black/80"
   751→                    : "bg-white hover:bg-gray-50",
   752→                  isSelected && !cell.isBlack && "ring-2 ring-blue-500 ring-inset z-10",
   753→                  isSelected && cell.isBlack && "ring-2 ring-blue-500 ring-inset z-10",
   754→                ].join(" ")}
   755→                onClick={() => handleCellClick(r, c)}
   756→                onMouseEnter={() => {
   757→                  // Hover preview for future drag-to-paint
   758→                }}
   759→              >
   760→                {cell.number !== null && (
   761→                  <span className="absolute top-0 left-0.5 text-[9px] sm:text-[10px] font-bold text-gray-500 leading-none">
   762→                    {cell.number}
   763→                  </span>
   764→                )}
   765→                {cell.letter && (
   766→                  <span className="font-bold text-black">{cell.letter}</span>
   767→                )}
   768→              </div>
   769→            );
   770→          })
   771→        )}
   772→      </div>
   773→    );
   774→  };
   775→
   776→  const renderClueList = (
   777→    direction: "across" | "down",
   778→    list: EditorClue[]
   779→  ) => {
   780→    if (list.length === 0) {
   781→      return (
   782→        <p className="text-sm text-muted-foreground italic py-4">
   783→          Aucun mot détecté. Entrez des lettres sur la grille pour former des mots de 2+ lettres.

   786→    }
   787→    return (
   788→      <div className="space-y-3">
   789→        {list.map((clue) => (

   791→            key={`${clue.number}-${clue.direction}`}
   792→            className="space-y-1.5 p-3 rounded-lg border bg-card"

   794→            <div className="flex items-center gap-2">
   795→              <Badge variant="outline" className="shrink-0 text-xs">
   796→                {clue.number}
   797→              </Badge>
   798→              <span className="text-xs text-muted-foreground">
   799→                ({clue.row + 1},{clue.col + 1})
   800→              </span>
   801→            </div>
   802→            <div className="space-y-1">
   803→              <Label className="text-xs text-muted-foreground">Mot</Label>
   804→              <Input
   805→                value={clue.word}
   806→                onChange={(e) =>
   807→                  handleWordChange(
   808→                    clue.number,
   809→                    clue.direction,
   810→                    e.target.value,
   811→                    clue.row,
   812→                    clue.col
   813→                  )
   814→                }
   815→                placeholder="Entrez le mot"
   816→                className="h-8 text-sm font-mono uppercase"
   817→                maxLength={20}
   818→              />
   819→            </div>
   820→            <div className="space-y-1">
   821→              <Label className="text-xs text-muted-foreground">Indice</Label>
   822→              <Textarea
   823→                value={clue.text}
   824→                onChange={(e) =>
   825→                  handleClueTextChange(clue.number, clue.direction, e.target.value)
   826→                }
   827→                placeholder="Entrez l'indice"
   828→                className="text-sm min-h-[60px] resize-y"
   829→                rows={2}
   830→              />
   831→            </div>
   832→          </div>
   833→        ))}
   834→      </div>
   835→    );
   836→  };
   837→
   838→  // ── Main render ──────────────────────────────────────────────────
   839→  return (
   840→    <Dialog open={open} onOpenChange={onOpenChange}>
   841→      <DialogContent
   842→        className="
   843→          max-w-[calc(100vw-1rem)] sm:max-w-[calc(100vw-2rem)] md:max-w-[95vw] lg:max-w-[90vw]
   844→          h-[calc(100vh-1rem)] sm:h-[calc(100vh-2rem)]
   845→          flex flex-col p-0 gap-0 overflow-hidden
   846→        "
   847→        showCloseButton
   848→      >
   849→        <DialogHeader className="px-4 pt-4 pb-2 shrink-0">
   850→          <DialogTitle>
   851→            {isEditing ? "Modifier le puzzle" : "Nouveau puzzle"}
   852→          </DialogTitle>
   853→          <DialogDescription>
   854→            {isEditing
   855→              ? "Modifiez la grille, les mots et les indices du puzzle."
   856→              : "Créez un nouveau mots croisés en dessinant la grille et en entrant les mots."}
   857→          </DialogDescription>
   858→        </DialogHeader>
   859→
   860→        {/* ── Puzzle settings ─────────────────────────────────────── */}
   861→        <div className="px-4 pb-2 shrink-0 space-y-2">
   862→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-6 gap-2">
   863→            <div className="space-y-1">
   864→              <Label htmlFor="puzzle-title" className="text-xs">
   865→                Titre *
   866→              </Label>
   867→              <Input
   868→                id="puzzle-title"
   869→                value={title}
   870→                onChange={(e) => setTitle(e.target.value)}
   871→                placeholder="Titre du puzzle"
   872→                className="h-8 text-sm"
   873→              />
   874→            </div>
   875→            <div className="space-y-1">
   876→              <Label htmlFor="puzzle-desc" className="text-xs">
   877→                Description
   878→              </Label>
   879→              <Input
   880→                id="puzzle-desc"
   881→                value={description}
   882→                onChange={(e) => setDescription(e.target.value)}
   883→                placeholder="Description optionnelle"
   884→                className="h-8 text-sm"
   885→              />
   886→            </div>
   887→            <div className="space-y-1">
   888→              <Label className="text-xs">Langue</Label>
   889→              <Select value={language} onValueChange={setLanguage}>
   890→                <SelectTrigger className="h-8 text-sm w-full">
   891→                  <Languages className="size-3 mr-1" />
   892→                  <SelectValue />
   893→                </SelectTrigger>
   894→                <SelectContent>
   895→                  <SelectItem value="fr">🇫🇷 Français</SelectItem>
   896→                  <SelectItem value="en">🇬🇧 English</SelectItem>
   897→                </SelectContent>
   898→              </Select>
   899→            </div>
   900→            <div className="space-y-1">
   901→              <Label className="text-xs">Catégorie</Label>
   902→              <Select value={categoryId} onValueChange={setCategoryId}>
   903→                <SelectTrigger className="h-8 text-sm w-full">
   904→                  <FolderOpen className="size-3 mr-1" />
   905→                  <SelectValue placeholder="Sans catégorie" />
   906→                </SelectTrigger>
   907→                <SelectContent>
   908→                  <SelectItem value={noneCategory}>Sans catégorie</SelectItem>
   909→                  {categories.map((cat) => (
   910→                    <SelectItem key={cat.id} value={cat.id}>
   911→                      {cat.icon || "🏷️"} {cat.language === "en" ? "🇬🇧" : "🇫🇷"} {cat.name}
   912→                    </SelectItem>
   913→                  ))}
   914→                </SelectContent>
   915→              </Select>
   916→            </div>
   917→            <div className="space-y-1">
   918→              <Label className="text-xs">Collection</Label>
   919→              <Select value={packId} onValueChange={setPackId}>
   920→                <SelectTrigger className="h-8 text-sm w-full">
   921→                  <Package className="size-3 mr-1" />
   922→                  <SelectValue placeholder="Sans collection" />
   923→                </SelectTrigger>
   924→                <SelectContent>
   925→                  <SelectItem value={nonePack}>Sans collection</SelectItem>
   926→                  {packs.map((pack) => (
   927→                    <SelectItem key={pack.id} value={pack.id}>
   928→                      {pack.icon || "📦"} {pack.language === "en" ? "🇬🇧" : "🇫🇷"} {pack.name}
   929→                    </SelectItem>
   930→                  ))}
   931→                </SelectContent>
   932→              </Select>
   933→            </div>
   934→            <div className="space-y-1">
   935→              <Label className="text-xs">Difficulté</Label>
   936→              <Select value={difficulty} onValueChange={setDifficulty}>
   937→                <SelectTrigger className="h-8 text-sm w-full">
   938→                  <SelectValue />
   939→                </SelectTrigger>
   940→                <SelectContent>
   941→                  <SelectItem value="1">Facile</SelectItem>
   942→                  <SelectItem value="2">Moyen</SelectItem>
   943→                  <SelectItem value="3">Difficile</SelectItem>
   944→                </SelectContent>
   945→              </Select>
   946→            </div>
   947→          </div>
   948→        </div>
   949→
   950→        <Separator className="shrink-0" />
   951→
   952→        {/* ── Grid size + toolbar ────────────────────────────────── */}
   953→        <div className="px-4 py-2 shrink-0 space-y-2">
   954→          {/* Grid size presets */}
   955→          <div className="flex flex-wrap items-center gap-1.5">
   956→            {gridSizePresets.map((preset) => {
   957→              const Icon = preset.icon;
   958→              const isActive = rowsInput === parseInt(preset.id.split('x')[0], 10) && colsInput === parseInt(preset.id.split('x')[1], 10);
   959→              return (
   960→                <button
   961→                  key={preset.id}
   962→                  type="button"
   963→                  onClick={() => handlePresetGridSize(preset.id)}
   964→                  className={`flex items-center gap-1 border rounded-md px-2 py-1 text-[11px] font-medium transition-colors ${
   965→                    isActive
   966→                      ? 'bg-primary text-primary-foreground border-primary'
   967→                      : 'bg-background text-muted-foreground hover:bg-muted hover:text-foreground'
   968→                  }`}
   969→                >
   970→                  <Icon className="size-3" />
   971→                  {preset.label}
   972→                </button>
   973→              );
   974→            })}
   975→            <div className="flex items-center gap-1.5 ml-auto">
   976→              <Label className="text-xs whitespace-nowrap">Lignes</Label>
   977→              <Input
   978→                type="number"
   979→                min={2}
   980→                max={20}
   981→                value={rowsInput}
   982→                onChange={(e) => setRowsInput(Number(e.target.value))}
   983→                className="h-7 w-14 text-sm text-center"
   984→              />
   985→            </div>
   986→            <div className="flex items-center gap-1.5">
   987→              <Label className="text-xs whitespace-nowrap">Colonnes</Label>
   988→              <Input
   989→                type="number"
   990→                min={2}
   991→                max={20}
   992→                value={colsInput}
   993→                onChange={(e) => setColsInput(Number(e.target.value))}
   994→                className="h-7 w-14 text-sm text-center"
   995→              />
   996→            </div>
   997→            <Button
   998→              variant="outline"
   999→              size="sm"
  1000→              className="h-7 text-xs"
  1001→              onClick={handleResizeGrid}
  1002→            >
  1003→              Appliquer
  1004→            </Button>
  1005→          </div>
  1006→          {/* Action buttons */}
  1007→          <div className="flex items-center gap-1.5">
  1008→            <Button
  1009→              variant="outline"
  1010→              size="sm"
  1011→              className="h-7 text-xs"
  1012→              onClick={handleClearLetters}
  1013→            >
  1014→              Grille vide
  1015→            </Button>
  1016→            <Button
  1017→              variant="outline"
  1018→              size="sm"
  1019→              className="h-7 text-xs text-destructive hover:text-destructive"
  1020→              onClick={handleClearAll}
  1021→            >

  1037→        {/* ── Main content: grid + clues ──────────────────────────── */}
  1038→        <div className="flex-1 overflow-hidden flex flex-col lg:flex-row min-h-0">
  1039→          {/* Grid area */}

  1043→            </div>
  1044→          </div>
  1045→
  1046→          {/* Clue editor sidebar (right on desktop, below on mobile) */}
  1047→          <div className="lg:w-[360px] xl:w-[400px] border-t lg:border-t-0 lg:border-l overflow-hidden flex flex-col min-h-0">
  1048→            <Tabs
  1049→              value={activeClueTab}
  1050→              onValueChange={setActiveClueTab}
  1051→              className="flex flex-col h-full"
  1052→            >
  1053→              <div className="shrink-0 px-2 pt-2">
  1054→                <TabsList className="w-full">
