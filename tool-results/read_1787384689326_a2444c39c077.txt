   161→function formatDate(dateStr: string | null): string {
   162→  if (!dateStr) return '—';
   163→  try {
   164→    return format(new Date(dateStr), 'dd MMM yyyy', { locale: fr });
   165→  } catch {
   166→    return '—';
   167→  }
   168→}

   170→function languageLabel(lang: string) {
   171→  return lang === 'en' ? '🇬🇧 EN' : '🇫🇷 FR';

   214→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {
   215→  const [newName, setNewName] = useState('');
   216→  const [newIcon, setNewIcon] = useState('🏷️');
   217→  const [newLanguage, setNewLanguage] = useState('fr');
   218→  const [creating, setCreating] = useState(false);
   219→
   220→  // Edit state
   221→  const [editingId, setEditingId] = useState<string | null>(null);
   222→  const [editName, setEditName] = useState('');
   223→  const [editIcon, setEditIcon] = useState('');
   224→  const [editLanguage, setEditLanguage] = useState('');
   225→  const [saving, setSaving] = useState(false);
   226→
   227→  // Delete state
   228→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   229→  const [deleting, setDeleting] = useState(false);
   230→
   231→  const handleCreate = async () => {
   232→    if (!newName.trim()) return;
   233→    setCreating(true);
   234→    try {
   235→      const res = await fetch('/api/categories', {
   236→        method: 'POST',
   237→        headers: { 'Content-Type': 'application/json' },
   238→        body: JSON.stringify({ name: newName.trim(), icon: newIcon, language: newLanguage }),
   239→      });
   240→      if (!res.ok) {
   241→        const err = await res.json().catch(() => ({}));
   242→        throw new Error((err as { error?: string }).error || 'Erreur');
   243→      }
   244→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   245→      setNewName('');
   246→      setNewIcon('🏷️');
   247→      onRefresh();
   248→    } catch (err) {
   249→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   250→    } finally {
   251→      setCreating(false);
   252→    }
   253→  };
   254→
   255→  const startEdit = (cat: CategoryItem) => {
   256→    setEditingId(cat.id);
   257→    setEditName(cat.name);
   258→    setEditIcon(cat.icon || '🏷️');
   259→    setEditLanguage(cat.language);
   260→  };
   261→
   262→  const handleSaveEdit = async () => {
   263→    if (!editingId || !editName.trim()) return;
   264→    setSaving(true);
   265→    try {
   266→      const res = await fetch('/api/categories', {
   267→        method: 'PUT',
   268→        headers: { 'Content-Type': 'application/json' },
   269→        body: JSON.stringify({ id: editingId, name: editName.trim(), icon: editIcon, language: editLanguage }),
   270→      });
   271→      if (!res.ok) {
   272→        const err = await res.json().catch(() => ({}));
   273→        throw new Error((err as { error?: string }).error || 'Erreur');
   274→      }
   275→      toast.success('Catégorie mise à jour');
   276→      setEditingId(null);
   277→      onRefresh();
   278→    } catch (err) {
   279→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   280→    } finally {
   281→      setSaving(false);
   282→    }
   283→  };
   284→
   285→  const handleDelete = async () => {
   286→    if (!deleteTarget) return;
   287→    setDeleting(true);
   288→    try {
   289→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   290→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   291→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   292→      setDeleteTarget(null);
   293→      onRefresh();
   294→    } catch (err) {
   295→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   296→    } finally {
   297→      setDeleting(false);
   298→    }
   299→  };
   300→
   301→  return (
   302→    <>
   303→      {/* Create new category row */}
   304→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   305→        <div className="w-16 sm:w-20 space-y-1">
   306→          <Label className="text-xs text-muted-foreground">Icône</Label>
   307→          <Input
   308→            placeholder="🏷️"
   309→            value={newIcon}
   310→            onChange={(e) => setNewIcon(e.target.value)}
   311→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   312→            className="h-8 text-sm text-center"
   313→            maxLength={4}
   314→          />
   315→        </div>
   316→        <div className="flex-1 w-full sm:w-auto space-y-1">
   317→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   318→          <Input
   319→            placeholder="Ex: Animaux, Sciences..."
   320→            value={newName}
   321→            onChange={(e) => setNewName(e.target.value)}
   322→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   323→            className="h-8 text-sm"
   324→          />
   325→        </div>
   326→        <div className="w-full sm:w-28 space-y-1">
   327→          <Label className="text-xs text-muted-foreground">Langue</Label>
   328→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   329→            <SelectTrigger className="h-8 text-sm">
   330→              <SelectValue />
   331→            </SelectTrigger>
   332→            <SelectContent>
   333→              <SelectItem value="fr">🇫🇷 Français</SelectItem>
   334→              <SelectItem value="en">🇬🇧 English</SelectItem>
   335→            </SelectContent>
   336→          </Select>
   337→        </div>
   338→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   339→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   340→          Ajouter
   341→        </Button>
   342→      </div>
   343→
   344→      {/* Category list */}
   345→      {categories.length === 0 ? (
   346→        <div className="text-center py-6 text-muted-foreground text-sm">
   347→          Aucune catégorie. Créez-en une ci-dessus.
   348→        </div>
   349→      ) : (
   350→        <div className="rounded-md border max-h-64 overflow-y-auto">
   351→          <Table>
   352→            <TableHeader>
   353→              <TableRow className="hover:bg-transparent">
   354→                <TableHead className="pl-3">Nom</TableHead>
   355→                <TableHead>Langue</TableHead>
   356→                <TableHead>Puzzles</TableHead>
   357→                <TableHead className="text-right pr-3">Actions</TableHead>
   358→              </TableRow>
   359→            </TableHeader>
   360→            <TableBody>
   361→              {categories.map((cat) => (
   362→                <TableRow key={cat.id}>
   363→                  <TableCell className="font-medium pl-3">
   364→                    {editingId === cat.id ? (
   365→                      <div className="flex items-center gap-1.5">
   366→                        <Input
   367→                          value={editIcon}
   368→                          onChange={(e) => setEditIcon(e.target.value)}
   369→                          className="h-7 text-sm w-12 text-center"
   370→                          maxLength={4}
   371→                          autoFocus
   372→                        />
   373→                        <Input
   374→                          value={editName}
   375→                          onChange={(e) => setEditName(e.target.value)}
   376→                          onKeyDown={(e) => e.key === 'Enter' && handleSaveEdit()}
   377→                          className="h-7 text-sm w-full"
   378→                        />
   379→                      </div>
   380→                    ) : (
   381→                      <span className="flex items-center gap-1.5">
   382→                        <span className="text-sm">{cat.icon || '🏷️'}</span>
   383→                        {cat.name}
   384→                      </span>
   385→                    )}
   386→                  </TableCell>
   387→                  <TableCell>
   388→                    {editingId === cat.id ? (
   389→                      <Select value={editLanguage} onValueChange={setEditLanguage}>
   390→                        <SelectTrigger className="h-7 text-xs w-24">
   391→                          <SelectValue />
   392→                        </SelectTrigger>
   393→                        <SelectContent>
   394→                          <SelectItem value="fr">🇫🇷 FR</SelectItem>
   395→                          <SelectItem value="en">🇬🇧 EN</SelectItem>
   396→                        </SelectContent>
   397→                      </Select>
   398→                    ) : (
   399→                      <Badge variant="outline" className="text-xs">
   400→                        {languageLabel(cat.language)}
   401→                      </Badge>
   402→                    )}
   403→                  </TableCell>
   404→                  <TableCell className="text-muted-foreground text-xs">
   405→                    {cat._count.puzzles}
   406→                  </TableCell>
   407→                  <TableCell className="text-right pr-3">
   408→                    <div className="flex items-center justify-end gap-1">
   409→                      {editingId === cat.id ? (
   410→                        <>
   411→                          <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={() => setEditingId(null)}>
   412→                            Annuler
   413→                          </Button>
   414→                          <Button size="sm" className="h-7 text-xs gap-1" onClick={handleSaveEdit} disabled={saving}>
   415→                            {saving ? <Loader2 className="size-3 animate-spin" /> : <Save className="size-3" />}
   416→                            OK
   417→                          </Button>
   418→                        </>
   419→                      ) : (
   420→                        <>
   421→                          <Button
   422→                            variant="ghost"
   423→                            size="sm"
   424→                            className="h-7 text-xs gap-1"
   425→                            onClick={() => startEdit(cat)}
   426→                          >
   427→                            <Pencil className="size-3" />
   428→                          </Button>
   429→                          <Button
   430→                            variant="ghost"
   431→                            size="sm"
   432→                            className="h-7 text-xs gap-1 text-destructive hover:text-destructive"
   433→                            onClick={() => setDeleteTarget(cat)}
   434→                          >
   435→                            <Trash2 className="size-3" />
   436→                          </Button>
   437→                        </>
   438→                      )}
   439→                    </div>
   440→                  </TableCell>
   441→                </TableRow>
   442→              ))}
   443→            </TableBody>
   444→          </Table>
   445→        </div>
   446→      )}
   447→
   448→      {/* Delete confirmation */}
   449→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   450→        <AlertDialogContent>
   451→          <AlertDialogHeader>
   452→            <AlertDialogTitle>Supprimer la catégorie</AlertDialogTitle>
   453→            <AlertDialogDescription>
   454→              Êtes-vous sûr de vouloir supprimer « {deleteTarget?.name} » ? Les puzzles associés ne seront pas supprimés mais seront décatégorisés.
   455→            </AlertDialogDescription>
   456→          </AlertDialogHeader>
   457→          <AlertDialogFooter>
   458→            <AlertDialogCancel disabled={deleting}>Annuler</AlertDialogCancel>
   459→            <AlertDialogAction
   460→              onClick={handleDelete}
   461→              disabled={deleting}
   462→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   463→            >
   464→              {deleting ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   465→              Supprimer
   466→            </AlertDialogAction>
   467→          </AlertDialogFooter>
   468→        </AlertDialogContent>
   469→      </AlertDialog>
   470→    </>
   471→  );
   472→}
