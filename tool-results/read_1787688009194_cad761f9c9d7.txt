   217→   217→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {
   218→   218→  const [newName, setNewName] = useState('');
   219→   219→  const [newIcon, setNewIcon] = useState('🏷️');
   220→   220→  const [newLanguage, setNewLanguage] = useState('fr');
   221→   221→  const [creating, setCreating] = useState(false);
   222→   222→
   223→   223→  // Edit state
   224→   224→  const [editingId, setEditingId] = useState<string | null>(null);
   225→   225→  const [editName, setEditName] = useState('');
   226→   226→  const [editIcon, setEditIcon] = useState('');
   227→   227→  const [editLanguage, setEditLanguage] = useState('');
   228→   228→  const [saving, setSaving] = useState(false);
   229→   229→
   230→   230→  // Delete state
   231→   231→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   232→   232→  const [deleting, setDeleting] = useState(false);
   233→   233→
   234→   234→  const handleCreate = async () => {
   235→   235→    if (!newName.trim()) return;
   236→   236→    setCreating(true);
   237→   237→    try {
   238→   238→      const res = await fetch('/api/categories', {
   239→   239→        method: 'POST',
   240→   240→        headers: { 'Content-Type': 'application/json' },
   241→   241→        body: JSON.stringify({ name: newName.trim(), icon: newIcon, language: newLanguage }),
   242→   242→      });
   243→   243→      if (!res.ok) {
   244→   244→        const err = await res.json().catch(() => ({}));
   245→   245→        throw new Error((err as { error?: string }).error || 'Erreur');
   246→   246→      }
   247→   247→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   248→   248→      setNewName('');
   249→   249→      setNewIcon('🏷️');
   250→   250→      onRefresh();
   251→   251→    } catch (err) {
   252→   252→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   253→   253→    } finally {
   254→   254→      setCreating(false);
   255→   255→    }
   256→   256→  };
   257→   257→
   258→   258→  const startEdit = (cat: CategoryItem) => {
   259→   259→    setEditingId(cat.id);
   260→   260→    setEditName(cat.name);
   261→   261→    setEditIcon(cat.icon || '🏷️');
   262→   262→    setEditLanguage(cat.language);
   263→   263→  };
   264→   264→
   265→   265→  const handleSaveEdit = async () => {
   266→   266→    if (!editingId || !editName.trim()) return;
   267→   267→    setSaving(true);
   268→   268→    try {
   269→   269→      const res = await fetch('/api/categories', {
   270→   270→        method: 'PUT',
   271→   271→        headers: { 'Content-Type': 'application/json' },
   272→   272→        body: JSON.stringify({ id: editingId, name: editName.trim(), icon: editIcon, language: editLanguage }),
   273→   273→      });
   274→   274→      if (!res.ok) {
   275→   275→        const err = await res.json().catch(() => ({}));
   276→   276→        throw new Error((err as { error?: string }).error || 'Erreur');
   277→   277→      }
   278→   278→      toast.success('Catégorie mise à jour');
   279→   279→      setEditingId(null);
   280→   280→      onRefresh();
   281→   281→    } catch (err) {
   282→   282→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   283→   283→    } finally {
   284→   284→      setSaving(false);
   285→   285→    }
   286→   286→  };
   287→   287→
   288→   288→  const handleDelete = async () => {
   289→   289→    if (!deleteTarget) return;
   290→   290→    setDeleting(true);
   291→   291→    try {
   292→   292→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   293→   293→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   294→   294→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   295→   295→      setDeleteTarget(null);
   296→   296→      onRefresh();
   297→   297→    } catch (err) {
   298→   298→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   299→   299→    } finally {
   300→   300→      setDeleting(false);
   301→   301→    }
   302→   302→  };
   303→   303→
   304→   304→  return (
   305→   305→    <>
   306→   306→      {/* Create new category row */}
   307→   307→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   308→   308→        <div className="w-16 sm:w-20 space-y-1">
   309→   309→          <Label className="text-xs text-muted-foreground">Icône</Label>
   310→   310→          <Input
   311→   311→            placeholder="🏷️"
   312→   312→            value={newIcon}
   313→   313→            onChange={(e) => setNewIcon(e.target.value)}
   314→   314→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   315→   315→            className="h-8 text-sm text-center"
   316→   316→            maxLength={4}
   317→   317→          />
   318→   318→        </div>
   319→   319→        <div className="flex-1 w-full sm:w-auto space-y-1">
   320→   320→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   321→   321→          <Input
   322→   322→            placeholder="Ex: Animaux, Sciences..."
   323→   323→            value={newName}
   324→   324→            onChange={(e) => setNewName(e.target.value)}
   325→   325→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   326→   326→            className="h-8 text-sm"
   327→   327→          />
   328→   328→        </div>
   329→   329→        <div className="w-full sm:w-28 space-y-1">
   330→   330→          <Label className="text-xs text-muted-foreground">Langue</Label>
   331→   331→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   332→   332→            <SelectTrigger className="h-8 text-sm">
   333→   333→              <SelectValue />
   334→   334→            </SelectTrigger>
   335→   335→            <SelectContent>
   336→   336→              <SelectItem value="fr">🇫🇷 Français</SelectItem>
   337→   337→              <SelectItem value="en">🇬🇧 English</SelectItem>
   338→   338→            </SelectContent>
   339→   339→          </Select>
   340→   340→        </div>
   341→   341→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   342→   342→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   343→   343→          Ajouter
   344→   344→        </Button>
   345→   345→      </div>
   346→   346→
   347→   347→      {/* Category list */}
   348→   348→      {categories.length === 0 ? (
   349→   349→        <div className="text-center py-6 text-muted-foreground text-sm">
   350→   350→          Aucune catégorie. Créez-en une ci-dessus.

   352→   352→      ) : (
   353→   353→        <div className="rounded-md border max-h-64 overflow-y-auto">
   354→   354→          <Table>
   355→   355→            <TableHeader>
   356→   356→              <TableRow className="hover:bg-transparent">
   357→   357→                <TableHead className="pl-3">Nom</TableHead>
   358→   358→                <TableHead>Langue</TableHead>
   359→   359→                <TableHead>Puzzles</TableHead>
   360→   360→                <TableHead className="text-right pr-3">Actions</TableHead>
   361→   361→              </TableRow>
   362→   362→            </TableHeader>
   363→   363→            <TableBody>
   364→   364→              {categories.map((cat) => (
   365→   365→                <TableRow key={cat.id}>
   366→   366→                  <TableCell className="font-medium pl-3">
   367→   367→                    {editingId === cat.id ? (
   368→   368→                      <div className="flex items-center gap-1.5">
   369→   369→                        <Input
   370→   370→                          value={editIcon}
   371→   371→                          onChange={(e) => setEditIcon(e.target.value)}
   372→   372→                          className="h-7 text-sm w-12 text-center"
   373→   373→                          maxLength={4}
   374→   374→                          autoFocus
   375→   375→                        />
   376→   376→                        <Input
   377→   377→                          value={editName}
   378→   378→                          onChange={(e) => setEditName(e.target.value)}
   379→   379→                          onKeyDown={(e) => e.key === 'Enter' && handleSaveEdit()}

   384→   384→                      <span className="flex items-center gap-1.5">
   385→   385→                        <span className="text-sm">{cat.icon || '🏷️'}</span>
   386→   386→                        {cat.name}
   387→   387→                      </span>
   388→   388→                    )}
   389→   389→                  </TableCell>
   390→   390→                  <TableCell>
