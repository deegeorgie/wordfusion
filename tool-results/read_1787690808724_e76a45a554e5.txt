   167→function formatDate(dateStr: string | null): string {
   168→  if (!dateStr) return '—';
   169→  try {
   170→    return format(new Date(dateStr), 'dd MMM yyyy', { locale: fr });
   171→  } catch {
   172→    return '—';
   173→  }
   174→}

   220→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {
   221→  const [newName, setNewName] = useState('');
   222→  const [newIcon, setNewIcon] = useState('🏷️');
   223→  const [newLanguage, setNewLanguage] = useState('fr');
   224→  const [creating, setCreating] = useState(false);
   225→
   226→  // Edit state
   227→  const [editingId, setEditingId] = useState<string | null>(null);
   228→  const [editName, setEditName] = useState('');
   229→  const [editIcon, setEditIcon] = useState('');
   230→  const [editLanguage, setEditLanguage] = useState('');
   231→  const [saving, setSaving] = useState(false);
   232→
   233→  // Delete state
   234→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   235→  const [deleting, setDeleting] = useState(false);
   236→
   237→  const handleCreate = async () => {
   238→    if (!newName.trim()) return;
   239→    setCreating(true);
   240→    try {
   241→      const res = await fetch('/api/categories', {
   242→        method: 'POST',
   243→        headers: { 'Content-Type': 'application/json' },
   244→        body: JSON.stringify({ name: newName.trim(), icon: newIcon, language: newLanguage }),
   245→      });
   246→      if (!res.ok) {
   247→        const err = await res.json().catch(() => ({}));
   248→        throw new Error((err as { error?: string }).error || 'Erreur');
   249→      }
   250→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   251→      setNewName('');
   252→      setNewIcon('🏷️');
   253→      onRefresh();
   254→    } catch (err) {
   255→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   256→    } finally {
   257→      setCreating(false);
   258→    }
   259→  };
   260→
   261→  const startEdit = (cat: CategoryItem) => {
   262→    setEditingId(cat.id);
   263→    setEditName(cat.name);
   264→    setEditIcon(cat.icon || '🏷️');
   265→    setEditLanguage(cat.language);
   266→  };
   267→
   268→  const handleSaveEdit = async () => {
   269→    if (!editingId || !editName.trim()) return;
   270→    setSaving(true);
   271→    try {
   272→      const res = await fetch('/api/categories', {
   273→        method: 'PUT',
   274→        headers: { 'Content-Type': 'application/json' },
   275→        body: JSON.stringify({ id: editingId, name: editName.trim(), icon: editIcon, language: editLanguage }),
   276→      });
   277→      if (!res.ok) {
   278→        const err = await res.json().catch(() => ({}));
   279→        throw new Error((err as { error?: string }).error || 'Erreur');
   280→      }
   281→      toast.success('Catégorie mise à jour');
   282→      setEditingId(null);
   283→      onRefresh();
   284→    } catch (err) {
   285→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   286→    } finally {
   287→      setSaving(false);
   288→    }
   289→  };
   290→
   291→  const handleDelete = async () => {
   292→    if (!deleteTarget) return;
   293→    setDeleting(true);
   294→    try {
   295→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   296→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   297→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   298→      setDeleteTarget(null);
   299→      onRefresh();
   300→    } catch (err) {
   301→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   302→    } finally {
   303→      setDeleting(false);
   304→    }
   305→  };
   306→
   307→  return (
   308→    <>
   309→      {/* Create new category row */}
   310→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   311→        <div className="w-16 sm:w-20 space-y-1">
   312→          <Label className="text-xs text-muted-foreground">Icône</Label>
   313→          <Input
   314→            placeholder="🏷️"
   315→            value={newIcon}
   316→            onChange={(e) => setNewIcon(e.target.value)}
   317→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   318→            className="h-8 text-sm text-center"
   319→            maxLength={4}
   320→          />
   321→        </div>
   322→        <div className="flex-1 w-full sm:w-auto space-y-1">
   323→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   324→          <Input
   325→            placeholder="Ex: Animaux, Sciences..."
   326→            value={newName}
   327→            onChange={(e) => setNewName(e.target.value)}
   328→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   329→            className="h-8 text-sm"
   330→          />
   331→        </div>
   332→        <div className="w-full sm:w-28 space-y-1">
   333→          <Label className="text-xs text-muted-foreground">Langue</Label>
   334→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   335→            <SelectTrigger className="h-8 text-sm">
   336→              <SelectValue />
   337→            </SelectTrigger>
   338→            <SelectContent>
   339→              <SelectItem value="fr">🇫🇷 Français</SelectItem>
   340→              <SelectItem value="en">🇬🇧 English</SelectItem>
   341→            </SelectContent>
   342→          </Select>
   343→        </div>
   344→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   345→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   346→          Ajouter
   347→        </Button>
   348→      </div>
   349→
   350→      {/* Category list */}
   351→      {categories.length === 0 ? (
   352→        <div className="text-center py-6 text-muted-foreground text-sm">
   353→          Aucune catégorie. Créez-en une ci-dessus.
   354→        </div>
   355→      ) : (
   356→        <div className="rounded-md border max-h-64 overflow-y-auto">
   357→          <Table>
   358→            <TableHeader>
   359→              <TableRow className="hover:bg-transparent">
   360→                <TableHead className="pl-3">Nom</TableHead>
   361→                <TableHead>Langue</TableHead>
   362→                <TableHead>Puzzles</TableHead>
   363→                <TableHead className="text-right pr-3">Actions</TableHead>
   364→              </TableRow>
   365→            </TableHeader>
   366→            <TableBody>
   367→              {categories.map((cat) => (
   368→                <TableRow key={cat.id}>
   369→                  <TableCell className="font-medium pl-3">
   370→                    {editingId === cat.id ? (
   371→                      <div className="flex items-center gap-1.5">
   372→                        <Input
   373→                          value={editIcon}
   374→                          onChange={(e) => setEditIcon(e.target.value)}
   375→                          className="h-7 text-sm w-12 text-center"
   376→                          maxLength={4}
   377→                          autoFocus
   378→                        />
   379→                        <Input
   380→                          value={editName}
   381→                          onChange={(e) => setEditName(e.target.value)}
   382→                          onKeyDown={(e) => e.key === 'Enter' && handleSaveEdit()}
   383→                          className="h-7 text-sm w-full"
   384→                        />
   385→                      </div>
   386→                    ) : (
   387→                      <span className="flex items-center gap-1.5">
   388→                        <span className="text-sm">{cat.icon || '🏷️'}</span>
   389→                        {cat.name}
   390→                      </span>
   391→                    )}
   392→                  </TableCell>
   393→                  <TableCell>
   394→                    {editingId === cat.id ? (
   395→                      <Select value={editLanguage} onValueChange={setEditLanguage}>
   396→                        <SelectTrigger className="h-7 text-xs w-24">
   397→                          <SelectValue />
   398→                        </SelectTrigger>
   399→                        <SelectContent>
   400→                          <SelectItem value="fr">🇫🇷 FR</SelectItem>
   401→                          <SelectItem value="en">🇬🇧 EN</SelectItem>
   402→                        </SelectContent>
   403→                      </Select>
   404→                    ) : (
   405→                      <Badge variant="outline" className="text-xs">
   406→                        {languageLabel(cat.language)}
   407→                      </Badge>
   408→                    )}
   409→                  </TableCell>
   410→                  <TableCell className="text-muted-foreground text-xs">
   411→                    {cat._count.puzzles}
   412→                  </TableCell>
   413→                  <TableCell className="text-right pr-3">
   414→                    <div className="flex items-center justify-end gap-1">
   415→                      {editingId === cat.id ? (
   416→                        <>
   417→                          <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={() => setEditingId(null)}>
   418→                            Annuler
   419→                          </Button>
   420→                          <Button size="sm" className="h-7 text-xs gap-1" onClick={handleSaveEdit} disabled={saving}>
   421→                            {saving ? <Loader2 className="size-3 animate-spin" /> : <Save className="size-3" />}
   422→                            OK
   423→                          </Button>
   424→                        </>
   425→                      ) : (
   426→                        <>
   427→                          <Button
   428→                            variant="ghost"
   429→                            size="sm"
   430→                            className="h-7 text-xs gap-1"
   431→                            onClick={() => startEdit(cat)}
   432→                          >
   433→                            <Pencil className="size-3" />
   434→                          </Button>
   435→                          <Button
   436→                            variant="ghost"
   437→                            size="sm"
   438→                            className="h-7 text-xs gap-1 text-destructive hover:text-destructive"
   439→                            onClick={() => setDeleteTarget(cat)}
   440→                          >
   441→                            <Trash2 className="size-3" />
   442→                          </Button>
   443→                        </>
   444→                      )}
   445→                    </div>
   446→                  </TableCell>
   447→                </TableRow>
   448→              ))}
   449→            </TableBody>
   450→          </Table>
   451→        </div>
   452→      )}
   453→
   454→      {/* Delete confirmation */}
   455→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   456→        <AlertDialogContent>
   457→          <AlertDialogHeader>
   458→            <AlertDialogTitle>Supprimer la catégorie</AlertDialogTitle>
   459→            <AlertDialogDescription>
   460→              Êtes-vous sûr de vouloir supprimer « {deleteTarget?.name} » ? Les puzzles associés ne seront pas supprimés mais seront décatégorisés.
   461→            </AlertDialogDescription>
   462→          </AlertDialogHeader>
   463→          <AlertDialogFooter>
   464→            <AlertDialogCancel disabled={deleting}>Annuler</AlertDialogCancel>
   465→            <AlertDialogAction
   466→              onClick={handleDelete}
   467→              disabled={deleting}
   468→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   469→            >
   470→              {deleting ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   471→              Supprimer
   472→            </AlertDialogAction>
   473→          </AlertDialogFooter>
   474→        </AlertDialogContent>
   475→      </AlertDialog>
   476→    </>
   477→  );
   478→}
