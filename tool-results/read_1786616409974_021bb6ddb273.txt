   193→   193→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {
   194→   194→  const [newName, setNewName] = useState('');
   195→   195→  const [newLanguage, setNewLanguage] = useState('fr');
   196→   196→  const [creating, setCreating] = useState(false);
   197→   197→
   198→   198→  // Edit state
   199→   199→  const [editingId, setEditingId] = useState<string | null>(null);
   200→   200→  const [editName, setEditName] = useState('');
   201→   201→  const [editLanguage, setEditLanguage] = useState('');
   202→   202→  const [saving, setSaving] = useState(false);
   203→   203→
   204→   204→  // Delete state
   205→   205→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   206→   206→  const [deleting, setDeleting] = useState(false);
   207→   207→
   208→   208→  const handleCreate = async () => {
   209→   209→    if (!newName.trim()) return;
   210→   210→    setCreating(true);
   211→   211→    try {
   212→   212→      const res = await fetch('/api/categories', {
   213→   213→        method: 'POST',
   214→   214→        headers: { 'Content-Type': 'application/json' },
   215→   215→        body: JSON.stringify({ name: newName.trim(), language: newLanguage }),
   216→   216→      });
   217→   217→      if (!res.ok) {
   218→   218→        const err = await res.json().catch(() => ({}));
   219→   219→        throw new Error((err as { error?: string }).error || 'Erreur');
   220→   220→      }
   221→   221→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   222→   222→      setNewName('');
   223→   223→      onRefresh();
   224→   224→    } catch (err) {
   225→   225→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   226→   226→    } finally {
   227→   227→      setCreating(false);
   228→   228→    }
   229→   229→  };
   230→   230→
   231→   231→  const startEdit = (cat: CategoryItem) => {
   232→   232→    setEditingId(cat.id);
   233→   233→    setEditName(cat.name);
   234→   234→    setEditLanguage(cat.language);
   235→   235→  };
   236→   236→
   237→   237→  const handleSaveEdit = async () => {
   238→   238→    if (!editingId || !editName.trim()) return;
   239→   239→    setSaving(true);
   240→   240→    try {
   241→   241→      const res = await fetch('/api/categories', {
   242→   242→        method: 'PUT',
   243→   243→        headers: { 'Content-Type': 'application/json' },
   244→   244→        body: JSON.stringify({ id: editingId, name: editName.trim(), language: editLanguage }),
   245→   245→      });
   246→   246→      if (!res.ok) {
   247→   247→        const err = await res.json().catch(() => ({}));
   248→   248→        throw new Error((err as { error?: string }).error || 'Erreur');
   249→   249→      }
   250→   250→      toast.success('Catégorie mise à jour');
   251→   251→      setEditingId(null);
   252→   252→      onRefresh();
   253→   253→    } catch (err) {
   254→   254→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   255→   255→    } finally {
   256→   256→      setSaving(false);
   257→   257→    }
   258→   258→  };
   259→   259→
   260→   260→  const handleDelete = async () => {
   261→   261→    if (!deleteTarget) return;
   262→   262→    setDeleting(true);
   263→   263→    try {
   264→   264→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   265→   265→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   266→   266→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   267→   267→      setDeleteTarget(null);
   268→   268→      onRefresh();
   269→   269→    } catch (err) {
   270→   270→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   271→   271→    } finally {
   272→   272→      setDeleting(false);
   273→   273→    }
   274→   274→  };
   275→   275→
   276→   276→  return (
   277→   277→    <>
   278→   278→      {/* Create new category row */}
   279→   279→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   280→   280→        <div className="flex-1 w-full sm:w-auto space-y-1">
   281→   281→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   282→   282→          <Input
   283→   283→            placeholder="Ex: Animaux, Sciences..."
   284→   284→            value={newName}
   285→   285→            onChange={(e) => setNewName(e.target.value)}
   286→   286→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   287→   287→            className="h-8 text-sm"
   288→   288→          />
   289→   289→        </div>
   290→   290→        <div className="w-full sm:w-28 space-y-1">
   291→   291→          <Label className="text-xs text-muted-foreground">Langue</Label>
   292→   292→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   293→   293→            <SelectTrigger className="h-8 text-sm">
   294→   294→              <SelectValue />
   295→   295→            </SelectTrigger>
   296→   296→            <SelectContent>
   297→   297→              <SelectItem value="fr">🇫🇷 Français</SelectItem>
   298→   298→              <SelectItem value="en">🇬🇧 English</SelectItem>
   299→   299→            </SelectContent>
   300→   300→          </Select>
   301→   301→        </div>
   302→   302→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   303→   303→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   304→   304→          Ajouter
   305→   305→        </Button>
   306→   306→      </div>
   307→   307→
   308→   308→      {/* Category list */}
   309→   309→      {categories.length === 0 ? (
   310→   310→        <div className="text-center py-6 text-muted-foreground text-sm">
   311→   311→          Aucune catégorie. Créez-en une ci-dessus.
   312→   312→        </div>
   313→   313→      ) : (
   314→   314→        <div className="rounded-md border max-h-64 overflow-y-auto">
   315→   315→          <Table>
   316→   316→            <TableHeader>
   317→   317→              <TableRow className="hover:bg-transparent">
   318→   318→                <TableHead className="pl-3">Nom</TableHead>
   319→   319→                <TableHead>Langue</TableHead>
   320→   320→                <TableHead>Puzzles</TableHead>
   321→   321→                <TableHead className="text-right pr-3">Actions</TableHead>
   322→   322→              </TableRow>
   323→   323→            </TableHeader>
   324→   324→            <TableBody>
   325→   325→              {categories.map((cat) => (
   326→   326→                <TableRow key={cat.id}>
   327→   327→                  <TableCell className="font-medium pl-3">
   328→   328→                    {editingId === cat.id ? (
   329→   329→                      <Input
   330→   330→                        value={editName}
   331→   331→                        onChange={(e) => setEditName(e.target.value)}
   332→   332→                        onKeyDown={(e) => e.key === 'Enter' && handleSaveEdit()}
   333→   333→                        className="h-7 text-sm w-full"
   334→   334→                        autoFocus
   335→   335→                      />
   336→   336→                    ) : (
   337→   337→                      <span className="flex items-center gap-1.5">
   338→   338→                        <Tag className="size-3 text-muted-foreground" />
   339→   339→                        {cat.name}
   340→   340→                      </span>
   341→   341→                    )}
   342→   342→                  </TableCell>
   343→   343→                  <TableCell>
   344→   344→                    {editingId === cat.id ? (
   345→   345→                      <Select value={editLanguage} onValueChange={setEditLanguage}>
   346→   346→                        <SelectTrigger className="h-7 text-xs w-24">
   347→   347→                          <SelectValue />
   348→   348→                        </SelectTrigger>
   349→   349→                        <SelectContent>
   350→   350→                          <SelectItem value="fr">🇫🇷 FR</SelectItem>
   351→   351→                          <SelectItem value="en">🇬🇧 EN</SelectItem>
   352→   352→                        </SelectContent>
   353→   353→                      </Select>

   355→   355→                      <Badge variant="outline" className="text-xs">
