   156→function formatDate(dateStr: string | null): string {
   157→  if (!dateStr) return '—';
   158→  try {
   159→    return format(new Date(dateStr), 'dd MMM yyyy', { locale: fr });
   160→  } catch {
   161→    return '—';
   162→  }
   163→}

   209→function CategoryManager({ categories, onRefresh }: { categories: CategoryItem[]; onRefresh: () => void }) {
   210→  const [newName, setNewName] = useState('');
   211→  const [newIcon, setNewIcon] = useState('🏷️');
   212→  const [newLanguage, setNewLanguage] = useState('fr');
   213→  const [creating, setCreating] = useState(false);
   214→
   215→  // Edit state
   216→  const [editingId, setEditingId] = useState<string | null>(null);
   217→  const [editName, setEditName] = useState('');
   218→  const [editIcon, setEditIcon] = useState('');
   219→  const [editLanguage, setEditLanguage] = useState('');
   220→  const [saving, setSaving] = useState(false);
   221→
   222→  // Delete state
   223→  const [deleteTarget, setDeleteTarget] = useState<CategoryItem | null>(null);
   224→  const [deleting, setDeleting] = useState(false);
   225→
   226→  const handleCreate = async () => {
   227→    if (!newName.trim()) return;
   228→    setCreating(true);
   229→    try {
   230→      const res = await fetch('/api/categories', {
   231→        method: 'POST',
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify({ name: newName.trim(), icon: newIcon, language: newLanguage }),
   234→      });
   235→      if (!res.ok) {
   236→        const err = await res.json().catch(() => ({}));
   237→        throw new Error((err as { error?: string }).error || 'Erreur');
   238→      }
   239→      toast.success(`Catégorie « ${newName.trim()} » créée`);
   240→      setNewName('');
   241→      setNewIcon('🏷️');
   242→      onRefresh();
   243→    } catch (err) {
   244→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   245→    } finally {
   246→      setCreating(false);
   247→    }
   248→  };
   249→
   250→  const startEdit = (cat: CategoryItem) => {
   251→    setEditingId(cat.id);
   252→    setEditName(cat.name);
   253→    setEditIcon(cat.icon || '🏷️');
   254→    setEditLanguage(cat.language);
   255→  };
   256→
   257→  const handleSaveEdit = async () => {
   258→    if (!editingId || !editName.trim()) return;
   259→    setSaving(true);
   260→    try {
   261→      const res = await fetch('/api/categories', {
   262→        method: 'PUT',
   263→        headers: { 'Content-Type': 'application/json' },
   264→        body: JSON.stringify({ id: editingId, name: editName.trim(), icon: editIcon, language: editLanguage }),
   265→      });
   266→      if (!res.ok) {
   267→        const err = await res.json().catch(() => ({}));
   268→        throw new Error((err as { error?: string }).error || 'Erreur');
   269→      }
   270→      toast.success('Catégorie mise à jour');
   271→      setEditingId(null);
   272→      onRefresh();
   273→    } catch (err) {
   274→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   275→    } finally {
   276→      setSaving(false);
   277→    }
   278→  };
   279→
   280→  const handleDelete = async () => {
   281→    if (!deleteTarget) return;
   282→    setDeleting(true);
   283→    try {
   284→      const res = await fetch(`/api/categories?id=${deleteTarget.id}`, { method: 'DELETE' });
   285→      if (!res.ok) throw new Error('Erreur lors de la suppression');
   286→      toast.success(`Catégorie « ${deleteTarget.name} » supprimée`);
   287→      setDeleteTarget(null);
   288→      onRefresh();
   289→    } catch (err) {
   290→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue');
   291→    } finally {
   292→      setDeleting(false);
   293→    }
   294→  };
   295→
   296→  return (
   297→    <>
   298→      {/* Create new category row */}
   299→      <div className="flex flex-col sm:flex-row items-start sm:items-end gap-2 mb-4">
   300→        <div className="w-16 sm:w-20 space-y-1">
   301→          <Label className="text-xs text-muted-foreground">Icône</Label>
   302→          <Input
   303→            placeholder="🏷️"
   304→            value={newIcon}
   305→            onChange={(e) => setNewIcon(e.target.value)}
   306→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   307→            className="h-8 text-sm text-center"
   308→            maxLength={4}
   309→          />
   310→        </div>
   311→        <div className="flex-1 w-full sm:w-auto space-y-1">
   312→          <Label className="text-xs text-muted-foreground">Nom de la catégorie</Label>
   313→          <Input
   314→            placeholder="Ex: Animaux, Sciences..."
   315→            value={newName}
   316→            onChange={(e) => setNewName(e.target.value)}
   317→            onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   318→            className="h-8 text-sm"
   319→          />
   320→        </div>
   321→        <div className="w-full sm:w-28 space-y-1">
   322→          <Label className="text-xs text-muted-foreground">Langue</Label>
   323→          <Select value={newLanguage} onValueChange={setNewLanguage}>
   324→            <SelectTrigger className="h-8 text-sm">
   325→              <SelectValue />
   326→            </SelectTrigger>
   327→            <SelectContent>
   328→              <SelectItem value="fr">🇫🇷 Français</SelectItem>
   329→              <SelectItem value="en">🇬🇧 English</SelectItem>
   330→            </SelectContent>
   331→          </Select>
   332→        </div>
   333→        <Button size="sm" className="h-8 text-xs gap-1.5 shrink-0" onClick={handleCreate} disabled={creating || !newName.trim()}>
   334→          {creating ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   335→          Ajouter
   336→        </Button>
   337→      </div>
   338→
   339→      {/* Category list */}
   340→      {categories.length === 0 ? (
   341→        <div className="text-center py-6 text-muted-foreground text-sm">
   342→          Aucune catégorie. Créez-en une ci-dessus.
   343→        </div>
   344→      ) : (
   345→        <div className="rounded-md border max-h-64 overflow-y-auto">
   346→          <Table>
   347→            <TableHeader>
   348→              <TableRow className="hover:bg-transparent">
   349→                <TableHead className="pl-3">Nom</TableHead>
   350→                <TableHead>Langue</TableHead>
   351→                <TableHead>Puzzles</TableHead>
   352→                <TableHead className="text-right pr-3">Actions</TableHead>
   353→              </TableRow>
   354→            </TableHeader>
   355→            <TableBody>
   356→              {categories.map((cat) => (
   357→                <TableRow key={cat.id}>
   358→                  <TableCell className="font-medium pl-3">
   359→                    {editingId === cat.id ? (
   360→                      <div className="flex items-center gap-1.5">
   361→                        <Input
   362→                          value={editIcon}
   363→                          onChange={(e) => setEditIcon(e.target.value)}
   364→                          className="h-7 text-sm w-12 text-center"
   365→                          maxLength={4}
   366→                          autoFocus
   367→                        />
   368→                        <Input
   369→                          value={editName}
   370→                          onChange={(e) => setEditName(e.target.value)}
   371→                          onKeyDown={(e) => e.key === 'Enter' && handleSaveEdit()}
   372→                          className="h-7 text-sm w-full"
   373→                        />
   374→                      </div>
   375→                    ) : (
   376→                      <span className="flex items-center gap-1.5">
   377→                        <span className="text-sm">{cat.icon || '🏷️'}</span>
   378→                        {cat.name}
   379→                      </span>
   380→                    )}
   381→                  </TableCell>
   382→                  <TableCell>
   383→                    {editingId === cat.id ? (
   384→                      <Select value={editLanguage} onValueChange={setEditLanguage}>
   385→                        <SelectTrigger className="h-7 text-xs w-24">
   386→                          <SelectValue />
   387→                        </SelectTrigger>
   388→                        <SelectContent>
   389→                          <SelectItem value="fr">🇫🇷 FR</SelectItem>
   390→                          <SelectItem value="en">🇬🇧 EN</SelectItem>
   391→                        </SelectContent>
   392→                      </Select>
   393→                    ) : (
   394→                      <Badge variant="outline" className="text-xs">
   395→                        {languageLabel(cat.language)}
   396→                      </Badge>
   397→                    )}
   398→                  </TableCell>
   399→                  <TableCell className="text-muted-foreground text-xs">
   400→                    {cat._count.puzzles}
   401→                  </TableCell>
   402→                  <TableCell className="text-right pr-3">
   403→                    <div className="flex items-center justify-end gap-1">
   404→                      {editingId === cat.id ? (
   405→                        <>
   406→                          <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={() => setEditingId(null)}>
   407→                            Annuler
   408→                          </Button>
   409→                          <Button size="sm" className="h-7 text-xs gap-1" onClick={handleSaveEdit} disabled={saving}>
   410→                            {saving ? <Loader2 className="size-3 animate-spin" /> : <Save className="size-3" />}
   411→                            OK
   412→                          </Button>
   413→                        </>
   414→                      ) : (
   415→                        <>
   416→                          <Button
   417→                            variant="ghost"
   418→                            size="sm"
   419→                            className="h-7 text-xs gap-1"
   420→                            onClick={() => startEdit(cat)}
   421→                          >
   422→                            <Pencil className="size-3" />
   423→                          </Button>
   424→                          <Button
   425→                            variant="ghost"
   426→                            size="sm"
   427→                            className="h-7 text-xs gap-1 text-destructive hover:text-destructive"
   428→                            onClick={() => setDeleteTarget(cat)}
   429→                          >
   430→                            <Trash2 className="size-3" />
   431→                          </Button>
   432→                        </>
   433→                      )}
   434→                    </div>
   435→                  </TableCell>
   436→                </TableRow>
   437→              ))}
   438→            </TableBody>
   439→          </Table>
   440→        </div>
   441→      )}
   442→
   443→      {/* Delete confirmation */}
   444→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   445→        <AlertDialogContent>
   446→          <AlertDialogHeader>
   447→            <AlertDialogTitle>Supprimer la catégorie</AlertDialogTitle>
   448→            <AlertDialogDescription>
   449→              Êtes-vous sûr de vouloir supprimer « {deleteTarget?.name} » ? Les puzzles associés ne seront pas supprimés mais seront décatégorisés.
   450→            </AlertDialogDescription>
   451→          </AlertDialogHeader>
   452→          <AlertDialogFooter>
   453→            <AlertDialogCancel disabled={deleting}>Annuler</AlertDialogCancel>
   454→            <AlertDialogAction
   455→              onClick={handleDelete}
   456→              disabled={deleting}
   457→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   458→            >
   459→              {deleting ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   460→              Supprimer
   461→            </AlertDialogAction>
   462→          </AlertDialogFooter>
   463→        </AlertDialogContent>
   464→      </AlertDialog>
   465→    </>
   466→  );
   467→}
