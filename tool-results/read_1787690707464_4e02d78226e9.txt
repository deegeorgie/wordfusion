   166→// ── Streak Heatmap ──────────────────────────────────────────────

   168→function StreakHeatmap({ streakData }: { streakData: StreakCompletion[] }) {
   169→  const days = useMemo(() => {
   170→    const today = new Date();
   171→    const result: { date: string; completed: boolean; count: number }[] = [];
   172→    for (let i = 29; i >= 0; i--) {
   173→      const d = new Date(today);
   174→      d.setDate(d.getDate() - i);
   175→      const dateStr = d.toISOString().slice(0, 10);
   176→      const completions = streakData.filter((c) => c.date === dateStr);
   177→      result.push({
   178→        date: dateStr,
   179→        completed: completions.length > 0,
   180→        count: completions.length,
   181→      });
   182→    }
   183→    return result;
   184→  }, [streakData]);
   185→
   186→  if (streakData.length === 0) return null;
   187→
   188→  return (
   189→    <div className="flex flex-wrap gap-1">
   190→      {days.map((day) => (
   191→        <div
   192→          key={day.date}
   193→          title={`${day.date}${day.completed ? ` — ${day.count} puzzle${day.count > 1 ? 's' : ''}` : ''}`}
   194→          className={cn(
   195→            'h-5 w-5 rounded-sm transition-colors',
   196→            day.completed
   197→              ? 'bg-emerald-500 dark:bg-emerald-400'
   198→              : day.date <= new Date().toISOString().slice(0, 10)
   199→                ? 'bg-muted'
   200→                : 'bg-transparent',
   201→          )}
   202→        />
   203→      ))}
   204→    </div>
   205→  );
   206→}

   210→function ConfettiOverlay() {
   211→  const particles = useMemo(() => {
   212→    const colors = ['#10b981', '#f59e0b', '#ef4444', '#3b82f6', '#8b5cf6', '#ec4899', '#14b8a6'];
   213→    return Array.from({ length: 50 }, (_, i) => ({
   214→      id: i,
   215→      x: Math.random() * 100,
   216→      y: Math.random() * -50 - 10,
   217→      color: colors[i % colors.length],
   218→      size: Math.random() * 8 + 4,
   219→      delay: Math.random() * 1.2,
   220→      duration: Math.random() * 2 + 2.5,
   221→    }));
   222→  }, []);
   223→
   224→  return (
   225→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden">
   226→      {particles.map((p) => (
   227→        <motion.div
   228→          key={p.id}
   229→          initial={{ x: `${p.x}vw`, y: `${p.y}vh`, opacity: 1, scale: 1 }}
   230→          animate={{ y: '110vh', opacity: 0, rotate: 540 }}
   231→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeIn' }}
   232→          className="absolute rounded-sm"
   233→          style={{ width: p.size, height: p.size, backgroundColor: p.color }}
   234→        />
   235→      ))}
   236→    </div>
   237→  );
   238→}
